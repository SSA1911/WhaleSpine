// ==== /z/ai-models/rs-v131wt/src/backend.rs
//! Backend seam: device memory as `Buf` (an address plus whatever keeps the memory alive) and the kernel ops the
//! model runs. Every op takes device addresses, so model code never sees backend types. `cuda` and `vulkan` implement it.

use crate::quant::GgmlType;
use crate::weights::Tensor;
use std::any::Any;

/// Blocks per head in decode attention (positions split across them). Scratch sizes depend on it.
pub const ATT_SPLITS: usize = 32;
/// Splits of the one-head-per-block decode attention used below ATT_GQA_MIN positions.
pub const ATT_SPLITS_1: usize = 16;
/// Positions from which decode attention shares K/V loads across a GQA group (break-even measured near 2K).
pub const ATT_GQA_MIN: i32 = 2048;
const _: () = assert!(ATT_SPLITS <= 256); // attn_merge keeps one weight per split in a 256-thread block
/// Positions per GEMM tile. Prefill cuts its tile lists at this size.
pub const GEMM_BM: usize = 128;

/// `len` bytes of device memory at device address `addr`, freed when the last clone drops.
#[derive(Clone)]
pub struct Buf {
    addr: u64,
    len: usize,
    _owner: std::sync::Arc<dyn Any + Send + Sync>,
}

impl Buf {
    pub fn new(addr: u64, len: usize, owner: impl Any + Send + Sync) -> Buf { Buf { addr, len, _owner: std::sync::Arc::new(owner) } }
    pub fn addr(&self) -> u64 { self.addr }
    pub fn len(&self) -> usize { self.len }
}

#[derive(Clone)]
pub struct GpuTensor {
    pub ty: GgmlType,
    pub rows: usize,
    pub cols: usize,
    pub ptr: u64,
    pub scale: f32,
    _buf: Buf,
}

impl GpuTensor {
    /// `rows` x `cols` of `ty` stored in `buf` from its start.
    pub fn new(ty: GgmlType, rows: usize, cols: usize, buf: Buf) -> GpuTensor { GpuTensor { ty, rows, cols, ptr: buf.addr, scale: 1.0, _buf: buf } }
    pub fn with_scale(mut self, scale: f32) -> GpuTensor { self.scale = scale; self }
    pub fn bytes(&self) -> usize { self.ty.row_bytes(self.cols as u64) * self.rows }
}

/// Values copied to and from device memory as raw bytes.
/// Safety: no padding bytes, and every bit pattern is a valid value.
pub unsafe trait Plain: Copy + Default + 'static {}
unsafe impl Plain for f32 {}
unsafe impl Plain for i32 {}
unsafe impl Plain for u32 {}
unsafe impl Plain for u16 {}
unsafe impl Plain for u8 {}
unsafe impl Plain for u64 {}

/// Device memory, recorded graphs and kernel ops. Host copies are synchronous; device work runs in queue order.
/// Pointer arguments are device addresses; strides are in floats unless named otherwise.
/// The backend `kind` ("cuda", "vulkan" or "hip") on GPU `device`; with no kind, CUDA if it loads, else Vulkan.
/// Refuses a `--device` list with several indices on the HIP backend: `hip::Gpu` opens one device per process
/// (hipSetDevice runs once, in `Gpu::new`), so a split over two cards would run both layers on the first one.
pub fn check_devices(kind: Option<&str>, devices: &[usize]) -> anyhow::Result<()> {
    if kind == Some("hip") && devices.len() > 1 {
        anyhow::bail!("the hip backend runs one device: --device {} needs cuda or vulkan", devices.iter().map(|d| d.to_string()).collect::<Vec<_>>().join(","));
    }
    Ok(())
}

/// Opens every device of a `--device` list, after the check above. Callers open a list through this, not one device at a time.
pub fn open_all(kind: Option<&str>, devices: &[usize]) -> anyhow::Result<Vec<Box<dyn Backend>>> {
    check_devices(kind, devices)?;
    devices.iter().map(|&d| open(kind, d)).collect()
}

pub fn open(kind: Option<&str>, device: usize) -> anyhow::Result<Box<dyn Backend>> {
    match kind {
        Some("cuda") => Ok(Box::new(crate::cuda::Gpu::new(device)?)),
        Some("vulkan") => Ok(Box::new(crate::vulkan::Vk::new(device)?)),
        Some("hip") => Ok(Box::new(crate::hip::Gpu::new(device)?)),
        Some(k) => anyhow::bail!("unknown backend {k:?} (cuda, vulkan or hip)"),
        None => match crate::cuda::Gpu::new(device) {
            Ok(g) => Ok(Box::new(g)),
            Err(c) if format!("{c}") == "CUDA init" => {
                let v = crate::vulkan::Vk::new(device).map_err(|v| anyhow::anyhow!("no GPU backend loads: CUDA: {c:#}; Vulkan: {v:#}"))?;
                eprintln!("CUDA unavailable ({c:#}), using Vulkan");
                Ok(Box::new(v))
            }
            Err(c) => Err(c),
        },
    }
}

/// Which usable Vulkan device each GPU index opens, so that an index is the same card on both backends. `cuda` is the
/// identifier of every CUDA device in CUDA's order (`None`: the driver gives none), `vk` the one of every usable
/// Vulkan device in the order Vulkan alone would count them. Index i is the position in `vk` of CUDA device i's
/// identifier, `None` when no usable Vulkan device has it. A CUDA device without identifier takes the next Vulkan
/// device that no CUDA device names, and the rest of those follow the CUDA devices: with no CUDA device, or none
/// that Vulkan knows, the order is `vk`'s own.
pub fn device_order<T: PartialEq>(cuda: &[Option<T>], vk: &[T]) -> Vec<Option<usize>> {
    let mut rest = (0..vk.len()).filter(|&i| !cuda.iter().flatten().any(|c| *c == vk[i]));
    let mut order: Vec<_> = cuda.iter().map(|c| match c { Some(c) => vk.iter().position(|v| v == c), None => rest.next() }).collect();
    order.extend(rest.map(Some));
    order
}

/// `gemm` `act` flag: x is unchanged since the previous gemm call, which read the same x, so its int8 copy is reused.
pub const SAME_X: i32 = 4;
/// `gemm_dense` `act` flag: the call is a step of a batch of requests (`step_batch`), so the backend may take a kernel
/// that sums the products in another order than the prefill GEMM. Paths of a single request leave it clear: their bits
/// do not change.
pub const BATCHED: i32 = 8;

/// The weight type `Backend::upload` stores a tensor of type `ty` as: F16 as F32 (no F16 kernels; F32 holds every F16
/// value exactly). Expert tensors are copied raw by `GpuModel::load_part`, not through `upload`.
pub fn upload_type(ty: GgmlType) -> GgmlType { if ty == GgmlType::F16 { GgmlType::F32 } else { ty } }

pub trait Backend: Send {
    /// "cuda", "vulkan" or "hip".
    fn name(&self) -> &'static str;
    /// The GPU's name as the driver reports it.
    fn device(&self) -> String;
    /// Create a lightweight handle sharing the same underlying GPU device/context for an auxiliary draft model.
    fn share(&self) -> anyhow::Result<Box<dyn Backend>> {
        anyhow::bail!("backend {} does not support device sharing", self.name())
    }
    /// Zeroed buffer of `bytes` (at least 1).
    fn alloc(&self, bytes: usize) -> anyhow::Result<Buf>;
    fn htod(&self, src: &[u8], dst: &Buf, off: usize) -> anyhow::Result<()>;
    fn dtoh(&self, src: &Buf, off: usize, dst: &mut [u8]) -> anyhow::Result<()>;
    fn dtod(&self, src: &Buf, src_off: usize, dst: &Buf, dst_off: usize, bytes: usize) -> anyhow::Result<()>;
    /// Zero all of `dst`.
    fn memset(&self, dst: &Buf) -> anyhow::Result<()>;
    /// Free and total device memory in bytes.
    fn mem_info(&self) -> anyhow::Result<(usize, usize)>;
    /// Wait for all queued device work.
    fn sync(&self) -> anyhow::Result<()>;
    /// Run the ops `f` queues as the graph recorded under `key`, recording it on first use. `f` runs only when
    /// recording, so its ops must read per-call values from device memory.
    fn graph(&self, key: u64, f: &mut dyn FnMut() -> anyhow::Result<()>) -> anyhow::Result<()>;
    /// Free the graph recorded under `key`, if there is one. For a caller whose keys change as it runs (a graph holds
    /// device memory, and CUDA stops recording at a fixed number of them).
    fn graph_drop(&self, _key: u64) -> anyhow::Result<()> { Ok(()) }
    /// Queue the ops of `f`, none of which reads what another writes, with no ordering among them (Vulkan: no
    /// barriers between them); ops after the batch see all their results.
    fn batch(&self, f: &mut dyn FnMut() -> anyhow::Result<()>) -> anyhow::Result<()> { f() }
    /// Queue the ops of `f` in order, its host copies beside its other ops (CUDA: on a side stream), so none of
    /// those ops may read what a copy writes. The copies see the ops before `f`. Ops after `f` see the copies of the
    /// `beside` before this one, not yet these: they run on beside the next `beside` too (`beside_join` waits).
    fn beside(&self, f: &mut dyn FnMut() -> anyhow::Result<()>) -> anyhow::Result<()> { f() }
    /// Ops after this see the copies of every `beside` so far.
    fn beside_join(&self) -> anyhow::Result<()> { Ok(()) }
    /// Submit the ops queued so far without waiting. Long prefills call it per layer: amdgpu resets the GPU when one
    /// submission runs over its timeout (10 s).
    fn kick(&self) -> anyhow::Result<()> { Ok(()) }
    /// Whether `beside` really overlaps its copies with its other ops (else it queues them in order).
    fn beside_overlaps(&self) -> bool { false }
    /// Whether a MoE combine gave up waiting for the CPU worker since the last call (checked after a wait). Its layer
    /// added nothing and the rest of that submission was skipped: the caller waits for the worker and runs the
    /// forward again from that combine. Clears the flag. Backends whose combines never give up return false.
    fn stalled(&self) -> bool { false }
    /// Whether the prefill GEMM for `ty` runs on int8 tensor cores on this backend: the type policy of `Gpu::mmq`,
    /// gated on the device. Tests build their reference math from this, since the CUDA launcher takes the f16
    /// `gemm_*` path where the arch has no int8 tensor cores (sm_70) even for an int8 type.
    fn mmq(&self, ty: GgmlType) -> bool { crate::cuda::Gpu::mmq(ty) }
    /// Whether this backend runs the Volta (sm_70) CUDA fallbacks, whose kernels sum the same f16 products in
    /// another order than `mma.sync`. A cross-backend comparison against or from such a backend is by tolerance.
    fn volta(&self) -> bool { false }
    /// Zeroed host memory of `words` u32 that kernels read and write at the `Buf` address. The host pointer is
    /// valid while the `Buf` lives.
    fn host_mailbox(&self, words: usize) -> anyhow::Result<(Buf, *mut u32)>;

    /// Refuse, before any weight is uploaded, if a pipeline these weight types (`GpuModel`'s upload types) need would
    /// exceed the device's shared-memory limit; `verbose` also lists each checked (kernel, specialization) pair.
    /// Vulkan computes it from the committed SPIR-V and the specialization constants; backends without a Workgroup
    /// shared-memory limit have nothing to check.
    fn preflight(&self, _types: &[GgmlType], _verbose: bool) -> anyhow::Result<()> { Ok(()) }

    fn upload(&self, t: &Tensor) -> anyhow::Result<GpuTensor> {
        let gt = match upload_type(t.ty) {
            GgmlType::F32 if t.ty == GgmlType::F16 => {
                let mut f = vec![0f32; t.rows * t.cols];
                crate::quant::dequant_row(GgmlType::F16, t.data, &mut f);
                let b = self.alloc(4 * f.len())?;
                self.htod(&f.iter().flat_map(|v| v.to_le_bytes()).collect::<Vec<u8>>(), &b, 0)?;
                GpuTensor::new(GgmlType::F32, t.rows, t.cols, b)
            }
            ty => {
                let b = self.alloc(t.data.len())?;
                self.htod(t.data, &b, 0)?;
                GpuTensor::new(ty, t.rows, t.cols, b)
            }
        };
        Ok(gt.with_scale(t.scale))
    }

    fn upload_raw(&self, t: &Tensor) -> anyhow::Result<GpuTensor> {
        let b = self.alloc(t.data.len())?;
        self.htod(t.data, &b, 0)?;
        Ok(GpuTensor::new(t.ty, t.rows, t.cols, b).with_scale(t.scale))
    }

    /// u16 words of a KV cache for `n` values: f16, or q8 with one f16 scale per 32 values.
    fn kv_words(&self, n: usize) -> usize;
    /// GEMM over a tile list: for each tile, positions m0 .. m0 + cnt get y[m, 0..rows] = W_tile x[xrow[m]] (x row m
    /// when `xrow` is 0). `tiles` is a device array of [w, m0 | cnt << 32]. `act`: 0 stores, 1 computes
    /// y = silu(y) * result (SwiGLU with the gate already in y), 2 adds to y; `| SAME_X` as below. `n_x`: rows of x
    /// the tiles read.
    fn gemm(&self, ty: GgmlType, rows: usize, cols: usize, tiles: u64, n_tiles: usize, x: u64, x_stride: usize, n_x: usize, xrow: u64, y: u64, y_stride: usize, act: i32) -> anyhow::Result<()>;
    /// Whether `gemm` and `gemm_dense` run on int8 tensor cores for `ty` (the alternative is f16 products), so tests
    /// and callers build their reference the way this backend will.
    fn int8_gemm(&self, ty: GgmlType) -> bool;
    /// y[m] (op) W x[m] for positions m < n, x and y rows `x_stride` / `y_stride` floats apart.
    fn gemm_dense(&self, w: &GpuTensor, n: usize, x: u64, x_stride: usize, y: u64, y_stride: usize, act: i32) -> anyhow::Result<()>;
    fn prequant_gemm_x(&self, _w: &GpuTensor, _n: usize, _x: u64, _x_stride: usize) -> anyhow::Result<()> { Ok(()) }
    /// y = W x (all rows of `w`).
    fn gemv(&self, w: &GpuTensor, x: u64, y: u64) -> anyhow::Result<()>;
    /// y[t] = W x[t] for t < nt (1 to 4; x[t] at x + 4 * t * x_stride bytes, y[t] at y + 4 * t * y_stride); each
    /// weight block is read once for all nt.
    fn gemvn(&self, w: &GpuTensor, x: u64, x_stride: usize, y: u64, y_stride: usize, nt: usize) -> anyhow::Result<()>;
    /// `gemvn` that reads x pre-quantized by `quant_xq` at `xq` (0 = quantize x itself; float weights read x).
    fn gemvq(&self, w: &GpuTensor, x: u64, x_stride: usize, xq: u64, y: u64, y_stride: usize, nt: usize) -> anyhow::Result<()>;
    /// `gemvq` of `a` into `ya` and of `b` into `yb` on the same x (columns alike; output rows `a.rows` / `b.rows` apart).
    fn gemvq2(&self, a: &GpuTensor, b: &GpuTensor, x: u64, x_stride: usize, xq: u64, ya: u64, yb: u64, nt: usize) -> anyhow::Result<()> {
        self.gemvq(a, x, x_stride, xq, ya, a.rows, nt)?;
        self.gemvq(b, x, x_stride, xq, yb, b.rows, nt)
    }
    /// Whether this backend supports indexed-row GEMV (`gemv_rows`).
    fn supports_gemv_rows(&self) -> bool { false }
    /// `y[i] = W[ids[i]] x` for `i < n_rows` (indexed subset of rows of `w`).
    fn gemv_rows(&self, _w: &GpuTensor, _x: u64, _ids: u64, _n_rows: usize, _y: u64) -> anyhow::Result<()> {
        anyhow::bail!("gemv_rows unsupported on this backend")
    }
    /// `y[t * n_rows + i] = W[ids[i]] x[t]` for `t < nt`, `i < n_rows`.
    fn gemv_rows_n(&self, w: &GpuTensor, x: u64, x_stride: usize, _xq: u64, ids: u64, n_rows: usize, y: u64, nt: usize) -> anyhow::Result<()> {
        for t in 0..nt {
            self.gemv_rows(w, x + (t * x_stride * 4) as u64, ids, n_rows, y + (t * n_rows * 4) as u64)?;
        }
        Ok(())
    }
    /// For j < k with ids[j] >= 0: y[j] = W_j x[j / xg], where W_j is the `rows` x `cols` matrix at w + ids[j] * stride
    /// bytes. `pair` = (w2, y2) also computes the same product for the matrices at byte offset w2 into the output at
    /// byte offset y2. `xq` (not 0): x pre-quantized by `quant_xq` (k / xg rows), read in place of x.
    fn moe_gemv(&self, ty: GgmlType, rows: usize, cols: usize, w: u64, stride: usize, ids: u64, k: usize, x: u64, x_stride: usize, xg: usize, y: u64, pair: Option<(i64, i64)>, xq: u64) -> anyhow::Result<()>;
    /// out = the first `rows` rows of `w` as floats (a test helper).
    fn dequant(&self, w: &GpuTensor, rows: usize, out: u64) -> anyhow::Result<()>;
    /// Quantize `nt` rows of n floats (row t at x + 4 * t * x_stride bytes) for `gemvq` / `moe_gemv` into `out`:
    /// the nt rows of n int8 values, then their f32 scales (one per 32 values), nt * (n + n / 8) bytes in all.
    fn quant_xq(&self, x: u64, x_stride: usize, n: usize, nt: usize, out: u64) -> anyhow::Result<()>;
    /// `rows` rows: out[r] = rmsnorm(x[r]) * w.
    fn rms_norm(&self, x: u64, x_stride: usize, w: u64, out: u64, out_stride: usize, n: usize, rows: usize, eps: f32) -> anyhow::Result<()>;
    /// `rms_norm` that also writes its output quantized as `quant_xq` does to `xq` (when not 0), and first adds `res`
    /// (rows `x_stride` apart, when not 0) into x in place.
    fn rms_norm_q(&self, x: u64, x_stride: usize, w: u64, out: u64, out_stride: usize, n: usize, rows: usize, eps: f32, xq: u64, res: u64) -> anyhow::Result<()>;
    /// Per-stream RMSNorm of nt stacks of hc rows of n (contiguous) with the hc x n gamma `w`.
    fn rms_norm_streams(&self, x: u64, w: u64, out: u64, n: usize, hc: usize, nt: usize, eps: f32) -> anyhow::Result<()> {
        for s in 0..hc {
            let o = 4 * (s * n) as u64;
            self.rms_norm(x + o, hc * n, w + o, out + o, hc * n, n, nt, eps)?;
        }
        Ok(())
    }
    /// out[r] = argmax of row r of x (`rows` rows of n floats), first index on ties.
    fn argmax_rows(&self, x: u64, n: usize, rows: usize, out: u64) -> anyhow::Result<()>;
    /// Row r: vector r % per (n floats) of token r / per, tokens `stride` floats apart.
    fn l2_norm(&self, x: u64, n: usize, per: usize, stride: usize, rows: usize, eps: f32) -> anyhow::Result<()>;
    /// `heads` rows of `stride` floats, `per` per token; token t is at position step[0] + t.
    fn rope_neox(&self, x: u64, stride: usize, heads: usize, per: usize, n_rot: usize, step: u64, base: f32) -> anyhow::Result<()> {
        self.rope_pairs(x, stride, heads, per, n_rot, step, base, 0, n_rot / 2)
    }
    /// `rope_neox` of pairs `lo..hi` only (M-RoPE: one call per section, each with its own position buffer).
    fn rope_pairs(&self, x: u64, stride: usize, heads: usize, per: usize, n_rot: usize, step: u64, base: f32, lo: usize, hi: usize) -> anyhow::Result<()>;
    /// M-RoPE (`ops::rope_mrope`): section s of the pairs turns by position `pos[2 * s]` (`pos`: four [position, _]
    /// i32 pairs); pairs past the sections use the last position.
    fn rope_mrope(&self, x: u64, stride: usize, heads: usize, per: usize, n_rot: usize, pos: u64, base: f32, sections: [usize; 4]) -> anyhow::Result<()> {
        let (half, mut lo) = (n_rot / 2, 0);
        for (s, n) in sections.into_iter().enumerate() {
            let hi = if s == 3 { half } else { (lo + n).min(half) };
            if hi > lo { self.rope_pairs(x, stride, heads, per, n_rot, pos + 8 * s as u64, base, lo, hi)?; }
            lo = hi;
        }
        Ok(())
    }
    /// `rms_norm` of `rows` heads of n (output rows of n), then `rope_neox` on them.
    fn rms_norm_rope(&self, x: u64, x_stride: usize, w: u64, out: u64, n: usize, rows: usize, eps: f32, per: usize, n_rot: usize, step: u64, base: f32) -> anyhow::Result<()> {
        self.rms_norm(x, x_stride, w, out, n, n, rows, eps)?;
        self.rope_neox(out, n, rows, per, n_rot, step, base)
    }
    /// `rms_norm_rope` of the k heads (`per` per token), then `kv_store` of them and v (rows of n, like out).
    fn rms_norm_rope_kv(&self, x: u64, x_stride: usize, w: u64, out: u64, n: usize, rows: usize, eps: f32, per: usize, n_rot: usize, step: u64, base: f32, v: u64, kc: u64, vc: u64) -> anyhow::Result<()>;
    /// `rms_norm_rope` of the q heads (`qrows` rows, `qper` per token), then `rms_norm_rope_kv` of the k heads (rows of n).
    fn rms_norm_rope_qk(&self, q: u64, q_stride: usize, qw: u64, qo: u64, qrows: usize, qper: usize, k: u64, kw: u64, ko: u64, krows: usize, kper: usize,
                        n: usize, eps: f32, n_rot: usize, step: u64, base: f32, v: u64, kc: u64, vc: u64) -> anyhow::Result<()> {
        self.rms_norm_rope(q, q_stride, qw, qo, n, qrows, eps, qper, n_rot, step, base)?;
        self.rms_norm_rope_kv(k, n, kw, ko, n, krows, eps, kper, n_rot, step, base, v, kc, vc)
    }
    /// x += y.
    fn add(&self, x: u64, y: u64, n: usize) -> anyhow::Result<()>;
    /// g = silu(g) * u.
    fn swiglu(&self, g: u64, u: u64, n: usize) -> anyhow::Result<()>;
    /// Beta*tanh(gate/beta) * sigmoid(gate) * lb*tanh(up/lb) as in `situ`, directly quantized as `quant_xq` into `xq`.
    fn situ_q(&self, _y: u64, _n: usize, _rows: usize, _scale: u64, _beta: f32, _lb: f32, _xq: u64) -> anyhow::Result<()> { anyhow::bail!("{}: situ_q not implemented", self.name()) }
    /// `x[i] = cap * tanh(x[i] / cap)` for `i < n`.
    fn softcap(&self, _x: u64, _n: usize, _cap: f32) -> anyhow::Result<()> { anyhow::bail!("{}: softcap not implemented", self.name()) }
    /// Inkling's activation on `rows` rows of 2n floats at `y`, n gates then n ups: each gate becomes
    /// beta*tanh(gate/beta) * sigmoid(gate) * lb*tanh(up/lb) (`lb` not above 0: times the up as it is).
    fn situ(&self, _y: u64, _n: usize, _rows: usize, _scale: u64, _beta: f32, _lb: f32) -> anyhow::Result<()> { anyhow::bail!("{}: SITU not implemented", self.name()) }
    // Inkling 2's decode token and batched prefill on the device (`inkling2::device`); the host math is in `inkling2::forward`.
    /// Depth mix: `out[j]` (n floats each, j < nq) = the softmax-weighted sum of the RMS-normed states, scored by
    /// query row j of `q`. The states are the `nb` rows of `blocks` (the newest last) and `cur`.
    fn ink_mix(&self, blocks: u64, nb: usize, cur: u64, q: u64, nq: usize, out: u64, n: usize) -> anyhow::Result<()> {
        self.ink_mix_n(blocks, nb, cur, q, nq, out, n, 1)
    }
    /// Batched depth mix across `nt` tokens: `blocks` `[nb][nt][n]`, `cur` `[nt][n]`, `q` `[nq][n]`, `out` `[nq][nt][n]`.
    fn ink_mix_n(&self, _blocks: u64, _nb: usize, _cur: u64, _q: u64, _nq: usize, _out: u64, _n: usize, _nt: usize) -> anyhow::Result<()> {
        anyhow::bail!("{}: Inkling 2 device decode not implemented", self.name())
    }
    /// `count` loop controllers: `out[j] = cur_j - tanh(forget)*cur_j + tanh(gain)*msg[j]`, `cur_j` row
    /// `min(j, nins - 1)` of `cur`. `down` (`[rank][2n]`), `gain` and `forget` (`[rank][n]`) are f16, a controller
    /// after another. `gate`: `count * rank` floats of scratch.

impl<'a> dyn Backend + 'a {
    /// Zeroed buffer of `n` values of `T`.
    pub fn zeros<T: Plain>(&self, n: usize) -> anyhow::Result<Buf> { self.alloc(n * size_of::<T>()) }

    /// Copy `src` into `dst` from element `at` on.
    pub fn put<T: Plain>(&self, src: &[T], dst: &Buf, at: usize) -> anyhow::Result<()> {
        let b = unsafe { std::slice::from_raw_parts(src.as_ptr().cast::<u8>(), size_of_val(src)) };
        self.htod(b, dst, at * size_of::<T>())
    }

    /// `n` values of `T` from `src`, starting at element `at`.
    pub fn get<T: Plain>(&self, src: &Buf, at: usize, n: usize) -> anyhow::Result<Vec<T>> {
        let mut v = vec![T::default(); n];
        let b = unsafe { std::slice::from_raw_parts_mut(v.as_mut_ptr().cast::<u8>(), n * size_of::<T>()) };
        self.dtoh(src, at * size_of::<T>(), b)?;
        Ok(v)
    }
}

// ==== /z/ai-models/rs-v131wt/src/chat.rs
//! Chat templates. Renders OpenAI-style messages and tools as llama.cpp's Jinja engine renders the model's
//! `tokenizer.chat_template`, and parses tool calls out of a reply. Two template families are covered, picked by
//! markers in the template text: Qwen3.6 (`<function=..>` XML calls) and Qwen3 2507 / Qwen3-Next (JSON calls).
//! `tests/serve.rs` checks the output byte for byte against llama-server's `/apply-template`.

use anyhow::{Context, bail};
use serde::de::{Deserialize, Deserializer, MapAccess, SeqAccess, Visitor};

/// A JSON value that keeps object key order. serde_json's `Map` sorts keys without its `preserve_order` feature,
/// and templates print tools and arguments in the client's order.
#[derive(Clone, Debug, PartialEq)]
pub enum Json { Null, Bool(bool), Num(serde_json::Number), Str(String), Arr(Vec<Json>), Obj(Vec<(String, Json)>) }

static NULL: Json = Json::Null;

impl Json {
    /// Field `k` of an object; null when missing or not an object.
    pub fn get(&self, k: &str) -> &Json {
        match self { Json::Obj(o) => o.iter().find(|(x, _)| x == k).map_or(&NULL, |x| &x.1), _ => &NULL }
    }
    pub fn as_str(&self) -> Option<&str> { if let Json::Str(s) = self { Some(s) } else { None } }
    /// Jinja's `tojson` as llama.cpp prints it: `", "` and `": "` separators, non-ASCII kept, floats as C `%g`.
    pub fn dump(&self) -> String {
        let mut s = String::new();
        self.dump_to(&mut s);
        s
    }
    fn dump_to(&self, s: &mut String) {
        let key = |k: &str| serde_json::to_string(k).unwrap();
        match self {
            Json::Null => s.push_str("null"),
            Json::Bool(b) => s.push_str(if *b { "true" } else { "false" }),
            Json::Num(n) => s.push_str(&num(n)),
            Json::Str(x) => s.push_str(&key(x)),
            Json::Arr(a) => {
                s.push('[');
                for (i, v) in a.iter().enumerate() {
                    if i > 0 { s.push_str(", "); }
                    v.dump_to(s);
                }
                s.push(']');
            }
            Json::Obj(o) => {
                s.push('{');
                for (i, (k, v)) in o.iter().enumerate() {
                    if i > 0 { s.push_str(", "); }
                    s.push_str(&key(k));
                    s.push_str(": ");
                    v.dump_to(s);
                }
                s.push('}');
            }
        }
    }
    /// A tool-call parameter value as the Qwen3.6 template prints it: lists and objects as JSON, other values
    /// through Jinja's `string` filter (Python spelling for booleans and null).
    fn param(&self) -> String {
        match self {
            Json::Str(s) => s.clone(),
            Json::Bool(b) => (if *b { "True" } else { "False" }).into(),
            Json::Null => "None".into(),
            Json::Num(n) => num(n),
            _ => self.dump(),
        }
    }
}

/// Number text: integers exact, floats like C's `%g` (6 significant digits), which is what llama.cpp prints.
fn num(n: &serde_json::Number) -> String {
    let Some(f) = n.as_f64().filter(|_| n.is_f64()) else { return n.to_string() };
    if f == 0.0 { return (if f.is_sign_negative() { "-0" } else { "0" }).into(); }
    let strip = |s: &str| if s.contains('.') { s.trim_end_matches('0').trim_end_matches('.').to_string() } else { s.to_string() };
    let e = format!("{f:.5e}");
    let (m, x) = e.split_once('e').unwrap();
    let x: i32 = x.parse().unwrap();
    if (-4..6).contains(&x) { strip(&format!("{f:.*}", (5 - x) as usize)) }
    else { format!("{}e{}{:02}", strip(m), if x < 0 { '-' } else { '+' }, x.abs()) }
}

impl<'de> Deserialize<'de> for Json {
    fn deserialize<D: Deserializer<'de>>(d: D) -> Result<Json, D::Error> {
        struct V;
        impl<'de> Visitor<'de> for V {
            type Value = Json;
            fn expecting(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result { f.write_str("JSON") }
            fn visit_unit<E>(self) -> Result<Json, E> { Ok(Json::Null) }
            fn visit_bool<E>(self, b: bool) -> Result<Json, E> { Ok(Json::Bool(b)) }
            fn visit_i64<E>(self, n: i64) -> Result<Json, E> { Ok(Json::Num(n.into())) }
            fn visit_u64<E>(self, n: u64) -> Result<Json, E> { Ok(Json::Num(n.into())) }
            fn visit_f64<E: serde::de::Error>(self, f: f64) -> Result<Json, E> {
                serde_json::Number::from_f64(f).map(Json::Num).ok_or_else(|| E::custom("bad number"))
            }
            fn visit_str<E>(self, s: &str) -> Result<Json, E> { Ok(Json::Str(s.into())) }
            fn visit_string<E>(self, s: String) -> Result<Json, E> { Ok(Json::Str(s)) }
            fn visit_seq<A: SeqAccess<'de>>(self, mut a: A) -> Result<Json, A::Error> {
                let mut v = Vec::new();
                while let Some(x) = a.next_element()? { v.push(x); }
                Ok(Json::Arr(v))
            }
            fn visit_map<A: MapAccess<'de>>(self, mut a: A) -> Result<Json, A::Error> {
                let mut v: Vec<(String, Json)> = Vec::new();
                // Key positions, so a body with many keys parses in linear time.
                let mut at = std::collections::HashMap::<String, usize>::new();
                while let Some((k, x)) = a.next_entry::<String, Json>()? {
                    // A repeated key keeps its first place and its last value, as in nlohmann's ordered_json.
                    match at.get(&k) { Some(&i) => v[i].1 = x, None => { at.insert(k.clone(), v.len()); v.push((k, x)) } }
                }
                Ok(Json::Obj(v))
            }
        }
        d.deserialize_any(V)
    }
}

/// One chat message. `content` is already text (content parts joined with newlines, as llama.cpp does), with one
/// `<image>` placeholder per image in `images`.
#[derive(Clone, Debug, PartialEq)]
pub struct Msg { pub role: String, pub content: String, pub reasoning: Option<String>, pub calls: Vec<Call>, pub images: Vec<String> }

/// A tool call in an assistant message; `args` is the parsed arguments object.
#[derive(Clone, Debug, PartialEq)]
pub struct Call { pub name: String, pub args: Json }

impl Msg {
    pub fn new(role: &str, content: &str) -> Msg { Msg { role: role.into(), content: content.into(), reasoning: None, calls: vec![], images: vec![] } }

    /// Parses one OpenAI message. Roles are limited to the ones the templates know, because the role goes into the
    /// prompt verbatim.
    pub fn from_json(m: &Json) -> anyhow::Result<Msg> {
        let role = m.get("role").as_str().context("message without role")?;
        if !["system", "user", "assistant", "tool"].contains(&role) { bail!("unknown role {role:?}"); }
        let mut images = vec![];
        let content = match m.get("content") {
            Json::Str(s) => s.clone(),
            Json::Arr(parts) => {
                let mut text = String::new();
                let mut first = true;
                for p in parts {
                    let piece = match p.get("type").as_str().unwrap_or("") {
                        "text" => p.get("text").as_str().unwrap_or("").to_string(),
                        "image_url" => {
                            let u = p.get("image_url");
                            let url = u.get("url").as_str().or_else(|| u.as_str()).unwrap_or("");
                            if url.is_empty() { continue; }
                            images.push(url.to_string());
                            "<image>".to_string()
                        }
                        "image" => {
                            let src = p.get("source");
                            let url = src.get("url").as_str().map(str::to_owned).or_else(|| {
                                (src.get("type").as_str() == Some("base64")).then(|| {
                                    format!("data:{};base64,{}", src.get("media_type").as_str().unwrap_or("image/png"), src.get("data").as_str().unwrap_or(""))
                                })
                            });
                            let Some(url) = url.filter(|u| !u.is_empty() && !u.ends_with(";base64,")) else { continue };
                            images.push(url);
                            "<image>".to_string()
                        }
                        _ => continue,
                    };
                    if !first { text.push('\n'); }
                    text.push_str(&piece);
                    first = false;
                }
                text
            }
            Json::Null if role == "assistant" => String::new(),
            Json::Null => bail!("{role} message without content"),
            _ => bail!("bad content"),
        };
        // Messages come from the network: a plain path would make the server read a local file.
        if let Some(u) = images.iter().find(|u| !["data:", "http://", "https://"].iter().any(|p| u.starts_with(p))) {
            bail!("image source {u:?} is not a data: or http(s) URL");
        }
        let calls = match m.get("tool_calls") {
            Json::Null => vec![],
            Json::Arr(a) if role == "assistant" => a.iter().map(|c| {
                let f = c.get("function");
                let name = f.get("name").as_str().context("tool call without function name")?.to_string();
                // llama.cpp parses string arguments, so both families print them as JSON.
                let args = match f.get("arguments") {
                    Json::Str(s) => serde_json::from_str(s).context("tool call arguments are not valid JSON")?,
                    Json::Null => bail!("tool call without arguments"),
                    a => a.clone(),
                };
                Ok(Call { name, args })
            }).collect::<anyhow::Result<_>>()?,
            _ => bail!("tool_calls must be an array on an assistant message"),
        };
        Ok(Msg { role: role.into(), content, reasoning: m.get("reasoning_content").as_str().map(Into::into), calls, images })
    }
}

/// A tool as llama.cpp hands it to the template: the type, then the function's name, description and parameters,
/// nothing else.
pub fn tool_def(t: &Json) -> anyhow::Result<Json> {
    if !matches!(t.get("type"), Json::Null) && t.get("type").as_str() != Some("function") { bail!("tool type must be \"function\""); }
    let f = t.get("function");
    let name = f.get("name").as_str().context("tool without function name")?;
    let or = |v: &Json, d: Json| if matches!(v, Json::Null) { d } else { v.clone() };
    Ok(Json::Obj(vec![("type".into(), Json::Str("function".into())), ("function".into(), Json::Obj(vec![
        ("name".into(), Json::Str(name.into())),
        ("description".into(), or(f.get("description"), Json::Str(String::new()))),
        ("parameters".into(), or(f.get("parameters"), Json::Obj(vec![]))),
    ]))]))
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub enum Template {
    /// Qwen3.6: `<function=name><parameter=key>` calls, contents trimmed, an empty think block when thinking is off.
    Xml,
    /// Qwen3 2507 and Qwen3-Next: `{"name": .., "arguments": ..}` calls. `reasoning` (2507) puts the turns after the
    /// last user query in a think block.
    Json { reasoning: bool },
    /// DeepSeek-V4.1: `<｜User｜>`/`<｜Assistant｜>` turns, think mode with a reasoning-effort line. Plain chat only;
    // ponytail: no DSML tool calls, add when serving DeepSeek tools matters.
    Ds,
    /// Recursive Flint: `<|system|>`/`<|user|>`/`<|assistant|>`/`<|observation|>` turns and compact
    /// `<tool_call>name<arg_key>..</arg_key><arg_value>..</arg_value></tool_call>` calls.
    Flint,
}

const XML_RULES: &str = "\n\nIf you choose to call a function ONLY reply in the following format with NO suffix:\n\n<tool_call>\n<function=example_function_name>\n<parameter=example_parameter_1>\nvalue_1\n</parameter>\n<parameter=example_parameter_2>\nThis is the value for the second parameter\nthat can span\nmultiple lines\n</parameter>\n</function>\n</tool_call>\n\n<IMPORTANT>\nReminder:\n- Function calls MUST follow the specified format: an inner <function=...></function> block must be nested within <tool_call></tool_call> XML tags\n- Required parameters MUST be specified\n- You may provide optional reasoning for your function call in natural language BEFORE the function call, but NOT after\n- If there is no function call available, answer the question like normal with your current knowledge and do not tell the user about function calls\n</IMPORTANT>";
const JSON_HEAD: &str = "# Tools\n\nYou may call one or more functions to assist with the user query.\n\nYou are provided with function signatures within <tools></tools> XML tags:\n<tools>";
const JSON_TAIL: &str = "\n</tools>\n\nFor each function call, return a json object with function name and arguments within <tool_call></tool_call> XML tags:\n<tool_call>\n{\"name\": <function-name>, \"arguments\": <args-json-object>}\n</tool_call><|im_end|>\n";
const TAG: &str = "<tool_call>";

const DS_TOOLS_HEAD: &str = "## Tools\n\nYou have access to a set of tools to help answer the user's question. You can invoke tools by writing a \"<｜DSML｜ calls>\" block like the following:\n\n<｜DSML｜ calls>\n<｜DSML｜ invoke name=\"$TOOL_NAME\">\n<｜DSML｜ parameter name=\"$PARAMETER_NAME\" string=\"true|false\">$PARAMETER_VALUE</｜DSML｜ parameter>\n...\n</｜DSML｜ invoke>\n<｜DSML｜ invoke name=\"$TOOL_NAME2\">\n...\n</｜DSML｜ invoke>\n</｜DSML｜ calls>\n\nString parameters should be specified as is and set `string=\"true\"`. For all other types (numbers, booleans, arrays, objects), pass the value in JSON format and set `string=\"false\"`.\n\nIf thinking_mode is enabled (triggered by <think>), you MUST output your complete reasoning inside <think>...</think> BEFORE any tool calls or final response.\n\nOtherwise, output directly after </think> with tool calls or final response.\n\n### Available Tool Schemas\n\n";
const DS_TOOLS_TAIL: &str = "\n\nYou MUST strictly follow the above defined tool name and parameter schemas to invoke tool calls.\n";
const DS_CALLS: &str = "<｜DSML｜ calls>";

/// DeepSeek-V4.1 (Unsloth template): tool results merge into the user turn, the tools go after the system text
/// (a new empty system message when there is none), and tools anywhere keep the reasoning of earlier turns.
// ponytail: tool results print in message order; the template re-sorts them by tool_call_id (Msg keeps no ids).
fn render_ds(msgs: &[Msg], tools: &[Json], think: bool, preserve: bool, generate: bool) -> String {
    enum It<'a> { Sys(&'a str), User(Vec<String>, bool), Asst(&'a Msg) }
    let mut items: Vec<It> = vec![];
    if !tools.is_empty() && msgs.first().is_none_or(|m| m.role != "system") { items.push(It::Sys("")); }
    for m in msgs {
        match m.role.as_str() {
            "tool" => {
                let p = format!("<tool_result>{}</tool_result>", m.content);
                match items.last_mut() { Some(It::User(v, _)) => v.push(p), _ => items.push(It::User(vec![p], false)) }
            }
            "user" => match items.last_mut() {
                Some(It::User(v, real)) => { v.push(m.content.clone()); *real = true }
                _ => items.push(It::User(vec![m.content.clone()], true)),
            },
            "assistant" => items.push(It::Asst(m)),
            "system" => items.push(It::Sys(&m.content)),
            _ => {}
        }
    }
    let last_user = items.iter().rposition(|x| matches!(x, It::User(_, true)) || matches!(x, It::Sys(_))).filter(|&i| i > 0 || matches!(items[0], It::User(..))).map_or(-1, |i| i as i64);
    let keep_all = preserve || !tools.is_empty();
    let mut s = String::from("<｜begin▁of▁sentence｜>");
    for (i, it) in items.iter().enumerate() {
        if i == 0 {
            if think || matches!(it, It::Sys(_)) { s += "<｜System｜>"; }
            if think { s += "Reasoning Effort: 75 (range 1-100, the higher the value, the more thorough the reasoning)\n\n"; }
        }
        match it {
            It::Sys(c) => {
                if i > 0 { s += "<｜System｜>"; }
                s += c;
                if i == 0 && !tools.is_empty() {
                    s += "\n\n";
                    s += DS_TOOLS_HEAD;
                    s += &tools.iter().map(|t| t.get("function").dump()).collect::<Vec<_>>().join("\n");
                    s += DS_TOOLS_TAIL;
                }
            }
            It::User(v, _) => { s += "<｜User｜>"; s += &v.join("\n\n"); }
            It::Asst(m) => {
                let (r, c) = split_think(m, &m.content);
                if think && (keep_all || i as i64 > last_user) { s += r; s += "</think>"; }
                s += c;
                if !m.calls.is_empty() {
                    let inv: Vec<String> = m.calls.iter().map(|call| {
                        let params: Vec<String> = match &call.args {
                            Json::Obj(o) => o.iter().map(|(k, v)| format!("<｜DSML｜ parameter name=\"{k}\" string=\"{}\">{}</｜DSML｜ parameter>",
                                v.as_str().is_some(), v.as_str().map_or_else(|| v.dump(), String::from))).collect(),
                            _ => vec![],
                        };
                        format!("<｜DSML｜ invoke name=\"{}\">\n{}\n</｜DSML｜ invoke>", call.name, params.join("\n"))
                    }).collect();
                    s += &format!("\n\n{DS_CALLS}\n{}\n</｜DSML｜ calls>", inv.join("\n"));
                }
                s += "<｜end▁of▁sentence｜>";
            }
        }
        let last = i + 1 == items.len();
        if (last || matches!(items[i + 1], It::Asst(_))) && (matches!(it, It::User(..)) || (matches!(it, It::Sys(_)) && i > 0)) && (generate || !last) {
            s += "<｜Assistant｜>";
            s += if think && (keep_all || i as i64 >= last_user) { "<think>" } else { "</think>" };
        }
    }
    s
}

/// Recursive Flint: escape the template's own markers in user text and drop NULs and think tags.
fn flint_text(s: &str) -> String {
    let mut t = s.replace('\0', "").replace(" thinking", "").replace("</think>", "");
    for m in ["<|begin_of_text|>", "<|end_of_text|>", "<|system|>", "<|developer|>", "<|user|>", "<|assistant|>", "<|observation|>"] {
        t = t.replace(m, &m.replace('<', "&lt;").replace('>', "&gt;"));
    }
    t
}

/// The tool block Flint puts in a system turn before the first non-system message.
fn flint_preamble(tools: &[Json]) -> String {
    let mut s = String::from("<|system|>\n# Tools\n\nYou may call one or more functions to assist with the user query.\n\nYou are provided with function signatures within <tools></tools> XML tags:\n<tools>");
    for t in tools {
        s += &t.dump();
    }
    s += "</tools>\n\nFor each function call, output the function name and arguments within the following XML format:\n<tool_call>{function-name}<arg_key>{arg-key-1}</arg_key><arg_value>{arg-value-1}</arg_value>...</tool_call>";
    s
}

/// Flint's chat template: turn markers with no trailing newline, `<tool_call>` calls and an always-on think block.
fn render_flint(msgs: &[Msg], tools: &[Json], generate: bool) -> String {
    let preamble = flint_preamble(tools);
    let mut pending = !tools.is_empty();
    let mut s = String::new();
    for m in msgs {
        if pending && m.role != "system" {
            s += &preamble;
            pending = false;
        }
        match m.role.as_str() {
            "system" => { s += "<|system|>"; s += &flint_text(&m.content); }
            "developer" => { s += "<|developer|>"; s += &flint_text(&m.content); }
            "user" => { s += "<|user|>"; s += &flint_text(&m.content); }
            "assistant" => {
                s += "<|assistant|>";
                let (r, c) = split_think(m, &m.content);
                let reasoning = m.reasoning.as_deref().unwrap_or(r);
                if !reasoning.is_empty() { s += " thinking"; s += reasoning; s += "</think>"; }
                s += c;
                for call in &m.calls {
                    s += "<tool_call>";
                    s += &call.name;
                    if let Json::Obj(o) = &call.args {
                        for (k, v) in o {
                            s += "<arg_key>";
                            s += k;
                            s += "</arg_key><arg_value>";
                            match v { Json::Str(x) => s += x, _ => s += &v.dump() }
                            s += "</arg_value>";
                        }
                    }
                    s += "</tool_call>";
                }
            }
            "tool" => { s += "<|observation|><tool_response>"; s += &flint_text(&m.content); s += "</tool_response>"; }
            _ => {}
        }
    }
    if pending { s += &preamble; }
    if generate { s += "<|assistant|>"; }
    s
}

/// One `<arg_value>` as JSON text. Values that parse as JSON keep their type; anything else is a string.
fn flint_arg(raw: &str) -> String {
    serde_json::from_str::<serde_json::Value>(raw).map_or_else(|_| serde_json::to_string(raw).unwrap(), |v| v.to_string())
}

/// Flint reply calls: `<tool_call>name<arg_key>k</arg_key><arg_value>v</arg_value>...</tool_call>`.
fn flint_calls(text: &str) -> Option<(String, Vec<(String, String)>)> {
    let from = text.rfind("</think>").map_or(0, |p| p + 8);
    let start = from + text[from..].find(TAG)?;
    let mut rest = &text[start..];
    let mut calls = Vec::new();
    loop {
        rest = rest.trim_start();
        if rest.is_empty() { break; }
        let (body, tail) = rest.strip_prefix(TAG)?.split_once("</tool_call>")?;
        rest = tail;
        let at = body.find("<arg_key>").unwrap_or(body.len());
        let name = body[..at].trim();
        if name.is_empty() { return None; }
        let mut args = String::from("{");
        let mut b = &body[at..];
        let mut first = true;
        while let Some(x) = b.strip_prefix("<arg_key>") {
            let (k, x) = x.split_once("</arg_key>")?;
            let x = x.strip_prefix("<arg_value>")?;
            let (v, x) = x.split_once("</arg_value>")?;
            b = x;
            if !first { args.push(','); }
            first = false;
            args += &serde_json::to_string(k.trim()).ok()?;
            args.push(':');
            args += &flint_arg(v);
        }
        if !b.trim().is_empty() { return None; }
        args.push('}');
        calls.push((name.to_string(), args));
    }
    Some((text[..start].trim_end().to_string(), calls))
}

/// Index of the last user turn that is not a tool response (the templates' `last_query_index`).
fn last_query(msgs: &[Msg], trim: bool) -> usize {
    msgs.iter().rposition(|m| {
        let c = if trim { m.content.trim() } else { &m.content };
        m.role == "user" && !(c.starts_with("<tool_response>") && c.ends_with("</tool_response>"))
    }).unwrap_or(msgs.len().saturating_sub(1))
}

/// Reasoning and answer of an assistant turn: `reasoning_content` when given, else the `<think>` part of `c`.
fn split_think<'a>(m: &'a Msg, c: &'a str) -> (&'a str, &'a str) {
    if let Some(r) = &m.reasoning { return (r, c); }
    if !c.contains("</think>") { return ("", c); }
    let r = c.split("</think>").next().unwrap().trim_end_matches('\n').rsplit("<think>").next().unwrap();
    (r.trim_start_matches('\n'), c.rsplit("</think>").next().unwrap().trim_start_matches('\n'))
}

impl Template {
    /// The family of a GGUF's `tokenizer.chat_template`. Files without one get Qwen3.6's format.
    pub fn detect(src: Option<&str>) -> Template {
        match src {
            Some(s) if s.contains("｜DSML｜") => Template::Ds,
            Some(s) if s.contains("<|observation|>") && s.contains("<|begin_of_text|>") => Template::Flint,
            Some(s) if !s.contains("<function=") && s.contains(TAG) => Template::Json { reasoning: s.contains("reasoning_content") },
            _ => Template::Xml,
        }
    }

    /// The prompt for `msgs` and `tools` (from `tool_def`). `generate` adds the assistant header the reply follows;
    /// `think` false closes the think block at once where the template has one. `preserve` prints the think block of
    /// every earlier reply, not only of those after the last user turn (Qwen3.6's `preserve_thinking`), so a new user
    /// turn leaves the text of the earlier ones as it was and their cached state can be reused.
    pub fn render(&self, msgs: &[Msg], tools: &[Json], think: bool, preserve: bool, generate: bool) -> String {
        let mut s = String::new();
        let tool_end = |s: &mut String, i: usize| {
            if i + 1 == msgs.len() || msgs[i + 1].role != "tool" { *s += "<|im_end|>\n"; }
        };
        match *self {
            Template::Flint => s += &render_flint(msgs, tools, generate),
            Template::Ds => s += &render_ds(msgs, tools, think, preserve, generate),
            Template::Xml => {
                let (n_sys, sys) = match msgs {
                    [a, b, ..] if a.role == "system" && b.role == "system" => (2, format!("{}\n{}", a.content.trim(), b.content.trim())),
                    [a, ..] if a.role == "system" => (1, a.content.trim().to_string()),
                    _ => (0, String::new()),
                };
                if !tools.is_empty() {
                    s += "<|im_start|>system\n# Tools\n\nYou have access to the following functions:\n\n<tools>";
                    for t in tools { s += "\n"; s += &t.dump(); }
                    s += "\n</tools>";
                    s += XML_RULES;
                    if !sys.is_empty() { s += "\n\n"; s += &sys; }
                    s += "<|im_end|>\n";
                } else if !sys.is_empty() {
                    s += &format!("<|im_start|>system\n{sys}<|im_end|>\n");
                }
                let lq = last_query(msgs, true);
                for (i, m) in msgs.iter().enumerate().skip(n_sys) {
                    let c = m.content.trim();
                    match m.role.as_str() {
                        "user" => s += &format!("<|im_start|>user\n{c}<|im_end|>\n"),
                        "assistant" => {
                            let (r, c) = split_think(m, c);
                            if preserve || i > lq { s += &format!("<|im_start|>assistant\n<think>\n{}\n</think>\n\n{c}", r.trim()); }
                            else { s += &format!("<|im_start|>assistant\n{c}"); }
                            for (k, call) in m.calls.iter().enumerate() {
                                s += if k > 0 { "\n" } else if c.trim().is_empty() { "" } else { "\n\n" };
                                s += &format!("<tool_call>\n<function={}>\n", call.name);
                                if let Json::Obj(o) = &call.args {
                                    for (k, v) in o { s += &format!("<parameter={k}>\n{}\n</parameter>\n", v.param()); }
                                }
                                s += "</function>\n</tool_call>";
                            }
                            s += "<|im_end|>\n";
                        }
                        "tool" => {
                            if i > 0 && msgs[i - 1].role != "tool" { s += "<|im_start|>user"; }
                            s += &format!("\n<tool_response>\n{c}\n</tool_response>");
                            tool_end(&mut s, i);
                        }
                        _ => {}  // later system messages are dropped
                    }
                }
                if generate { s += if think { "<|im_start|>assistant\n<think>\n" } else { "<|im_start|>assistant\n<think>\n\n</think>\n\n" }; }
            }
            Template::Json { reasoning } => {
                let sys = msgs.first().filter(|m| m.role == "system").map(|m| m.content.as_str());
                if !tools.is_empty() {
                    s += "<|im_start|>system\n";
                    if let Some(c) = sys { s += c; s += "\n\n"; }
                    s += JSON_HEAD;
                    for t in tools { s += "\n"; s += &t.dump(); }
                    s += JSON_TAIL;
                } else if let Some(c) = sys {
                    s += &format!("<|im_start|>system\n{c}<|im_end|>\n");
                }
                let lq = last_query(msgs, false);
                for (i, m) in msgs.iter().enumerate() {
                    let c = m.content.as_str();
                    match m.role.as_str() {
                        "user" => s += &format!("<|im_start|>user\n{c}<|im_end|>\n"),
                        "system" if i > 0 => s += &format!("<|im_start|>system\n{c}<|im_end|>\n"),
                        "assistant" => {
                            let (r, c) = if reasoning { split_think(m, c) } else { ("", c) };
                            if reasoning && i > lq && (i + 1 == msgs.len() || !r.is_empty()) {
                                s += &format!("<|im_start|>assistant\n<think>\n{}\n</think>\n\n{}", r.trim_matches('\n'), c.trim_start_matches('\n'));
                            } else {
                                s += &format!("<|im_start|>assistant\n{c}");
                            }
                            for (k, call) in m.calls.iter().enumerate() {
                                if k > 0 || !c.is_empty() { s += "\n"; }
                                s += &format!("<tool_call>\n{{\"name\": \"{}\", \"arguments\": {}}}\n</tool_call>", call.name, call.args.dump());
                            }
                            s += "<|im_end|>\n";
                        }
                        "tool" => {
                            if i == 0 || msgs[i - 1].role != "tool" { s += "<|im_start|>user"; }
                            s += &format!("\n<tool_response>\n{c}\n</tool_response>");
                            tool_end(&mut s, i);
                        }
                        _ => {}
                    }
                }
                if generate { s += "<|im_start|>assistant\n"; }
            }
        }
        s
    }

    /// The prompt that ends inside the last message when it is an assistant message without tool calls, so the
    /// model continues that reply (prefill, as llama.cpp's server and the Anthropic Messages API do). `None` when
    /// the last message is not one.
    pub fn render_open(&self, msgs: &[Msg], tools: &[Json], think: bool, preserve: bool) -> Option<String> {
        msgs.last().filter(|m| m.role == "assistant" && m.calls.is_empty())?;
        let s = self.render(msgs, tools, think, preserve, false);
        let end = match self { Template::Ds => "<｜end▁of▁sentence｜>", Template::Flint => "", _ => "<|im_end|>\n" };
        Some(s.strip_suffix(end).unwrap_or(&s).to_string())
    }

    /// The text that starts a tool call in a reply.
    pub fn call_tag(&self) -> &'static str { if let Template::Ds = self { DS_CALLS } else { TAG } }

    /// Tool calls at the end of a reply: the text before them, and the name and arguments (JSON text) of each call.
    /// None when the reply has no call or a call is malformed; the reply then goes out as plain text.
    // ponytail: a literal `<tool_call>` inside the thinking part makes the reply plain text; strip the think block
    // first if thinking with tools matters.
    pub fn parse_calls(&self, text: &str, tools: &[Json]) -> Option<(String, Vec<(String, String)>)> {
        if let Template::Ds = self { return ds_calls(text); }
        if let Template::Flint = self { return flint_calls(text); }
        let start = text.find(TAG)?;
        let mut rest = &text[start..];
        let mut calls = Vec::new();
        loop {
            rest = rest.trim_start();
            if rest.is_empty() { break; }
            let inner;
            (inner, rest) = rest.strip_prefix(TAG)?.split_once("</tool_call>")?;
            calls.push(match self { Template::Ds | Template::Flint => unreachable!(), Template::Xml => xml_call(inner, tools)?, Template::Json { .. } => json_call(inner)? });
        }
        Some((text[..start].trim_end().to_string(), calls))
    }
}

/// DeepSeek reply: a `<｜DSML｜ calls>` block of `invoke` items with `parameter name=".." string="true|false"`. A
/// parameter is a string when the model says so, else JSON (a string if it does not parse). The reply's thinking part
/// is skipped, so a tag inside it is not taken for a call.
// ponytail: not run against the real model (the weights are deleted); the format is the template's own.
fn ds_calls(text: &str) -> Option<(String, Vec<(String, String)>)> {
    let from = text.rfind("</think>").map_or(0, |p| p + 8);
    let start = from + text[from..].find(DS_CALLS)?;
    let mut rest = text[start..].strip_prefix(DS_CALLS)?;
    let mut calls = Vec::new();
    loop {
        rest = rest.trim_start();
        if let Some(x) = rest.strip_prefix("</｜DSML｜ calls>") {
            if !x.trim().is_empty() || calls.is_empty() { return None; }
            break;
        }
        let (name, mut r) = rest.strip_prefix("<｜DSML｜ invoke name=\"")?.split_once("\">")?;
        let mut args = Vec::new();
        loop {
            r = r.trim_start();
            if let Some(x) = r.strip_prefix("</｜DSML｜ invoke>") { rest = x; break; }
            let (head, x) = r.strip_prefix("<｜DSML｜ parameter name=\"")?.split_once("\">")?;
            let (k, string) = head.split_once("\" string=\"")?;
            let (v, x) = x.split_once("</｜DSML｜ parameter>")?;
            r = x;
            args.push((k.to_string(), if string == "true" { Json::Str(v.into()) } else { serde_json::from_str(v).unwrap_or_else(|_| Json::Str(v.into())) }));
        }
        calls.push((name.to_string(), Json::Obj(args).dump()));
    }
    Some((text[..start].trim_end().to_string(), calls))
}

/// `{"name": .., "arguments": {..}}`; arguments given as a JSON string are accepted too.
fn json_call(inner: &str) -> Option<(String, String)> {
    let v: Json = serde_json::from_str(inner.trim()).ok()?;
    let args = match v.get("arguments") {
        Json::Str(s) => serde_json::from_str(s).ok()?,
        a => a.clone(),
    };
    let Json::Obj(_) = args else { return None };
    Some((v.get("name").as_str()?.to_string(), args.dump()))
}

/// `<function=name>` with `<parameter=key>\nvalue\n</parameter>` items. A value is a string when the tool's schema
/// says so, else JSON if it parses as JSON, else a string.
fn xml_call(inner: &str, tools: &[Json]) -> Option<(String, String)> {
    let (name, mut r) = inner.trim_start().strip_prefix("<function=")?.split_once('>')?;
    let props = tools.iter().map(|t| t.get("function")).find(|f| f.get("name").as_str() == Some(name))
        .map_or(&NULL, |f| f.get("parameters").get("properties"));
    let mut args = Vec::new();
    loop {
        r = r.trim_start();
        if let Some(x) = r.strip_prefix("</function>") {
            if !x.trim().is_empty() { return None; }
            break;
        }
        let (k, x) = r.strip_prefix("<parameter=")?.split_once('>')?;
        let (v, x) = x.split_once("</parameter>")?;
        r = x;
        let v = v.strip_prefix('\n').unwrap_or(v);
        let v = v.strip_suffix('\n').unwrap_or(v);
        let string = props.get(k).get("type").as_str() == Some("string");
        // The template prints earlier calls' booleans and null in Python spelling, and the model copies it.
        let py = |v: &str| match v.trim() { "True" => Some(Json::Bool(true)), "False" => Some(Json::Bool(false)), "None" => Some(Json::Null), _ => None };
        args.push((k.to_string(), if string { Json::Str(v.into()) } else { serde_json::from_str(v).ok().or_else(|| py(v)).unwrap_or_else(|| Json::Str(v.into())) }));
    }
    Some((name.to_string(), Json::Obj(args).dump()))
}

/// Streams reply text but holds back everything from a `<tool_call>` tag on (unless `no_tag`), so a call is never
/// sent as content, and ends the reply before the first of the `stops` strings.
#[derive(Default)]
pub struct Hold { pub text: String, sent: usize, held: bool, no_tag: bool, tag: &'static str, stops: Vec<String>, pub stopped: bool }

impl Hold {
    pub fn new(tag: bool, stops: Vec<String>) -> Hold { Hold { no_tag: !tag, stops, ..Hold::default() } }
    /// Holds back from `tag` instead of `<tool_call>` (the DeepSeek call block).
    pub fn tagged(mut self, tag: &'static str) -> Hold { self.tag = tag; self }
    /// Adds a piece of the reply; returns the text that is safe to send now.
    pub fn push(&mut self, piece: &str) -> &str {
        let from = self.sent;
        if self.stopped { return ""; }
        self.text += piece;
        // A stop string starts in the unsent text: a tail that could start one is never sent.
        if let Some(p) = self.stops.iter().filter_map(|s| self.text[from..].find(s.as_str())).min() {
            self.text.truncate(from + p);
            self.stopped = true;
        }
        if !self.held {
            self.sent = match self.text[from..].find(self.tag()).filter(|_| !self.no_tag) {
                Some(p) => { self.held = true; from + p }
                None if self.stopped => self.text.len(),
                None => (self.text.len() - self.tail()).max(from),
            };
        }
        &self.text[from..self.sent]
    }
    fn tag(&self) -> &'static str { if self.tag.is_empty() { TAG } else { self.tag } }
    /// The longest end of the text that could be the start of the tag or of a stop string.
    fn tail(&self) -> usize {
        (!self.no_tag).then_some(self.tag()).into_iter().chain(self.stops.iter().map(String::as_str))
            .flat_map(|s| s.char_indices().skip(1).map(move |(k, _)| &s[..k]))
            .filter(|p| self.text.ends_with(p)).map(str::len).max().unwrap_or(0)
    }
    /// The text not sent yet.
    pub fn rest(&self) -> &str { &self.text[self.sent..] }
}

/// Splits a reply that starts inside an open think block into reasoning and answer at `</think>`,
/// holding back a tail that could still become the tag.
pub struct Think { open: bool, lead: bool, buf: String }

impl Think {
    const TAG: &'static str = "</think>";
    pub fn new(open: bool) -> Think { Think { open, lead: false, buf: String::new() } }
    /// (reasoning, answer) that are safe to send after adding `piece`.
    pub fn feed(&mut self, piece: &str) -> (String, String) {
        self.buf += piece;
        if !self.open { let b = std::mem::take(&mut self.buf); return (String::new(), self.answer(b)); }
        if let Some(p) = self.buf.find(Self::TAG) {
            self.open = false;
            self.lead = true;
            let r = self.buf[..p].trim_end_matches('\n').to_string();
            let rest = self.buf[p + Self::TAG.len()..].to_string();
            self.buf.clear();
            let a = self.answer(rest);
            return (r, a);
        }
        let keep = (1..Self::TAG.len()).rev().find(|&k| self.buf.ends_with(&Self::TAG[..k])).unwrap_or(0);
        let cut = self.buf.len() - keep;
        let r = self.buf[..cut].to_string();
        self.buf.drain(..cut);
        (r, String::new())
    }
    /// The answer without the blank lines after `</think>`.
    fn answer(&mut self, a: String) -> String {
        if !self.lead { return a; }
        let t = a.trim_start_matches('\n');
        self.lead = t.is_empty();
        t.to_string()
    }
    /// What is still held when the reply ends.
    pub fn finish(&mut self) -> (String, String) {
        let b = std::mem::take(&mut self.buf);
        if self.open { (b, String::new()) } else { (String::new(), self.answer(b)) }
    }
}

/// What differs between the supported architectures; everything else comes from GGUF metadata.
struct Arch {
    /// DeltaNet layers with full attention every `full_attention_interval` layers; otherwise all attention.
    hybrid: bool,
    /// `attn_q` is [query | sigmoid output gate] per head (2 * head_dim wide).
    q_gate: bool,
    /// DeltaNet value head h reads key head h / (nv / nk) (llama.cpp qwen3next), not h % nk.
    k_interleave: bool,
    /// Canonical tensor suffix to this architecture's suffix.
    aliases: &'static [(&'static str, &'static str)],
}

fn arch(name: &str) -> Option<Arch> {
    const FFN_NORM: &[(&str, &str)] = &[("post_attention_norm.weight", "ffn_norm.weight")];
    const SENSENOVA_LOOP: &[(&str, &str)] = &[
        ("mhc_attn.weight", "hc_attn_fn.weight"),
        ("mhc_attn.bias", "hc_attn_fn.bias"),
        ("mhc_ffn.weight", "hc_ffn_fn.weight"),
        ("mhc_ffn.bias", "hc_ffn_fn.bias"),
    ];
    Some(match name {
        "qwen35" | "qwen35moe" => Arch { hybrid: true, q_gate: true, k_interleave: false, aliases: &[] },
        // SenseNova-Loop: Qwen3.5 hybrid DeltaNet/Attn with shared-weight looped middle layers and mHC Sinkhorn routing.
        "sensenova_loop" => Arch { hybrid: true, q_gate: true, k_interleave: false, aliases: SENSENOVA_LOOP },
        // Qwen3.8-Flash-Next: qwen35-style DeltaNet pairing, plus hyper-connections and PLE read from metadata.
        "qwen4exp" => Arch { hybrid: true, q_gate: true, k_interleave: false, aliases: &[] },
        "qwen3next" => Arch { hybrid: true, q_gate: true, k_interleave: true, aliases: &[] },
        // Recursive Flint: Qwen-style attention with sliding windows, per-head QK norms and attention sinks.
        "flint" => Arch { hybrid: false, q_gate: false, k_interleave: false, aliases: FFN_NORM },
        "qwen3" | "qwen3moe" => Arch { hybrid: false, q_gate: false, k_interleave: false, aliases: FFN_NORM },
        _ => return None,
    })
}

/// `prism.hadamard.*` of agentionai's Gyro files. Every routed expert matrix is stored as `W diag(s) H^T` over blocks
/// of `block` input columns (H the normalized Sylvester-Walsh-Hadamard matrix, s a vector of +1 and -1), so its input
/// x has to become `H (s x)` first: the layer input before gate and up, the SwiGLU output before down. The router and
/// the shared expert see the unrotated values.
#[derive(Clone)]
pub struct Hadamard { pub block: usize, signs: Vec<Vec<f32>> }

impl std::fmt::Debug for Hadamard {
    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
        write!(f, "Hadamard {{ block: {}, widths: {:?} }}", self.block, self.signs.iter().map(|s| s.len()).collect::<Vec<_>>())
    }
}

impl Hadamard {
    fn from_gguf(g: &Gguf) -> anyhow::Result<Option<Hadamard>> {
        use crate::gguf::Value;
        let key = |k: &str| g.meta.get(&format!("prism.hadamard.{k}"));
        let Some(version) = key("version") else { return Ok(None) };
        let text = |k: &str| key(k).and_then(|v| v.as_str()).unwrap_or("");
        let arr = |k: &str| key(k).and_then(|v| v.as_arr()).unwrap_or(&[]);
        let block = key("block_size").and_then(|v| v.as_u64()).unwrap_or(0) as usize;
        anyhow::ensure!(version.as_u64() == Some(1) && text("transform") == "normalized-sylvester-walsh-hadamard" && text("axis") == "input-last-dimension"
            && block > 1 && block.is_power_of_two() && arr("inverse_weight_names").is_empty(),
            "unsupported prism.hadamard metadata (version {version:?}, transform {:?}, axis {:?}, block_size {block})", text("transform"), text("axis"));
        // ponytail: all routed experts rotated or none (true for the Gyro files); a table per layer if a file ever
        // rotates only some of them.
        let mut named: Vec<&str> = arr("weight_names").iter().filter_map(|v| v.as_str()).collect();
        let mut routed: Vec<&str> = g.tensors.keys().map(|n| n.as_str())
            .filter(|n| n.starts_with("blk.") && ["gate", "up", "down"].iter().any(|r| n.ends_with(&format!(".ffn_{r}_exps.weight")))).collect();
        named.sort_unstable();
        routed.sort_unstable();
        anyhow::ensure!(named == routed, "prism.hadamard.weight_names lists {} tensors, the file has {} routed expert tensors: only files with every routed expert rotated are supported", named.len(), routed.len());
        let mut widths: Vec<usize> = routed.iter().map(|n| g.tensors[*n].dims[0] as usize).collect();
        widths.sort_unstable();
        widths.dedup();
        anyhow::ensure!(widths.iter().all(|w| w % block == 0), "prism.hadamard.block_size {block} does not divide the expert widths {widths:?}");
        let signs = match text("sign_mode") {
            "identity" => widths.iter().map(|&w| vec![1f32; w]).collect(),
            "explicit" => {
                let sw: Vec<usize> = arr("sign_widths").iter().filter_map(|v| v.as_u64()).map(|v| v as usize).collect();
                let sv: Vec<f32> = arr("sign_values").iter().map(|v| match *v {
                    Value::I8(v) => v as f32, Value::I16(v) => v as f32, Value::I32(v) => v as f32, Value::I64(v) => v as f32,
                    _ => v.as_f32().unwrap_or(0.0),
                }).collect();
                anyhow::ensure!(sw.iter().sum::<usize>() == sv.len() && sv.iter().all(|v| v.abs() == 1.0), "prism.hadamard sign_widths / sign_values are inconsistent");
                let mut at = 0;
                let all: Vec<Vec<f32>> = sw.iter().map(|&w| { at += w; sv[at - w..at].to_vec() }).collect();
                anyhow::ensure!(widths.iter().all(|w| all.iter().any(|s| s.len() == *w)), "prism.hadamard has no sign vector for one of the widths {widths:?}");
                all
            }
            mode => bail!("unsupported prism.hadamard.sign_mode {mode:?}"),
        };
        Ok(Some(Hadamard { block, signs }))
    }

    /// The signs for an input of `n` values.
    pub fn signs(&self, n: usize) -> &[f32] {
        self.signs.iter().find(|s| s.len() == n).unwrap_or_else(|| panic!("no prism.hadamard signs for width {n}"))
    }

    /// `x` becomes `H (s x)`, block by block.
    pub fn rotate(&self, x: &mut [f32]) {
        let signs = self.signs(x.len());
        for (v, s) in x.iter_mut().zip(signs) { *v *= s; }
        Self::fwht(x, self.block);
    }

    /// `x` becomes `H x` per `block` values: the fast Walsh-Hadamard transform in natural (Sylvester) order, where
    /// entry (r, c) of H is (-1)^popcount(r & c) / sqrt(block).
    fn fwht(x: &mut [f32], block: usize) {
        let scale = 1.0 / (block as f32).sqrt();
        for v in x.chunks_exact_mut(block) {
            let mut len = 1;
            while len < block {
                for i in (0..block).step_by(2 * len) {
                    for j in i..i + len { let (a, c) = (v[j], v[j + len]); v[j] = a + c; v[j + len] = a - c; }
                }
                len *= 2;
            }
            for a in v { *a *= scale; }
        }
    }

    pub fn rotated(&self, x: &[f32]) -> Vec<f32> { let mut y = x.to_vec(); self.rotate(&mut y); y }

    /// The inverse of `rotate`: H is its own inverse, so `x` becomes `s (H x)`.
    pub fn unrotate(&self, x: &mut [f32]) {
        let signs = self.signs(x.len());
        Self::fwht(x, self.block);
        for (v, s) in x.iter_mut().zip(signs) { *v *= s; }
    }
}

#[test]
fn hadamard_rotation_is_the_matrix_product() {
    let h = Hadamard { block: 8, signs: vec![(0..16).map(|i| if i % 3 == 0 { -1.0 } else { 1.0 }).collect()] };
    let x: Vec<f32> = (0..16).map(|i| (i * i) as f32 - 20.0).collect();
    let y = h.rotated(&x);
    for r in 0..16 {
        let want: f32 = (0..8).map(|c| { let i: usize = r / 8 * 8 + c; (if (r % 8 & c as usize).count_ones() % 2 == 1 { -1.0 } else { 1.0 }) * h.signs(16)[i] * x[i] }).sum::<f32>() / 8f32.sqrt();
        assert!((y[r] - want).abs() < 1e-4, "{r}: {} vs {want}", y[r]);
    }
    let mut back = y;
    h.unrotate(&mut back);
    assert!(back.iter().zip(&x).all(|(a, b)| (a - b).abs() < 1e-4), "unrotate: {back:?} vs {x:?}");
}

/// Architectures with their own forward and server outside `ModelConfig` (`src/main.rs` dispatches on them).
pub const OWN_ARCHS: [&str; 6] = ["gemma4", "glm5-next", "deepseek41", crate::hy4::ARCH, crate::inkling2::ARCH, crate::inkling2::ARCH_RUNS];

impl ModelConfig {
    pub fn from_gguf(g: &Gguf) -> anyhow::Result<ModelConfig> {
        let arch = g.get("general.architecture")?.as_str().context("general.architecture")?.to_string();
        let Some(a) = self::arch(&arch) else {
            ensure!(!OWN_ARCHS.contains(&arch.as_str()), "'{arch}' models run with chat, generate, bench and serve only");
            bail!("unsupported architecture '{arch}' (supported: qwen3, qwen3moe, qwen3next, qwen35, qwen35moe, qwen4exp, flint, sensenova_loop)")
        };
        let opt = |k: &str| g.meta.get(&format!("{arch}.{k}")).and_then(|v| v.as_u64()).unwrap_or(0) as usize;
        let req = |k: &str| -> anyhow::Result<usize> {
            let key = format!("{arch}.{k}");
            Ok(g.get(&key)?.as_u64().with_context(|| format!("{key}: not an integer"))? as usize)
        };
        let f = |k: &str| g.meta.get(&format!("{arch}.{k}")).and_then(|v| v.as_f32());
        let n_nextn = opt("nextn_predict_layers");
        let n_layer = req("block_count")? - n_nextn;
        let flags = |k: &str| -> anyhow::Result<Option<Vec<bool>>> {
            let key = format!("{arch}.{k}");
            let Some(v) = g.meta.get(&key) else { return Ok(None) };
            let a = v.as_arr().with_context(|| format!("{key}: not an array"))?;
            anyhow::ensure!(a.len() >= n_layer, "{key} has {} entries for {n_layer} layers", a.len());
            Ok(Some(a[..n_layer].iter().map(|v| match v {
                crate::gguf::Value::Bool(b) => *b,
                _ => v.as_u64().unwrap_or(0) != 0,
            }).collect()))
        };
        let layers: Vec<LayerKind> = if let Some(rec) = flags("attention.recurrent_layers")? {
            rec.into_iter().map(|r| if r { LayerKind::Delta } else { LayerKind::Attn }).collect()
        } else if a.hybrid {
            let every = match opt("full_attention_interval") { 0 => 4, v => v };
            (0..n_layer).map(|il| if (il + 1) % every == 0 { LayerKind::Attn } else { LayerKind::Delta }).collect()
        } else {
            vec![LayerKind::Attn; n_layer]
        };
        let has_delta = layers.contains(&LayerKind::Delta);
        let ssm = |k: &str| if has_delta { req(k) } else { Ok(opt(k)) };
        let (n_embd, n_head) = (req("embedding_length")?, req("attention.head_count")?);
        let head_dim = match opt("attention.key_length") { 0 => n_embd / n_head, v => v };
        let n_expert = opt("expert_count");
        anyhow::ensure!(n_expert <= 512, "{n_expert} experts (at most 512 supported)");
        let n_dense = opt("leading_dense_block_count");
        anyhow::ensure!(n_dense <= n_layer, "{n_dense} leading dense blocks for {n_layer} layers");
        let mhc_streams = opt("mhc.streams");
        let loop_start = opt("loop.start");
        let loop_end = opt("loop.end");
        let loop_count = opt("loop.count").max(1);
        let mhc_layer_kinds: Vec<u8> = match g.meta.get(&format!("{arch}.mhc.layer_kinds")).and_then(|v| v.as_arr()) {
            Some(arr) if arr.len() >= n_layer => arr[..n_layer].iter().map(|v| v.as_u64().unwrap_or(0) as u8).collect(),
            _ if mhc_streams > 0 && opt("mhc.end") > opt("mhc.start") => {
                let (ms, me) = (opt("mhc.start"), opt("mhc.end"));
                let total_mhc = (ms + (me - ms) * loop_count).min(n_layer);
                (0..n_layer).map(|il| {
                    if il < ms || il >= total_mhc { 0 }
                    else if il == ms { 1 }
                    else if il + 1 == total_mhc { 2 }
                    else { 4 }
                }).collect()
            }
            _ => Vec::new(),
        };
        anyhow::ensure!(mhc_streams == 0 || matches!(mhc_streams, 2 | 4), "unsupported mhc.streams {mhc_streams} (expected 2 or 4)");
        let has_blk = |il: usize| {
            g.tensors.contains_key(&format!("blk.{il}.attn_norm.weight"))
                || g.tensors.contains_key(&format!("blk.{il}.hc_attn_norm.weight"))
        };
        let loop_span = if loop_end > loop_start {
            Some((loop_start, loop_end))
        } else if !mhc_layer_kinds.is_empty() {
            let s = mhc_layer_kinds.iter().position(|&k| k & 1 != 0);
            let e = s.and_then(|st| mhc_layer_kinds[st..].iter().position(|&k| k & 2 != 0).map(|o| st + o + 1));
            s.zip(e).filter(|&(st, en)| en > st)
        } else {
            None
        };
        let first_missing = (0..n_layer).find(|&il| !has_blk(il));
        let phys_layer: Vec<usize> = (0..n_layer).map(|il| {
            if has_blk(il) {
                il
            } else if let Some((st, en)) = loop_span {
                st + (il.saturating_sub(st)) % (en - st)
            } else if let Some(fm) = first_missing.filter(|&fm| fm > 0) {
                let miss_end = (fm..n_layer).find(|&j| has_blk(j)).unwrap_or(n_layer);
                let span = (miss_end - fm).min(fm).max(1);
                (fm - span) + (il - fm) % span
            } else {
                il
            }
        }).collect();
        let swa = flags("attention.sliding_window_pattern")?.unwrap_or_else(|| vec![false; n_layer]);
        let rope_layer = flags("attention.rope_pattern")?.unwrap_or_else(|| vec![true; n_layer]);
        let n_swa = opt("attention.sliding_window");
        anyhow::ensure!(!swa.contains(&true) || n_swa > 0, "sliding-window layers without attention.sliding_window");
        let emb = g.tensors.get("token_embd.weight").context("missing tensor token_embd.weight")?;
        let c = ModelConfig {
            n_layer,
            n_nextn,
            n_embd,
            n_head,
            n_head_kv: req("attention.head_count_kv")?,
            head_dim,
            n_rot: match opt("rope.dimension_count") { 0 => head_dim, v => v },
            mrope_sections: g.meta.get(&format!("{arch}.rope.dimension_sections")).and_then(|v| v.as_arr()).and_then(|a| {
                if a.len() < 4 { return None; }
                let mut s = [0usize; 4];
                for (o, v) in s.iter_mut().zip(a) { *o = v.as_u64()? as usize; }
                Some(s)
            }),
            rope_base: f("rope.freq_base").unwrap_or(10000.0),
            eps: f("attention.layer_norm_rms_epsilon").context("missing rms epsilon")?,
            n_vocab: emb.dims[1] as usize,
            n_ff: opt("feed_forward_length"),
            n_expert,
            n_expert_used: opt("expert_used_count"),
            n_ff_exp: opt("expert_feed_forward_length"),
            n_ff_shexp: opt("expert_shared_feed_forward_length"),
            n_dense,
            expert_weights_scale: f("expert_weights_scale").unwrap_or(0.0),
            router_sigmoid: arch == "flint" && opt("expert_gating_func") == 2,
            hadamard: Hadamard::from_gguf(g)?,
            n_swa,
            swa,
            rope_layer,
            attn_sinks: g.tensors.contains_key("blk.0.attn_sinks.weight"),
            ssm_d_inner: ssm("ssm.inner_size")?,
            ssm_d_state: ssm("ssm.state_size")?,
            ssm_n_group: ssm("ssm.group_count")?,
            ssm_dt_rank: ssm("ssm.time_step_rank")?,
            ssm_conv: ssm("ssm.conv_kernel")?,
            hc_n: opt("hyper_connection.count"),
            hc_rank: opt("hyper_connection.low_rank"),
            mhc_streams,
            mhc_layer_kinds,
            phys_layer,
            n_embd_ple: opt("embedding_length_per_layer_input"),
            idx_n_head: opt("attention.indexer.head_count"),
            idx_head_dim: opt("attention.indexer.key_length"),
            idx_top_k: opt("attention.indexer.top_k"),
            layers,
            q_gate: a.q_gate,
            k_interleave: a.k_interleave,
            aliases: a.aliases,
            arch,
            eagle3: None,
            dflash: None,
        };
        // Kernels divide by these and index value heads by nv / nk key heads.
        let (nk, nv) = (c.ssm_n_group, c.ssm_dt_rank);
        anyhow::ensure!(!has_delta || (nk > 0 && nv > 0 && nv % nk == 0 && c.ssm_d_inner % nv == 0),
            "bad ssm dims: group_count {nk}, time_step_rank {nv}, inner_size {}", c.ssm_d_inner);
        Ok(c)
    }

    pub fn is_moe(&self) -> bool { self.n_expert > 0 }
    /// Layer `il` uses the MoE FFN (dense-lead architectures start with `n_dense` dense layers).
    pub fn is_moe_layer(&self, il: usize) -> bool { self.n_expert > 0 && il >= self.n_dense }
    /// DeltaNet value head size (0 without DeltaNet layers).
    pub fn ssm_dv(&self) -> usize { self.ssm_d_inner / self.ssm_dt_rank.max(1) }
}
// ==== /z/ai-models/rs-v131wt/src/cpu_moe.rs
//! Fast CPU routed-expert path: activations quantized to Q8_K blocks, integer dot products per weight type
//! (ported from llama.cpp `ggml_vec_dot_*_q8_K_generic`, MIT License, ggml authors). Types without an integer
//! kernel fall back to f32 dequantization.

use crate::model::Model;
use crate::ops;
use crate::quant::{GgmlType, KVALUES_FP4, tables::*};
use crate::weights::Tensor;
use half::f16;
use rayon::prelude::*;

/// `bs` holds the sums of the 16 groups of 16 `qs`, for the min terms of Q2_K and Q4_K.
pub struct Q8K { d: f32, qs: [i8; 256], bs: [i16; 16] }

/// Signs of IQ2/IQ3 sub-blocks as ±1 bytes, indexed by the 7-bit sign field.
static SIGNS: [[i8; 8]; 128] = {
    let mut t = [[1i8; 8]; 128];
    let mut i = 0;
    while i < 128 {
        let mut j = 0;
        while j < 8 { if KSIGNS_IQ2XS[i] & (1 << j) != 0 { t[i][j] = -1; } j += 1; }
        i += 1;
    }
    t
};

/// Quantizes `x` in blocks of 256; a short last block is zero-padded (Q2_0 rows can hold 64 * k values).
pub fn quantize_q8k(x: &[f32]) -> Vec<Q8K> {
    x.chunks(256).map(|b| {
        let mut pad = [0f32; 256];
        let b = if b.len() == 256 { b } else { pad[..b.len()].copy_from_slice(b); &pad[..] };
        // The first value of largest magnitude (a max over |v| first, which vectorizes).
        let amax = b.iter().fold(0f32, |m, &v| m.max(v.abs()));
        if amax == 0.0 { return Q8K { d: 0.0, qs: [0; 256], bs: [0; 16] }; }
        let max = *b.iter().find(|v| v.abs() == amax).unwrap();
        let is = -127.0 / max;
        let mut qs = [0i8; 256];
        #[cfg(target_feature = "avx2")]
        avx2::quant_block(is, b, &mut qs);
        #[cfg(not(target_feature = "avx2"))]
        for (q, &v) in qs.iter_mut().zip(b) { *q = ((is * v).round_ties_even() as i32).min(127) as i8; }
        let mut bs = [0i16; 16];
        for (s, c) in bs.iter_mut().zip(qs.chunks_exact(16)) { *s = c.iter().map(|&v| v as i16).sum(); }
        Q8K { d: 1.0 / is, qs, bs }
    }).collect()
}

fn d16(b: &[u8]) -> f32 { f16::from_le_bytes([b[0], b[1]]).to_f32() }

#[inline]
fn dot8(g: [u8; 8], s: &[i8; 8], q: &[i8]) -> i32 {
    (0..8).map(|j| g[j] as i32 * s[j] as i32 * q[j] as i32).sum()
}

fn dot_iq2xs(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(74).zip(xq) {
        let sc = &b[66..74];
        let mut bsum = 0i32;
        for ib in 0..8 {
            let mut s = [0i32; 2];
            for l in 0..4 {
                let o = 2 + 2 * (4 * ib + l);
                let q = u16::from_le_bytes([b[o], b[o + 1]]);
                s[l / 2] += dot8(IQ2XS_GRID[(q & 511) as usize].to_le_bytes(), &SIGNS[(q >> 9) as usize], &y.qs[32 * ib + 8 * l..]);
            }
            bsum += s[0] * (2 * (sc[ib] & 0xF) as i32 + 1) + s[1] * (2 * (sc[ib] >> 4) as i32 + 1);
        }
        sumf += d16(b) * y.d * bsum as f32;
    }
    0.125 * sumf
}

fn dot_iq2xxs(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(66).zip(xq) {
        let mut bsum = 0i32;
        for ib in 0..8 {
            let aux = u32::from_le_bytes(b[6 + 8 * ib..10 + 8 * ib].try_into().unwrap());
            let s: i32 = (0..4).map(|l| dot8(IQ2XXS_GRID[b[2 + 8 * ib + l] as usize].to_le_bytes(), &SIGNS[((aux >> (7 * l)) & 127) as usize], &y.qs[32 * ib + 8 * l..])).sum();
            bsum += s * (2 * (aux >> 28) as i32 + 1);
        }
        sumf += d16(b) * y.d * bsum as f32;
    }
    0.125 * sumf
}

fn dot_iq2s(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(82).zip(xq) {
        let (qs, signs, qh, sc) = (&b[2..34], &b[34..66], &b[66..74], &b[74..82]);
        let mut bsum = 0i32;
        for ib in 0..8 {
            let s: [i32; 2] = std::array::from_fn(|h| (2 * h..2 * h + 2).map(|l| {
                let g = IQ2S_GRID[qs[4 * ib + l] as usize | ((qh[ib] as usize) << (8 - 2 * l) & 0x300)].to_le_bytes();
                let sg: [i8; 8] = std::array::from_fn(|j| 1 - 2 * (signs[4 * ib + l] >> j & 1) as i8);
                dot8(g, &sg, &y.qs[32 * ib + 8 * l..])
            }).sum());
            bsum += s[0] * (2 * (sc[ib] & 0xF) as i32 + 1) + s[1] * (2 * (sc[ib] >> 4) as i32 + 1);
        }
        sumf += d16(b) * y.d * bsum as f32;
    }
    0.125 * sumf
}

/// Dot of 8 IQ1 grid values (-1, 0, 1) with `q`.
fn dot1(g: u64, q: &[i8]) -> i32 { g.to_le_bytes().iter().zip(&q[..8]).map(|(&a, &b)| a as i8 as i32 * b as i32).sum() }

fn dot_iq1s(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(50).zip(xq) {
        let (mut s, mut s1) = (0i32, 0i32);
        for ib in 0..8 {
            let qh = u16::from_le_bytes([b[34 + 2 * ib], b[35 + 2 * ib]]) as usize;
            let ls = 2 * ((qh >> 12) & 7) as i32 + 1;
            s += ls * (0..4).map(|l| dot1(IQ1S_GRID[b[2 + 4 * ib + l] as usize | ((qh >> (3 * l)) & 7) << 8], &y.qs[32 * ib + 8 * l..])).sum::<i32>();
            s1 += ls * if qh & 0x8000 != 0 { -1 } else { 1 } * (y.bs[2 * ib] + y.bs[2 * ib + 1]) as i32;
        }
        sumf += d16(b) * y.d * (s as f32 + 0.125 * s1 as f32);
    }
    sumf
}

/// IQ1_M block scale: the top 4 bits of the four scale words.
fn iq1m_d(b: &[u8]) -> f32 {
    let sc = |i: usize| u16::from_le_bytes([b[48 + 2 * i], b[49 + 2 * i]]);
    f16::from_bits((sc(0) >> 12) | ((sc(1) >> 8) & 0xF0) | ((sc(2) >> 4) & 0xF00) | (sc(3) & 0xF000)).to_f32()
}

/// Scale of values 16 * (2 * ib + h)..+16 of an IQ1_M block.
fn iq1m_ls(b: &[u8], ib: usize, h: usize) -> i32 {
    2 * ((u16::from_le_bytes([b[48 + ib / 2 * 2], b[49 + ib / 2 * 2]]) >> (6 * (ib % 2) + 3 * h)) & 7) as i32 + 1
}

fn dot_iq1m(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(56).zip(xq) {
        let (mut s1, mut s2) = (0i32, 0i32);
        for ib in 0..8 {
            for l in 0..4 {
                let qh = b[32 + 2 * ib + l / 2] as usize >> (4 * (l % 2));
                let q = &y.qs[32 * ib + 8 * l..32 * ib + 8 * l + 8];
                let ls = iq1m_ls(b, ib, l / 2);
                s1 += ls * dot1(IQ1S_GRID[b[4 * ib + l] as usize | (qh & 7) << 8], q);
                s2 += ls * if qh & 8 != 0 { -1 } else { 1 } * q.iter().map(|&v| v as i32).sum::<i32>();
            }
        }
        sumf += iq1m_d(b) * y.d * (s1 as f32 + 0.125 * s2 as f32);
    }
    sumf
}

fn dot_iq3xxs(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(98).zip(xq) {
        let (qs, gas) = (&b[2..66], &b[66..98]);
        let mut bsum = 0i32;
        for ib in 0..8 {
            let aux = u32::from_le_bytes(gas[4 * ib..4 * ib + 4].try_into().unwrap());
            let mut s = 0i32;
            for l in 0..4 {
                let g1 = IQ3XXS_GRID[qs[8 * ib + 2 * l] as usize].to_le_bytes();
                let g2 = IQ3XXS_GRID[qs[8 * ib + 2 * l + 1] as usize].to_le_bytes();
                let g = [g1[0], g1[1], g1[2], g1[3], g2[0], g2[1], g2[2], g2[3]];
                s += dot8(g, &SIGNS[((aux >> (7 * l)) & 127) as usize], &y.qs[32 * ib + 8 * l..]);
            }
            bsum += s * (2 * (aux >> 28) as i32 + 1);
        }
        sumf += d16(b) * y.d * bsum as f32;
    }
    0.25 * sumf
}

fn dot_iq3s(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(110).zip(xq) {
        let (qs, qh, signs, sc) = (&b[2..66], &b[66..74], &b[74..106], &b[106..110]);
        let mut bsum = 0i32;
        for ib in 0..8 {
            let mut s = 0i32;
            for l in 0..4 {
                let g = |k: usize, sh: usize| IQ3S_GRID[(qs[8 * ib + 2 * l + k] as usize) | ((qh[ib] as usize) << sh & 256)].to_le_bytes();
                let (g1, g2) = (g(0, 8 - 2 * l), g(1, 7 - 2 * l));
                let g = [g1[0], g1[1], g1[2], g1[3], g2[0], g2[1], g2[2], g2[3]];
                let sg: [i8; 8] = std::array::from_fn(|j| if signs[4 * ib + l] >> j & 1 != 0 { -1 } else { 1 });
                s += dot8(g, &sg, &y.qs[32 * ib + 8 * l..]);
            }
            bsum += s * (2 * (sc[ib / 2] >> (4 * (ib % 2)) & 0xF) as i32 + 1);
        }
        sumf += d16(b) * y.d * bsum as f32;
    }
    sumf
}

fn dot_iq4xs(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(136).zip(xq) {
        let dd = d16(b) * y.d;
        let sh = u16::from_le_bytes([b[2], b[3]]);
        let (sl, qs) = (&b[4..8], &b[8..136]);
        for ib in 0..8 {
            let ls = ((sl[ib / 2] >> (4 * (ib % 2))) & 0xF) as i32 | ((((sh >> (2 * ib)) & 3) as i32) << 4);
            let q8 = &y.qs[32 * ib..32 * ib + 32];
            let s: i32 = (0..16).map(|j| {
                let q = qs[16 * ib + j];
                q8[j] as i32 * KVALUES_IQ4NL[(q & 0xF) as usize] as i32 + q8[j + 16] as i32 * KVALUES_IQ4NL[(q >> 4) as usize] as i32
            }).sum();
            sumf += dd * (ls - 32) as f32 * s as f32;
        }
    }
    sumf
}

/// IQ4_NL: 32 values d * KVALUES_IQ4NL[q] per block, eight blocks per Q8_K block.
fn dot_iq4nl(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(18).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 8], i % 8 * 32);
        let s: i32 = (0..16).map(|j| {
            let q = b[2 + j];
            KVALUES_IQ4NL[(q & 0xF) as usize] as i32 * y.qs[o + j] as i32 + KVALUES_IQ4NL[(q >> 4) as usize] as i32 * y.qs[o + j + 16] as i32
        }).sum();
        d16(b) * y.d * s as f32
    }).sum()
}

/// NVFP4: 64 values per block in four sub-blocks of 16, each with its own UE4M3 scale; four blocks per Q8_K block.
fn dot_nvfp4(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(36).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 4], i % 4 * 64);
        (0..4).map(|s| {
            let q8 = &y.qs[o + 16 * s..];
            let v: i32 = (0..8).map(|j| {
                let q = b[4 + 8 * s + j];
                KVALUES_FP4[(q & 0xF) as usize] as i32 * q8[j] as i32 + KVALUES_FP4[(q >> 4) as usize] as i32 * q8[j + 8] as i32
            }).sum();
            crate::quant::ue4m3(b[s]) * y.d * v as f32
        }).sum::<f32>()
    }).sum()
}

/// Sum over the 16 groups of 16 of `sc(group) * (a · q)`, written so the compiler vectorizes it.
#[inline(always)]
fn grouped(a: &[i8; 256], q: &[i8; 256], sc: impl Fn(usize) -> i32) -> i32 {
    (0..16).map(|j| sc(j) * (16 * j..16 * j + 16).map(|i| a[i] as i32 * q[i] as i32).sum::<i32>()).sum()
}

fn dot_q2k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(84).zip(xq) {
        let sc = &b[0..16];
        let mut a = [0i8; 256];
        for i in 0..256 { a[i] = (b[16 + 32 * (i / 128) + i % 32] >> (2 * (i / 32 % 4)) & 3) as i8; }
        let s = grouped(&a, &y.qs, |j| (sc[j] & 0xF) as i32);
        let m: i32 = (0..16).map(|j| (sc[j] >> 4) as i32 * y.bs[j] as i32).sum();
        sumf += y.d * d16(&b[80..]) * s as f32 - y.d * d16(&b[82..]) * m as f32;
    }
    sumf
}

fn dot_q3k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(110).zip(xq) {
        let (hm, sc) = (&b[0..32], crate::quant::q3k_scales(b));
        let mut a = [0i8; 256];
        for i in 0..256 {
            let k = i / 32 % 4 + 4 * (i / 128);
            a[i] = (b[32 + 32 * (i / 128) + i % 32] >> (2 * (k % 4)) & 3) as i8 - if hm[i % 32] >> k & 1 != 0 { 0 } else { 4 };
        }
        sumf += d16(&b[108..]) * y.d * grouped(&a, &y.qs, |j| sc[j] as i32 - 32) as f32;
    }
    sumf
}

/// Q2_0: 64 values (q - 1) * d with q in 0..4, four to a byte; four blocks per Q8_K block.
fn dot_q2_0(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(18).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 4], i % 4 * 64);
        let s: i32 = (0..64).map(|j| (b[2 + j / 4] >> (2 * (j % 4)) & 3) as i32 * y.qs[o + j] as i32).sum();
        let m: i32 = y.bs[o / 16..o / 16 + 4].iter().map(|&v| v as i32).sum();
        d16(b) * y.d * (s - m) as f32
    }).sum()
}

/// TQ1_0: five base-3 trits per byte in qs, four in qh; values -1, 0, 1 times d.
fn dot_tq1_0(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(54).zip(xq) {
        let trit = |v: u8, n: usize| { const POW3: [u8; 5] = [1, 3, 9, 27, 81]; ((v.wrapping_mul(POW3[n]) as u16 * 3) >> 8) as i32 - 1 };
        let mut s = 0i32;
        for n in 0..5 { for m in 0..32 { s += trit(b[m], n) * y.qs[32 * n + m] as i32; } }
        for n in 0..5 { for m in 0..16 { s += trit(b[32 + m], n) * y.qs[160 + 16 * n + m] as i32; } }
        for n in 0..4 { for j in 0..4 { s += trit(b[48 + j], n) * y.qs[240 + 4 * n + j] as i32; } }
        sumf += d16(&b[52..]) * y.d * s as f32;
    }
    sumf
}

/// TQ2_0: 64 bytes of 2-bit codes (code - 1 in -1..2), then f16 d; 256 values a block.
fn dot_tq2_0(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(66).zip(xq) {
        let mut s = 0i32;
        for j in 0..2 { for l in 0..4 { for m in 0..32 {
            let code = (b[32 * j + m] >> (2 * l)) & 3;
            s += (code as i32 - 1) * y.qs[128 * j + 32 * l + m] as i32;
        } } }
        sumf += d16(&b[64..]) * y.d * s as f32;
    }
    sumf
}

/// Q4_0 (`T` = 2, 18 bytes), Q4_1 (3, 20), Q5_0 (6, 22), Q5_1 (7, 24): block bytes, offset of the 32 high bits
/// (0 = none) and of the 16 bytes of nibbles.
const fn q4_layout(t: u8) -> (usize, usize, usize) { match t { 2 => (18, 0, 2), 3 => (20, 0, 4), 6 => (22, 2, 6), _ => (24, 4, 8) } }
/// The block's m in value = d * q + m (q the unsigned 4- or 5-bit code): its own for Q4_1 / Q5_1, -8 d or -16 d for
/// Q4_0 / Q5_0.
fn q4_min(t: u8, b: &[u8]) -> f32 { match t { 3 | 7 => d16(&b[2..]), 2 => -8.0 * d16(b), _ => -16.0 * d16(b) } }

/// Q4_0, Q4_1, Q5_0, Q5_1 (`q4_layout`): 32 values d * q + m per block, eight blocks per Q8_K block.
fn dot_q4<const T: u8>(row: &[u8], xq: &[Q8K]) -> f32 {
    let (bb, oh, oq) = q4_layout(T);
    row.chunks_exact(bb).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 8], i % 8 * 32);
        let (qh, qs) = if oh > 0 { (u32::from_le_bytes(b[oh..oh + 4].try_into().unwrap()), &b[oq..oq + 16]) } else { (0, &b[oq..oq + 16]) };
        let s: i32 = (0..16).map(|j| {
            let (l, h) = ((qs[j] & 0xF) as i32 | ((qh >> j) & 1) as i32 * 16, (qs[j] >> 4) as i32 | ((qh >> (j + 16)) & 1) as i32 * 16);
            l * y.qs[o + j] as i32 + h * y.qs[o + j + 16] as i32
        }).sum();
        y.d * (d16(b) * s as f32 + q4_min(T, b) * (y.bs[o / 16] as i32 + y.bs[o / 16 + 1] as i32) as f32)
    }).sum()
}

/// Q8_0: 32 values d * q per block, eight blocks per Q8_K block.
fn dot_q8_0(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(34).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 8], i % 8 * 32);
        let s: i32 = (0..32).map(|j| b[2 + j] as i8 as i32 * y.qs[o + j] as i32).sum();
        d16(b) * y.d * s as f32
    }).sum()
}

fn dot_q4k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(144).zip(xq) {
        let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
        let mut a = [0i8; 256];
        for i in 0..256 { a[i] = (b[16 + 32 * (i / 64) + i % 32] >> (4 * (i / 32 % 2)) & 0xF) as i8; }
        let s = grouped(&a, &y.qs, |j| sm[j / 2].0 as i32);
        let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
        sumf += d16(b) * y.d * s as f32 - d16(&b[2..]) * y.d * m as f32;
    }
    sumf
}

/// Q5_K: Q4_K plus a fifth bit per value from `qh` (bit 2 * (i / 64) + i / 32 % 2 of byte i % 32).
fn dot_q5k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(176).zip(xq) {
        let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
        let mut a = [0i8; 256];
        for j in 0..4 {
            for l in 0..32 {
                let (q, h) = (b[48 + 32 * j + l], b[16 + l] >> (2 * j));
                a[64 * j + l] = (q & 0xF | (h & 1) << 4) as i8;
                a[64 * j + 32 + l] = (q >> 4 | (h & 2) << 3) as i8;
            }
        }
        let s = grouped(&a, &y.qs, |j| sm[j / 2].0 as i32);
        let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
        sumf += d16(b) * y.d * s as f32 - d16(&b[2..]) * y.d * m as f32;
    }
    sumf
}

fn dot_q6k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(210).zip(xq) {
        let mut a = [0i8; 256];
        for n in 0..2 {
            let (ql, qh, a) = (&b[64 * n..], &b[128 + 32 * n..], &mut a[128 * n..]);
            for l in 0..32 {
                a[l] = ((ql[l] & 0xF) | ((qh[l] & 3) << 4)) as i8 - 32;
                a[l + 32] = ((ql[l + 32] & 0xF) | (((qh[l] >> 2) & 3) << 4)) as i8 - 32;
                a[l + 64] = ((ql[l] >> 4) | (((qh[l] >> 4) & 3) << 4)) as i8 - 32;
                a[l + 96] = ((ql[l + 32] >> 4) | ((qh[l] >> 6) << 4)) as i8 - 32;
            }
        }
        let s: i32 = (0..16).map(|j| b[192 + j] as i8 as i32 * (16 * j..16 * j + 16).map(|i| a[i] as i32 * y.qs[i] as i32).sum::<i32>()).sum();
        sumf += d16(&b[208..]) * y.d * s as f32;
    }
    sumf
}

/// AVX2 versions of the kernels above (llama.cpp `arch/x86/quants.c`).
#[cfg(target_feature = "avx2")]
mod avx2 {
    use super::{Q8K, SIGNS, d16, iq1m_d};
    use crate::quant::tables::*;
    use std::arch::x86_64::*;

    #[inline(always)]
    unsafe fn hsum(v: __m256) -> f32 { unsafe {
        let s = _mm_add_ps(_mm256_castps256_ps128(v), _mm256_extractf128_ps(v, 1));
        let s = _mm_add_ps(s, _mm_movehl_ps(s, s));
        _mm_cvtss_f32(_mm_add_ss(s, _mm_movehdup_ps(s)))
    } }
    /// qs = round(is * b), as the scalar path (|is * b| <= 127, so the saturating packs equal its min(127)).
    pub fn quant_block(is: f32, b: &[f32], qs: &mut [i8; 256]) { unsafe {
        let s = _mm256_set1_ps(is);
        let q = |i: usize| _mm256_cvtps_epi32(_mm256_round_ps(_mm256_mul_ps(s, _mm256_loadu_ps(b[i..i + 8].as_ptr())), _MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC));
        let perm = _mm256_setr_epi32(0, 4, 1, 5, 2, 6, 3, 7);
        for j in (0..256).step_by(32) {
            let p = _mm256_packs_epi16(_mm256_packs_epi32(q(j), q(j + 8)), _mm256_packs_epi32(q(j + 16), q(j + 24)));
            _mm256_storeu_si256(qs[j..j + 32].as_mut_ptr() as *mut __m256i, _mm256_permutevar8x32_epi32(p, perm));
        }
    } }
    #[inline(always)]
    fn sg(i: usize) -> i64 { unsafe { std::mem::transmute::<[i8; 8], i64>(SIGNS[i]) } }

    pub fn dot_iq2xs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(74).zip(xq) {
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let v = u64::from_le_bytes(b[2 + 8 * ib..10 + 8 * ib].try_into().unwrap());
                    let (q0, q1, q2, q3) = (v as u16, (v >> 16) as u16, (v >> 32) as u16, (v >> 48) as u16);
                    let g = |q: u16| IQ2XS_GRID[(q & 511) as usize] as i64;
                    let grid = _mm256_set_epi64x(g(q3), g(q2), g(q1), g(q0));
                    let signs = _mm256_set_epi64x(sg((q3 >> 9) as usize), sg((q2 >> 9) as usize), sg((q1 >> 9) as usize), sg((q0 >> 9) as usize));
                    let q8 = _mm256_loadu_si256(y.qs.as_ptr().add(32 * ib) as *const __m256i);
                    let dot = _mm256_maddubs_epi16(grid, _mm256_sign_epi8(q8, signs));
                    let sc = b[66 + ib];
                    let ls = _mm256_set_m128i(_mm_set1_epi16(2 * (sc >> 4) as i16 + 1), _mm_set1_epi16(2 * (sc & 0xF) as i16 + 1));
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(dot, ls));
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            0.125 * hsum(acc)
        }
    }

    pub fn dot_iq3xxs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(98).zip(xq) {
                let (qs, gas) = (&b[2..66], &b[66..98]);
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let v = u64::from_le_bytes(qs[8 * ib..8 * ib + 8].try_into().unwrap());
                    let g = |i: usize| IQ3XXS_GRID[(v >> (8 * i)) as u8 as usize] as i32;
                    let grid = _mm256_set_epi32(g(7), g(6), g(5), g(4), g(3), g(2), g(1), g(0));
                    let aux = u32::from_le_bytes(gas[4 * ib..4 * ib + 4].try_into().unwrap());
                    let s = |l: u32| sg(((aux >> (7 * l)) & 127) as usize);
                    let signs = _mm256_set_epi64x(s(3), s(2), s(1), s(0));
                    let q8 = _mm256_loadu_si256(y.qs.as_ptr().add(32 * ib) as *const __m256i);
                    let dot = _mm256_maddubs_epi16(grid, _mm256_sign_epi8(q8, signs));
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(dot, _mm256_set1_epi16(2 * (aux >> 28) as i16 + 1)));
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            0.25 * hsum(acc)
        }
    }

    pub fn dot_iq3s(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            // Byte k of a lane picks sign byte k / 8, then bit k % 8 (llama.cpp k_mask1 / k_mask2).
            let m1 = _mm256_set_epi64x(0x0303030303030303, 0x0202020202020202, 0x0101010101010101, 0);
            let m2 = _mm256_set1_epi64x(0x8040201008040201u64 as i64);
            let shl = _mm256_set_epi32(1, 2, 3, 4, 5, 6, 7, 8);
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(110).zip(xq) {
                let (qs, qh) = (&b[2..66], &b[66..74]);
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let lo = _mm256_cvtepu8_epi32(_mm_loadl_epi64(qs[8 * ib..].as_ptr() as *const __m128i));
                    let hi = _mm256_and_si256(_mm256_sllv_epi32(_mm256_set1_epi32(qh[ib] as i32), shl), _mm256_set1_epi32(256));
                    let grid = _mm256_i32gather_epi32(IQ3S_GRID.as_ptr() as *const i32, _mm256_or_si256(lo, hi), 4);
                    let sb = i32::from_le_bytes(b[74 + 4 * ib..78 + 4 * ib].try_into().unwrap());
                    let s = _mm256_cmpeq_epi8(_mm256_and_si256(_mm256_shuffle_epi8(_mm256_set1_epi32(sb), m1), m2), m2);
                    let q8 = _mm256_loadu_si256(y.qs.as_ptr().add(32 * ib) as *const __m256i);
                    let dot = _mm256_maddubs_epi16(grid, _mm256_sub_epi8(_mm256_xor_si256(s, q8), s));
                    let ls = 2 * (b[106 + ib / 2] >> (4 * (ib % 2)) & 0xF) as i16 + 1;
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(dot, _mm256_set1_epi16(ls)));
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            hsum(acc)
        }
    }

    pub fn dot_iq4xs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let values = _mm_loadu_si128(KVALUES_IQ4NL.as_ptr() as *const __m128i);
            let m4 = _mm_set1_epi8(0xF);
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(136).zip(xq) {
                let sh = u16::from_le_bytes([b[2], b[3]]);
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let bits = _mm_loadu_si128(b.as_ptr().add(8 + 16 * ib) as *const __m128i);
                    let w = _mm256_set_m128i(_mm_shuffle_epi8(values, _mm_and_si128(_mm_srli_epi16(bits, 4), m4)),
                                             _mm_shuffle_epi8(values, _mm_and_si128(bits, m4)));
                    let q8 = _mm256_loadu_si256(y.qs.as_ptr().add(32 * ib) as *const __m256i);
                    let p16 = _mm256_maddubs_epi16(_mm256_sign_epi8(w, w), _mm256_sign_epi8(q8, w));
                    let ls = ((b[4 + ib / 2] >> (4 * (ib % 2))) & 0xF) as i16 | ((((sh >> (2 * ib)) & 3) as i16) << 4);
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(p16, _mm256_set1_epi16(ls - 32)));
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            hsum(acc)
        }
    }

    /// Ternary `TQ1_0` (54 bytes = 256 trits in `{-1, 0, +1}`, f16 `d`): unpack `u_n = trit + 1` in `{0, 1, 2}`
    /// via `(3 * (v * 3^n mod 256)) >> 8` (`> 85` and `> 170`), dot against signed `Q8K` with `_mm256_maddubs_epi16`,
    /// and subtract `y.bs` once per block to account for the `-1` offset.
    pub fn dot_tq1_0(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let flip = _mm256_set1_epi8(-128i8);
            let th1 = _mm256_set1_epi8(-43i8);
            let th2 = _mm256_set1_epi8(42i8);
            let zero = _mm256_setzero_si256();
            let ones16 = _mm256_set1_epi16(1);
            let trits_u = |q: __m256i| -> __m256i {
                let qb = _mm256_xor_si256(q, flip);
                _mm256_sub_epi8(zero, _mm256_add_epi8(_mm256_cmpgt_epi8(qb, th1), _mm256_cmpgt_epi8(qb, th2)))
            };
            let mul3 = |q: __m256i| _mm256_add_epi8(_mm256_add_epi8(q, q), q);
            let mul3_128 = |q: __m128i| _mm_add_epi8(_mm_add_epi8(q, q), q);
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(54).zip(xq) {
                let mut sum16 = zero;
                let mut q0 = ld(b, 0);
                for n in 0..5 {
                    sum16 = _mm256_add_epi16(sum16, _mm256_maddubs_epi16(trits_u(q0), q8(y, 32 * n)));
                    if n < 4 { q0 = mul3(q0); }
                }
                let q1_0 = _mm_loadu_si128(b.as_ptr().add(32) as *const __m128i);
                let p01 = _mm256_set_m128i(mul3_128(q1_0), q1_0);
                sum16 = _mm256_add_epi16(sum16, _mm256_maddubs_epi16(trits_u(p01), q8(y, 160)));
                let p23 = mul3(mul3(p01));
                sum16 = _mm256_add_epi16(sum16, _mm256_maddubs_epi16(trits_u(p23), q8(y, 192)));
                let q1_4 = mul3_128(_mm256_extracti128_si256::<1>(p23));
                let bm = |m: u8| u32::from_le_bytes([b[48].wrapping_mul(m), b[49].wrapping_mul(m), b[50].wrapping_mul(m), b[51].wrapping_mul(m)]) as i32;
                let qh = _mm_set_epi32(bm(27), bm(9), bm(3), bm(1));
                let p4h = _mm256_set_m128i(qh, q1_4);
                sum16 = _mm256_add_epi16(sum16, _mm256_maddubs_epi16(trits_u(p4h), q8(y, 224)));
                let bs = _mm256_loadu_si256(y.bs.as_ptr() as *const __m256i);
                let sum32 = _mm256_madd_epi16(_mm256_sub_epi16(sum16, bs), ones16);
                acc = _mm256_fmadd_ps(_mm256_set1_ps(d16(&b[52..]) * y.d), _mm256_cvtepi32_ps(sum32), acc);
            }
            hsum(acc)
        }
    }

    /// NVFP4 (see `super::dot_nvfp4`): 64 values per 36-byte block across 4 sub-blocks of 16, processed two sub-blocks
    /// (32 values) per AVX2 step with `_mm_shuffle_epi8` + `_mm256_maddubs_epi16`.
    pub fn dot_nvfp4(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let values = _mm_loadu_si128(crate::quant::KVALUES_FP4.as_ptr() as *const __m128i);
            let (m4, ones) = (_mm_set1_epi8(0xF), _mm256_set1_epi16(1));
            let mut acc = _mm256_setzero_ps();
            for (i, b) in row.chunks_exact(36).enumerate() {
                let (y, o) = (&xq[i / 4], (i % 4) * 64);
                for h in 0..2 {
                    let s0 = 2 * h;
                    let bits0 = _mm_loadl_epi64(b.as_ptr().add(4 + 8 * s0) as *const __m128i);
                    let bits1 = _mm_loadl_epi64(b.as_ptr().add(12 + 8 * s0) as *const __m128i);
                    let w0 = _mm_unpacklo_epi64(
                        _mm_shuffle_epi8(values, _mm_and_si128(bits0, m4)),
                        _mm_shuffle_epi8(values, _mm_and_si128(_mm_srli_epi16(bits0, 4), m4)),
                    );
                    let w1 = _mm_unpacklo_epi64(
                        _mm_shuffle_epi8(values, _mm_and_si128(bits1, m4)),
                        _mm_shuffle_epi8(values, _mm_and_si128(_mm_srli_epi16(bits1, 4), m4)),
                    );
                    let w = _mm256_set_m128i(w1, w0);
                    let p16 = _mm256_maddubs_epi16(_mm256_sign_epi8(w, w), _mm256_sign_epi8(q8(y, o + 32 * h), w));
                    let p32 = _mm256_madd_epi16(p16, ones);
                    let d = _mm256_set_m128(
                        _mm_set1_ps(crate::quant::ue4m3(b[s0 + 1]) * y.d),
                        _mm_set1_ps(crate::quant::ue4m3(b[s0]) * y.d),
                    );
                    acc = _mm256_fmadd_ps(d, _mm256_cvtepi32_ps(p32), acc);
                }
            }
            hsum(acc)
        }
    }

    /// IQ4_NL (see `super::dot_iq4nl`): two 32-value blocks per step so the two lookups and FMAs overlap.
    pub fn dot_iq4nl(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let values = _mm_loadu_si128(KVALUES_IQ4NL.as_ptr() as *const __m128i);
            let (m4, ones) = (_mm_set1_epi8(0xF), _mm256_set1_epi16(1));
            let (mut acc0, mut acc1) = (_mm256_setzero_ps(), _mm256_setzero_ps());
            let blk = |b: &[u8], y: &Q8K, o: usize| {
                let bits = _mm_loadu_si128(b.as_ptr().add(2) as *const __m128i);
                let w = _mm256_set_m128i(_mm_shuffle_epi8(values, _mm_and_si128(_mm_srli_epi16(bits, 4), m4)),
                                         _mm_shuffle_epi8(values, _mm_and_si128(bits, m4)));
                let p16 = _mm256_maddubs_epi16(_mm256_sign_epi8(w, w), _mm256_sign_epi8(q8(y, o), w));
                (_mm256_cvtepi32_ps(_mm256_madd_epi16(p16, ones)), ds(b, y.d))
            };
            let mut it = row.chunks_exact(36).enumerate();
            for (i, b2) in &mut it {
                let (y, o) = (&xq[2 * i / 8], 2 * i % 8 * 32);
                let (s0, d0) = blk(&b2[..18], y, o);
                let (s1, d1) = blk(&b2[18..], y, o + 32);
                acc0 = _mm256_fmadd_ps(d0, s0, acc0);
                acc1 = _mm256_fmadd_ps(d1, s1, acc1);
            }
            let n = row.len() / 18;
            if n % 2 == 1 {
                let i = n - 1;
                let (s, d) = blk(&row[18 * i..18 * i + 18], &xq[i / 8], i % 8 * 32);
                acc0 = _mm256_fmadd_ps(d, s, acc0);
            }
            hsum(_mm256_add_ps(acc0, acc1))
        }
    }

    /// `dot_iq4nl` of one row with four activations (see `super::int_dot4`).
    pub fn dot4_iq4nl(row: &[u8], xq: [&[Q8K]; 4]) -> [f32; 4] {
        unsafe {
            let values = _mm_loadu_si128(KVALUES_IQ4NL.as_ptr() as *const __m128i);
            let (m4, ones) = (_mm_set1_epi8(0xF), _mm256_set1_epi16(1));
            // Sums of the even and of the odd blocks, as the two accumulators of `dot_iq4nl`.
            let (mut even, mut odd) = ([_mm256_setzero_ps(); 4], [_mm256_setzero_ps(); 4]);
            let blk = |b: &[u8], i: usize, acc: &mut [__m256; 4]| {
                let bits = _mm_loadu_si128(b.as_ptr().add(2) as *const __m128i);
                let w = _mm256_set_m128i(_mm_shuffle_epi8(values, _mm_and_si128(_mm_srli_epi16(bits, 4), m4)),
                                         _mm_shuffle_epi8(values, _mm_and_si128(bits, m4)));
                let aw = _mm256_sign_epi8(w, w);
                for k in 0..4 {
                    let y = &xq[k][i / 8];
                    let p16 = _mm256_maddubs_epi16(aw, _mm256_sign_epi8(q8(y, i % 8 * 32), w));
                    acc[k] = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(_mm256_madd_epi16(p16, ones)), acc[k]);
                }
            };
            let mut it = row.chunks_exact(36).enumerate();
            for (i, b2) in &mut it {
                blk(&b2[..18], 2 * i, &mut even);
                blk(&b2[18..], 2 * i + 1, &mut odd);
            }
            let n = row.len() / 18;
            if n % 2 == 1 { blk(&row[18 * (n - 1)..18 * n], n - 1, &mut even); }
            std::array::from_fn(|k| hsum(_mm256_add_ps(even[k], odd[k])))
        }
    }

    /// Pair sums `p` (16 i16) times `lo` (first 8, values 0..16) and `hi` (last 8), summed into 8 i32.
    #[inline(always)]
    unsafe fn udot_scale(p: __m256i, lo: i32, hi: i32) -> __m256i { unsafe {
        _mm256_madd_epi16(p, _mm256_set_m128i(_mm_set1_epi16(hi as i16), _mm_set1_epi16(lo as i16)))
    } }
    /// `(w · q8)` per 16 values times `lo` (first 16) and `hi` (last 16), for 32 weights in 0..=127.
    #[inline(always)]
    unsafe fn udot32(w: __m256i, q8: __m256i, lo: i32, hi: i32) -> __m256i { unsafe { udot_scale(_mm256_maddubs_epi16(w, q8), lo, hi) } }
    /// Eight copies of `d16(b) * d`, the same bits. A scalar f16 load compiles to `vpinsrw` into a register that the
    /// accumulator may hold, and then each block waits for the block before it; a broadcast has no such input.
    #[inline(always)]
    unsafe fn ds(b: &[u8], d: f32) -> __m256 { unsafe {
        #[cfg(target_feature = "f16c")]
        return _mm256_mul_ps(_mm256_cvtph_ps(_mm_set1_epi16(i16::from_le_bytes([b[0], b[1]]))), _mm256_set1_ps(d));
        #[cfg(not(target_feature = "f16c"))]
        _mm256_set1_ps(d16(b) * d)
    } }
    #[inline(always)]
    unsafe fn q8(y: &Q8K, i: usize) -> __m256i { unsafe { _mm256_loadu_si256(y.qs.as_ptr().add(i) as *const __m256i) } }
    #[inline(always)]
    unsafe fn ld(b: &[u8], i: usize) -> __m256i { unsafe { _mm256_loadu_si256(b[i..i + 32].as_ptr() as *const __m256i) } }
    /// Crumb `k` (bits 2k, 2k+1) of every byte.
    #[inline(always)]
    unsafe fn crumb(v: __m256i, k: usize) -> __m256i { unsafe {
        _mm256_and_si256(_mm256_srl_epi16(v, _mm_cvtsi32_si128(2 * k as i32)), _mm256_set1_epi8(3))
    } }

    pub fn dot_iq2xxs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(66).zip(xq) {
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let g = |l: usize| IQ2XXS_GRID[b[2 + 8 * ib + l] as usize] as i64;
                    let aux = u32::from_le_bytes(b[6 + 8 * ib..10 + 8 * ib].try_into().unwrap());
                    let s = |l: u32| sg(((aux >> (7 * l)) & 127) as usize);
                    let grid = _mm256_set_epi64x(g(3), g(2), g(1), g(0));
                    let dot = _mm256_maddubs_epi16(grid, _mm256_sign_epi8(q8(y, 32 * ib), _mm256_set_epi64x(s(3), s(2), s(1), s(0))));
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(dot, _mm256_set1_epi16(2 * (aux >> 28) as i16 + 1)));
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            0.125 * hsum(acc)
        }
    }

    pub fn dot_iq2s(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            // As in dot_iq3s: byte k of a lane picks sign byte k / 8, then bit k % 8.
            let m1 = _mm256_set_epi64x(0x0303030303030303, 0x0202020202020202, 0x0101010101010101, 0);
            let m2 = _mm256_set1_epi64x(0x8040201008040201u64 as i64);
            // Grid index high bits: word p of the 32 indices takes qh[p / 4] << (8 - 2 * (p % 4)) & 0x300.
            let hq0 = _mm256_set_epi64x(0x8003800380038003u64 as i64, 0x8002800280028002u64 as i64, 0x8001800180018001u64 as i64, 0x8000800080008000u64 as i64);
            let hq1 = _mm256_add_epi16(hq0, _mm256_set1_epi16(4));
            let hmul = _mm256_set1_epi64x(0x0004_0010_0040_0100);
            let h3 = _mm256_set1_epi16(0x300);
            // Scale words: the low lane takes byte 2 ib of the 16 block scales, the high lane byte 2 ib + 1.
            let sm0 = _mm256_set_epi64x(0x8001800180018001u64 as i64, 0x8001800180018001u64 as i64, 0x8000800080008000u64 as i64, 0x8000800080008000u64 as i64);
            let (sstep, n15) = (_mm256_set1_epi16(2), _mm_set1_epi8(15));
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(82).zip(xq) {
                let qh = _mm256_broadcastsi128_si256(_mm_loadl_epi64(b[66..74].as_ptr() as *const __m128i));
                let mut idx = [0u16; 32];
                for (h, hq) in [hq0, hq1].into_iter().enumerate() {
                    let lo = _mm256_cvtepu8_epi16(_mm_loadu_si128(b[2 + 16 * h..18 + 16 * h].as_ptr() as *const __m128i));
                    let hi = _mm256_and_si256(_mm256_mullo_epi16(_mm256_shuffle_epi8(qh, hq), hmul), h3);
                    _mm256_storeu_si256(idx[16 * h..].as_mut_ptr() as *mut __m256i, _mm256_or_si256(lo, hi));
                }
                let sc = _mm_loadl_epi64(b[74..82].as_ptr() as *const __m128i);
                let sc = _mm_unpacklo_epi8(_mm_and_si128(sc, n15), _mm_and_si128(_mm_srli_epi16(sc, 4), n15));
                let sc = _mm256_broadcastsi128_si256(_mm_add_epi8(_mm_add_epi8(sc, sc), _mm_set1_epi8(1)));
                let (mut sm, mut sumi) = (sm0, _mm256_setzero_si256());
                for ib in 0..8 {
                    let g = |l: usize| IQ2S_GRID[idx[4 * ib + l] as usize & 1023] as i64;
                    let grid = _mm256_set_epi64x(g(3), g(2), g(1), g(0));
                    let sb = i32::from_le_bytes(b[34 + 4 * ib..38 + 4 * ib].try_into().unwrap());
                    let s = _mm256_cmpeq_epi8(_mm256_and_si256(_mm256_shuffle_epi8(_mm256_set1_epi32(sb), m1), m2), m2);
                    let dot = _mm256_maddubs_epi16(grid, _mm256_sub_epi8(_mm256_xor_si256(s, q8(y, 32 * ib)), s));
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(dot, _mm256_shuffle_epi8(sc, sm)));
                    sm = _mm256_add_epi16(sm, sstep);
                }
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            0.125 * hsum(acc)
        }
    }

    /// Products of 32 IQ1 grid values (-1, 0, 1) with q8, in pairs (maddubs needs an unsigned first operand).
    #[inline(always)]
    unsafe fn dot1(g: __m256i, q: __m256i) -> __m256i { unsafe { _mm256_maddubs_epi16(_mm256_sign_epi8(g, g), _mm256_sign_epi8(q, g)) } }

    pub fn dot_iq1s(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            let mut sumf = 0.0;
            for (b, y) in row.chunks_exact(50).zip(xq) {
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let qh = u16::from_le_bytes([b[34 + 2 * ib], b[35 + 2 * ib]]) as usize;
                    let g = |l: usize| IQ1S_GRID[b[2 + 4 * ib + l] as usize | ((qh >> (3 * l)) & 7) << 8] as i64;
                    let ls = 2 * ((qh >> 12) & 7) as i32 + 1;
                    let p = dot1(_mm256_set_epi64x(g(3), g(2), g(1), g(0)), q8(y, 32 * ib));
                    sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(p, _mm256_set1_epi16(ls as i16)));
                }
                // Delta term: sum of ±scale (sign in bit 15 of qh) times each 32 values' q8 sum.
                let qh = _mm_loadu_si128(b[34..50].as_ptr() as *const __m128i);
                let ls = _mm_or_si128(_mm_slli_epi16(_mm_and_si128(_mm_srli_epi16(qh, 12), _mm_set1_epi16(7)), 1), _mm_set1_epi16(1));
                let ls = _mm_sign_epi16(ls, _mm_or_si128(qh, _mm_set1_epi16(1)));
                let bs = |i: usize| _mm_loadu_si128(y.bs[8 * i..].as_ptr() as *const __m128i);
                let s = _mm_add_epi32(_mm_madd_epi16(_mm_unpacklo_epi16(ls, ls), bs(0)), _mm_madd_epi16(_mm_unpackhi_epi16(ls, ls), bs(1)));
                let s = _mm_add_epi32(s, _mm_shuffle_epi32(s, 0b01_00_11_10));
                let s1 = _mm_cvtsi128_si32(_mm_add_epi32(s, _mm_shuffle_epi32(s, 0b10_11_00_01)));
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(sumi), acc);
                sumf += d16(b) * y.d * 0.125 * s1 as f32;
            }
            hsum(acc) + sumf
        }
    }

    /// IQ1_M weights plus 9 (8 * grid + 8, plus 2 when the delta is +1), indexed by the 8 grid bits, then 3 high grid
    /// bits and the delta sign bit (a qh nibble).
    static IQ1M_W: std::sync::LazyLock<Box<[u64; 4096]>> = std::sync::LazyLock::new(|| {
        let mut t = Box::new([0u64; 4096]);
        for (i, w) in t.iter_mut().enumerate() {
            let d = if i & 2048 != 0 { 8 } else { 10 };
            *w = u64::from_le_bytes(IQ1S_GRID[i & 2047].to_le_bytes().map(|g| (8 * g as i8 + d) as u8));
        }
        t
    });

    pub fn dot_iq1m(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (t, nine) = (&**IQ1M_W, _mm256_set1_epi8(9));
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(56).zip(xq) {
                // Weights 8 * grid + delta (-9..=9), so one 0.125 at the end gives grid + delta / 8. The table holds
                // them plus 9 (unsigned for maddubs), taken off the pair sums again with 9 * q8.
                let sc = u64::from_le_bytes(b[48..56].try_into().unwrap());
                let ls: [i16; 16] = std::array::from_fn(|i| 2 * ((sc >> (16 * (i / 4) + 3 * (i % 4))) & 7) as i16 + 1);
                let mut sumi = _mm256_setzero_si256();
                for ib in 0..8 {
                    let ls = |h: usize| ls[2 * ib + h] as i32;
                    let qh = u16::from_le_bytes([b[32 + 2 * ib], b[33 + 2 * ib]]) as usize;
                    let g = |l: usize| t[b[4 * ib + l] as usize | ((qh >> (4 * l)) & 15) << 8] as i64;
                    let q = q8(y, 32 * ib);
                    let p = _mm256_sub_epi16(_mm256_maddubs_epi16(_mm256_set_epi64x(g(3), g(2), g(1), g(0)), q), _mm256_maddubs_epi16(nine, q));
                    sumi = _mm256_add_epi32(sumi, udot_scale(p, ls(0), ls(1)));
                }
                acc = _mm256_fmadd_ps(_mm256_set1_ps(iq1m_d(b) * y.d), _mm256_cvtepi32_ps(sumi), acc);
            }
            0.125 * hsum(acc)
        }
    }

    pub fn dot_q2k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(84).zip(xq) {
                let sc = &b[0..16];
                let mut sumi = _mm256_setzero_si256();
                for n in 0..2 {
                    let bits = ld(b, 16 + 32 * n);
                    for k in 0..4 {
                        let g = 8 * n + 2 * k;
                        sumi = _mm256_add_epi32(sumi, udot32(crumb(bits, k), q8(y, 32 * (4 * n + k)), (sc[g] & 0xF) as i32, (sc[g + 1] & 0xF) as i32));
                    }
                }
                let mins = _mm256_cvtepu8_epi16(_mm_and_si128(_mm_srli_epi16(_mm_loadu_si128(sc.as_ptr() as *const __m128i), 4), _mm_set1_epi8(0xF)));
                let m = _mm256_madd_epi16(mins, _mm256_loadu_si256(y.bs.as_ptr() as *const __m256i));
                acc = _mm256_fmadd_ps(_mm256_set1_ps(y.d * d16(&b[80..])), _mm256_cvtepi32_ps(sumi), acc);
                acc = _mm256_fmadd_ps(_mm256_set1_ps(-y.d * d16(&b[82..])), _mm256_cvtepi32_ps(m), acc);
            }
            hsum(acc)
        }
    }

    pub fn dot_q3k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let mut acc = _mm256_setzero_ps();
            for (b, y) in row.chunks_exact(110).zip(xq) {
                let (hm, sc) = (ld(b, 0), crate::quant::q3k_scales(b));
                let mut sumi = _mm256_setzero_si256();
                for n in 0..2 {
                    let bits = ld(b, 32 + 32 * n);
                    for k in 0..4 {
                        let kk = 4 * n + k;
                        // Weight = crumb - 4 where the high bit is clear; signs moved onto q8 for maddubs.
                        let mask = _mm256_set1_epi8(1 << kk);
                        let low = _mm256_cmpeq_epi8(_mm256_and_si256(hm, mask), _mm256_setzero_si256());
                        let w = _mm256_sub_epi8(crumb(bits, k), _mm256_and_si256(low, _mm256_set1_epi8(4)));
                        let p = _mm256_maddubs_epi16(_mm256_sign_epi8(w, w), _mm256_sign_epi8(q8(y, 32 * kk), w));
                        let s = _mm256_set_m128i(_mm_set1_epi16(sc[2 * kk + 1] as i16 - 32), _mm_set1_epi16(sc[2 * kk] as i16 - 32));
                        sumi = _mm256_add_epi32(sumi, _mm256_madd_epi16(p, s));
                    }
                }
                acc = _mm256_fmadd_ps(_mm256_set1_ps(y.d * d16(&b[108..])), _mm256_cvtepi32_ps(sumi), acc);
            }
            hsum(acc)
        }
    }

    /// The 64 crumbs of a Q2_0 block in element order, as two vectors of 32 bytes.
    #[cfg(all(target_feature = "avx512vbmi", target_feature = "avx512vl"))]
    #[inline(always)]
    unsafe fn q2_0_crumbs(b: &[u8]) -> (__m256i, __m256i) { unsafe {
        // Output byte i of qword t takes bits 16 t + 2 i of the source qword: element 8 t + i of 32.
        let ctl = _mm256_set_epi64x(0x3e3c3a3836343230, 0x2e2c2a2826242220, 0x1e1c1a1816141210, 0x0e0c0a0806040200);
        let h = |o: usize| _mm256_and_si256(_mm256_multishift_epi64_epi8(ctl, _mm256_set1_epi64x(i64::from_le_bytes(b[o..o + 8].try_into().unwrap()))), _mm256_set1_epi8(3));
        (h(2), h(10))
    } }
    #[cfg(not(all(target_feature = "avx512vbmi", target_feature = "avx512vl")))]
    #[inline(always)]
    unsafe fn q2_0_crumbs(b: &[u8]) -> (__m256i, __m256i) { unsafe {
        let v = _mm_loadu_si128(b[2..18].as_ptr() as *const __m128i);
        let c = |k: i32| _mm_and_si128(_mm_srl_epi16(v, _mm_cvtsi32_si128(2 * k)), _mm_set1_epi8(3));
        let (c0, c1, c2, c3) = (c(0), c(1), c(2), c(3));
        // Re-interleave the crumbs into element order: byte k holds elements 4k..4k+3.
        let (lo01, lo23) = (_mm_unpacklo_epi8(c0, c1), _mm_unpacklo_epi8(c2, c3));
        let (hi01, hi23) = (_mm_unpackhi_epi8(c0, c1), _mm_unpackhi_epi8(c2, c3));
        (_mm256_set_m128i(_mm_unpackhi_epi16(lo01, lo23), _mm_unpacklo_epi16(lo01, lo23)),
         _mm256_set_m128i(_mm_unpackhi_epi16(hi01, hi23), _mm_unpacklo_epi16(hi01, hi23)))
    } }
    pub fn dot_q2_0(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (mut acc, mut sumf) = (_mm256_setzero_ps(), 0f32);
            let ones = _mm256_set1_epi16(1);
            for (sb, y) in row.chunks(72).zip(xq) {
                // The sums of y over each 64 values, in lanes 0, 1, 4 and 5.
                let bs = _mm256_madd_epi16(_mm256_loadu_si256(y.bs.as_ptr() as *const __m256i), ones);
                let mut m = [0f32; 8];
                _mm256_storeu_ps(m.as_mut_ptr(), _mm256_cvtepi32_ps(_mm256_hadd_epi32(bs, bs)));
                for (j, b) in sb.chunks_exact(18).enumerate() {
                    let (w0, w1) = q2_0_crumbs(b);
                    let p = _mm256_add_epi16(_mm256_maddubs_epi16(w0, q8(y, 64 * j)), _mm256_maddubs_epi16(w1, q8(y, 64 * j + 32)));
                    let d = d16(b) * y.d;
                    acc = _mm256_fmadd_ps(_mm256_set1_ps(d), _mm256_cvtepi32_ps(_mm256_madd_epi16(p, ones)), acc);
                    sumf -= d * m[j + j / 2 * 2];
                }
            }
            hsum(acc) + sumf
        }
    }

    /// Q4_0, Q4_1, Q5_0, Q5_1 (scalar `dot_q4`).
    pub fn dot_q4<const T: u8>(row: &[u8], xq: &[Q8K]) -> f32 {
        let (bb, oh, oq) = super::q4_layout(T);
        unsafe {
            let (m4, ones, mut acc, mut sumf) = (_mm_set1_epi8(0xF), _mm256_set1_epi16(1), _mm256_setzero_ps(), 0f32);
            for (i, b) in row.chunks_exact(bb).enumerate() {
                let (y, o) = (&xq[i / 8], i % 8 * 32);
                let q = _mm_loadu_si128(b[oq..oq + 16].as_ptr() as *const __m128i);
                let mut w = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(q, 4), m4), _mm_and_si128(q, m4));
                if oh > 0 {  // bit j of qh as 16 in byte j (llama.cpp bytes_from_bits_32)
                    let qh = i32::from_le_bytes(b[oh..oh + 4].try_into().unwrap());
                    let sh = _mm256_shuffle_epi8(_mm256_set1_epi32(qh), _mm256_set_epi64x(0x0303030303030303, 0x0202020202020202, 0x0101010101010101, 0));
                    let set = _mm256_cmpeq_epi8(_mm256_or_si256(sh, _mm256_set1_epi64x(0x7fbfdfeff7fbfdfe)), _mm256_set1_epi64x(-1));
                    w = _mm256_or_si256(w, _mm256_and_si256(set, _mm256_set1_epi8(16)));
                }
                let s = _mm256_madd_epi16(_mm256_maddubs_epi16(w, q8(y, o)), ones);
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(s), acc);
                sumf += super::q4_min(T, b) * y.d * (y.bs[o / 16] as i32 + y.bs[o / 16 + 1] as i32) as f32;
            }
            hsum(acc) + sumf
        }
    }

    pub fn dot_q4k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let m4 = _mm256_set1_epi8(0xF);
            let mut acc = _mm256_setzero_ps();
            let mut sumf = 0.0;
            for (b, y) in row.chunks_exact(144).zip(xq) {
                let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
                let mut sumi = _mm256_setzero_si256();
                for j in 0..4 {
                    let bits = ld(b, 16 + 32 * j);
                    let (s0, s1) = (sm[2 * j].0 as i32, sm[2 * j + 1].0 as i32);
                    sumi = _mm256_add_epi32(sumi, udot32(_mm256_and_si256(bits, m4), q8(y, 64 * j), s0, s0));
                    sumi = _mm256_add_epi32(sumi, udot32(_mm256_and_si256(_mm256_srli_epi16(bits, 4), m4), q8(y, 64 * j + 32), s1, s1));
                }
                let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
                acc = _mm256_fmadd_ps(_mm256_set1_ps(y.d * d16(b)), _mm256_cvtepi32_ps(sumi), acc);
                sumf -= y.d * d16(&b[2..]) * m as f32;
            }
            hsum(acc) + sumf
        }
    }

    /// Q8_0 (scalar `dot_q8_0`): |w| times q8 with w's sign, as maddubs needs an unsigned first operand.
    pub fn dot_q8_0(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (ones, mut acc) = (_mm256_set1_epi16(1), _mm256_setzero_ps());
            for (i, b) in row.chunks_exact(34).enumerate() {
                let (y, o) = (&xq[i / 8], i % 8 * 32);
                let w = _mm256_loadu_si256(b[2..34].as_ptr() as *const __m256i);
                let p = _mm256_maddubs_epi16(_mm256_sign_epi8(w, w), _mm256_sign_epi8(q8(y, o), w));
                acc = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(_mm256_madd_epi16(p, ones)), acc);
            }
            hsum(acc)
        }
    }

    /// `dot_q8_0` of one row with four activations (see `super::int_dot4`).
    pub fn dot4_q8_0(row: &[u8], xq: [&[Q8K]; 4]) -> [f32; 4] {
        unsafe {
            let (ones, mut acc) = (_mm256_set1_epi16(1), [_mm256_setzero_ps(); 4]);
            for (i, b) in row.chunks_exact(34).enumerate() {
                let w = _mm256_loadu_si256(b[2..34].as_ptr() as *const __m256i);
                let aw = _mm256_sign_epi8(w, w);
                for k in 0..4 {
                    let y = &xq[k][i / 8];
                    let p = _mm256_maddubs_epi16(aw, _mm256_sign_epi8(q8(y, i % 8 * 32), w));
                    acc[k] = _mm256_fmadd_ps(ds(b, y.d), _mm256_cvtepi32_ps(_mm256_madd_epi16(p, ones)), acc[k]);
                }
            }
            acc.map(|a| hsum(a))
        }
    }

    pub fn dot_q5k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (m4, one) = (_mm256_set1_epi8(0xF), _mm256_set1_epi8(1));
            // Bit k of every qh byte as 16.
            let hb = |qh: __m256i, k: i32| _mm256_slli_epi16(_mm256_and_si256(_mm256_srl_epi16(qh, _mm_cvtsi32_si128(k)), one), 4);
            let mut acc = _mm256_setzero_ps();
            let mut sumf = 0.0;
            for (b, y) in row.chunks_exact(176).zip(xq) {
                let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
                let qh = ld(b, 16);
                let mut sumi = _mm256_setzero_si256();
                for j in 0..4 {
                    let bits = ld(b, 48 + 32 * j);
                    let (s0, s1) = (sm[2 * j].0 as i32, sm[2 * j + 1].0 as i32);
                    let w0 = _mm256_or_si256(_mm256_and_si256(bits, m4), hb(qh, 2 * j as i32));
                    let w1 = _mm256_or_si256(_mm256_and_si256(_mm256_srli_epi16(bits, 4), m4), hb(qh, 2 * j as i32 + 1));
                    sumi = _mm256_add_epi32(sumi, udot32(w0, q8(y, 64 * j), s0, s0));
                    sumi = _mm256_add_epi32(sumi, udot32(w1, q8(y, 64 * j + 32), s1, s1));
                }
                let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
                acc = _mm256_fmadd_ps(_mm256_set1_ps(y.d * d16(b)), _mm256_cvtepi32_ps(sumi), acc);
                sumf -= y.d * d16(&b[2..]) * m as f32;
            }
            hsum(acc) + sumf
        }
    }

    /// Q6_K with unsigned 6-bit codes (maddubs needs an unsigned first operand); the -32 offset comes off through
    /// the Q8_K group sums.
    pub fn dot_q6k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let m4 = _mm256_set1_epi8(0xF);
            let mut acc = _mm256_setzero_ps();
            let mut sumf = 0.0;
            for (b, y) in row.chunks_exact(210).zip(xq) {
                let sc = |g: usize| b[192 + g] as i8 as i32;
                let mut sumi = _mm256_setzero_si256();
                for n in 0..2 {
                    let (la, lb, qh) = (ld(b, 64 * n), ld(b, 64 * n + 32), ld(b, 128 + 32 * n));
                    let hi = |k: usize| _mm256_slli_epi16(crumb(qh, k), 4);
                    let w = [
                        _mm256_or_si256(_mm256_and_si256(la, m4), hi(0)),
                        _mm256_or_si256(_mm256_and_si256(lb, m4), hi(1)),
                        _mm256_or_si256(_mm256_and_si256(_mm256_srli_epi16(la, 4), m4), hi(2)),
                        _mm256_or_si256(_mm256_and_si256(_mm256_srli_epi16(lb, 4), m4), hi(3)),
                    ];
                    for (k, w) in w.into_iter().enumerate() {
                        let g = 8 * n + 2 * k;
                        sumi = _mm256_add_epi32(sumi, udot32(w, q8(y, 128 * n + 32 * k), sc(g), sc(g + 1)));
                    }
                }
                let m: i32 = (0..16).map(|g| sc(g) * y.bs[g] as i32).sum();
                let d = y.d * d16(&b[208..]);
                acc = _mm256_fmadd_ps(_mm256_set1_ps(d), _mm256_cvtepi32_ps(sumi), acc);
                sumf -= 32.0 * d * m as f32;
            }
            hsum(acc) + sumf
        }
    }
}

#[cfg(all(target_arch = "x86_64", target_feature = "avx512f", target_feature = "avx512bw", target_feature = "avx512vnni", target_feature = "avx512vbmi"))]
mod avx512 {
    use super::{Q8K, d16};
    use crate::quant::tables::*;
    use std::arch::x86_64::*;

    #[inline(always)]
    unsafe fn q8_512(y: &Q8K, i: usize) -> __m512i { unsafe { _mm512_loadu_si512(y.qs.as_ptr().add(i) as *const __m512i) } }
    #[inline(always)]
    unsafe fn mul3_512(v: __m512i) -> __m512i { unsafe { _mm512_add_epi8(_mm512_add_epi8(v, v), v) } }
    #[inline(always)]
    unsafe fn mul3_256(v: __m256i) -> __m256i { unsafe { _mm256_add_epi8(_mm256_add_epi8(v, v), v) } }
    #[inline(always)]
    unsafe fn mul3_128(v: __m128i) -> __m128i { unsafe { _mm_add_epi8(_mm_add_epi8(v, v), v) } }

    /// Q8_0 (2 blocks = 64 values per 512-bit `vpdpbusd`).
    pub fn dot_q8_0(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let zero = _mm512_setzero_si512();
            let mut acc = _mm512_setzero_ps();
            for (j, b2) in row.chunks_exact(68).enumerate() {
                let (y, o) = (&xq[j / 4], (j % 4) * 64);
                let w0 = _mm256_loadu_si256(b2[2..34].as_ptr() as *const __m256i);
                let w1 = _mm256_loadu_si256(b2[36..68].as_ptr() as *const __m256i);
                let w = _mm512_inserti64x4(_mm512_castsi256_si512(w0), w1, 1);
                let q = q8_512(y, o);
                let neg = _mm512_movepi8_mask(w);
                let p32 = _mm512_dpbusd_epi32(zero, _mm512_abs_epi8(w), _mm512_mask_sub_epi8(q, neg, zero, q));
                let dv = _mm512_mask_blend_ps(0xFF00, _mm512_set1_ps(d16(b2) * y.d), _mm512_set1_ps(d16(&b2[34..]) * y.d));
                acc = _mm512_fmadd_ps(dv, _mm512_cvtepi32_ps(p32), acc);
            }
            let mut sum = _mm512_reduce_add_ps(acc);
            let n = row.len() / 34;
            if n % 2 == 1 {
                let i = n - 1;
                let b = &row[34 * i..34 * i + 34];
                let (y, o) = (&xq[i / 8], (i % 8) * 32);
                let s: i32 = b[2..34].iter().zip(&y.qs[o..o + 32]).map(|(&w, &q)| w as i8 as i32 * q as i32).sum();
                sum += d16(b) * y.d * s as f32;
            }
            sum
        }
    }

    /// Q4_0, Q4_1, Q5_0, Q5_1: two 32-value blocks per 512-bit `vpdpbusd` + 64-bit mask high-bit expansion.
    pub fn dot_q4<const T: u8>(row: &[u8], xq: &[Q8K]) -> f32 {
        let (bb, oh, oq) = super::q4_layout(T);
        unsafe {
            let (m4, sixteen, zero) = (_mm_set1_epi8(0xF), 16i8, _mm512_setzero_si512());
            let mut acc = _mm512_setzero_ps();
            let mut sumf = 0.0f32;
            for (j, b2) in row.chunks_exact(2 * bb).enumerate() {
                let (b0, b1) = (&b2[..bb], &b2[bb..]);
                let (y, o) = (&xq[j / 4], (j % 4) * 64);
                let q0 = _mm_loadu_si128(b0[oq..oq + 16].as_ptr() as *const __m128i);
                let q1 = _mm_loadu_si128(b1[oq..oq + 16].as_ptr() as *const __m128i);
                let w0 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(q0, 4), m4), _mm_and_si128(q0, m4));
                let w1 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(q1, 4), m4), _mm_and_si128(q1, m4));
                let mut w = _mm512_inserti64x4(_mm512_castsi256_si512(w0), w1, 1);
                if oh > 0 {
                    let qh0 = u32::from_le_bytes(b0[oh..oh + 4].try_into().unwrap()) as u64;
                    let qh1 = u32::from_le_bytes(b1[oh..oh + 4].try_into().unwrap()) as u64;
                    w = _mm512_or_si512(w, _mm512_maskz_set1_epi8(qh0 | (qh1 << 32), sixteen));
                }
                let p32 = _mm512_dpbusd_epi32(zero, w, q8_512(y, o));
                let dv = _mm512_mask_blend_ps(0xFF00, _mm512_set1_ps(d16(b0) * y.d), _mm512_set1_ps(d16(b1) * y.d));
                acc = _mm512_fmadd_ps(dv, _mm512_cvtepi32_ps(p32), acc);
                let bs0 = (y.bs[o / 16] as i32 + y.bs[o / 16 + 1] as i32) as f32;
                let bs1 = (y.bs[o / 16 + 2] as i32 + y.bs[o / 16 + 3] as i32) as f32;
                sumf += y.d * (super::q4_min(T, b0) * bs0 + super::q4_min(T, b1) * bs1);
            }
            let mut sum = _mm512_reduce_add_ps(acc) + sumf;
            let n = row.len() / bb;
            if n % 2 == 1 {
                let i = n - 1;
                let b = &row[bb * i..bb * i + bb];
                let (y, o) = (&xq[i / 8], (i % 8) * 32);
                let mut s = 0i32;
                let qh = if oh > 0 { u32::from_le_bytes(b[oh..oh + 4].try_into().unwrap()) } else { 0 };
                for l in 0..16 {
                    s += ((b[oq + l] & 0xF) as i32 | ((qh as i32 >> l) & 1) << 4) * y.qs[o + l] as i32;
                    s += ((b[oq + l] >> 4) as i32 | ((qh as i32 >> (l + 16)) & 1) << 4) * y.qs[o + 16 + l] as i32;
                }
                sum += d16(b) * y.d * s as f32 + super::q4_min(T, b) * y.d * (y.bs[o / 16] as i32 + y.bs[o / 16 + 1] as i32) as f32;
            }
            sum
        }
    }

    /// IQ4_NL: two 32-value blocks via broadcasted 512-bit `vpshufb` + `vpdpbusd`.
    pub fn dot_iq4nl(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let values = _mm512_broadcast_i32x4(_mm_loadu_si128(KVALUES_IQ4NL.as_ptr() as *const __m128i));
            let (m4, zero) = (_mm_set1_epi8(0xF), _mm512_setzero_si512());
            let mut acc = _mm512_setzero_ps();
            for (j, b2) in row.chunks_exact(36).enumerate() {
                let (b0, b1) = (&b2[..18], &b2[18..]);
                let (y, o) = (&xq[j / 4], (j % 4) * 64);
                let bits0 = _mm_loadu_si128(b0[2..].as_ptr() as *const __m128i);
                let bits1 = _mm_loadu_si128(b1[2..].as_ptr() as *const __m128i);
                let idx0 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(bits0, 4), m4), _mm_and_si128(bits0, m4));
                let idx1 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(bits1, 4), m4), _mm_and_si128(bits1, m4));
                let w = _mm512_shuffle_epi8(values, _mm512_inserti64x4(_mm512_castsi256_si512(idx0), idx1, 1));
                let q = q8_512(y, o);
                let neg = _mm512_movepi8_mask(w);
                let p32 = _mm512_dpbusd_epi32(zero, _mm512_abs_epi8(w), _mm512_mask_sub_epi8(q, neg, zero, q));
                let dv = _mm512_mask_blend_ps(0xFF00, _mm512_set1_ps(d16(b0) * y.d), _mm512_set1_ps(d16(b1) * y.d));
                acc = _mm512_fmadd_ps(dv, _mm512_cvtepi32_ps(p32), acc);
            }
            let mut sum = _mm512_reduce_add_ps(acc);
            let n = row.len() / 18;
            if n % 2 == 1 {
                let i = n - 1;
                let b = &row[18 * i..18 * i + 18];
                let (y, o) = (&xq[i / 8], (i % 8) * 32);
                let mut s = 0i32;
                for l in 0..16 {
                    s += KVALUES_IQ4NL[(b[2 + l] & 0xF) as usize] as i32 * y.qs[o + l] as i32;
                    s += KVALUES_IQ4NL[(b[2 + l] >> 4) as usize] as i32 * y.qs[o + 16 + l] as i32;
                }
                sum += d16(b) * y.d * s as f32;
            }
            sum
        }
    }

    /// IQ4_XS: 4 pairs of 32-value sub-blocks per super-block via 512-bit `vpshufb` + `vpdpbusd`.
    pub fn dot_iq4xs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let values = _mm512_broadcast_i32x4(_mm_loadu_si128(KVALUES_IQ4NL.as_ptr() as *const __m128i));
            let (m4, zero) = (_mm_set1_epi8(0xF), _mm512_setzero_si512());
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(136).zip(xq) {
                let sh = u16::from_le_bytes([b[2], b[3]]);
                let mut sumi = zero;
                for p in 0..4 {
                    let ib0 = 2 * p;
                    let ib1 = ib0 + 1;
                    let bits0 = _mm_loadu_si128(b.as_ptr().add(8 + 16 * ib0) as *const __m128i);
                    let bits1 = _mm_loadu_si128(b.as_ptr().add(8 + 16 * ib1) as *const __m128i);
                    let idx0 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(bits0, 4), m4), _mm_and_si128(bits0, m4));
                    let idx1 = _mm256_set_m128i(_mm_and_si128(_mm_srli_epi16(bits1, 4), m4), _mm_and_si128(bits1, m4));
                    let w = _mm512_shuffle_epi8(values, _mm512_inserti64x4(_mm512_castsi256_si512(idx0), idx1, 1));
                    let q = q8_512(y, 64 * p);
                    let neg = _mm512_movepi8_mask(w);
                    let d32 = _mm512_dpbusd_epi32(zero, _mm512_abs_epi8(w), _mm512_mask_sub_epi8(q, neg, zero, q));
                    let sc = |ib: usize| (((b[4 + ib / 2] >> (4 * (ib % 2))) & 0xF) as i32 | ((((sh >> (2 * ib)) & 3) as i32) << 4)) - 32;
                    let sv = _mm512_mask_set1_epi32(_mm512_set1_epi32(sc(ib0)), 0xFF00, sc(ib1));
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(d32, sv));
                }
                sumf += d16(b) * y.d * _mm512_reduce_add_epi32(sumi) as f32;
            }
            sumf
        }
    }

    /// Q4_K: 4 x 64 weights per super-block using 512-bit `vpdpbusd`.
    pub fn dot_q4k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (m4, zero) = (_mm512_set1_epi8(0xF), _mm512_setzero_si512());
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(144).zip(xq) {
                let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
                let mut sumi = zero;
                for j in 0..4 {
                    let bits256 = _mm256_loadu_si256(b[16 + 32 * j..].as_ptr() as *const __m256i);
                    let bits = _mm512_inserti64x4(_mm512_castsi256_si512(bits256), bits256, 1);
                    let hi = _mm512_srli_epi16(bits, 4);
                    let w = _mm512_and_si512(_mm512_mask_blend_epi32(0xFF00, bits, hi), m4);
                    let d32 = _mm512_dpbusd_epi32(zero, w, q8_512(y, 64 * j));
                    let sv = _mm512_mask_set1_epi32(_mm512_set1_epi32(sm[2 * j].0 as i32), 0xFF00, sm[2 * j + 1].0 as i32);
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(d32, sv));
                }
                let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
                sumf += y.d * d16(b) * _mm512_reduce_add_epi32(sumi) as f32 - y.d * d16(&b[2..]) * m as f32;
            }
            sumf
        }
    }

    /// Q6_K: 4 x 64 weights per super-block using 512-bit `vpdpbusd`.
    pub fn dot_q6k(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let (m4, m2, zero) = (_mm512_set1_epi8(0xF), _mm512_set1_epi8(3), _mm512_setzero_si512());
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(210).zip(xq) {
                let sc = |g: usize| b[192 + g] as i8 as i32;
                let mut sumi = zero;
                for n in 0..2 {
                    let ql = _mm512_loadu_si512(b[64 * n..].as_ptr() as *const __m512i);
                    let qh256 = _mm256_loadu_si256(b[128 + 32 * n..].as_ptr() as *const __m256i);
                    let qh = _mm512_inserti64x4(_mm512_castsi256_si512(qh256), qh256, 1);
                    let lo_h = _mm512_and_si512(ql, m4);
                    let hi_h = _mm512_and_si512(_mm512_srli_epi16(ql, 4), m4);
                    let h0 = _mm512_slli_epi16(_mm512_and_si512(_mm512_mask_blend_epi32(0xFF00, qh, _mm512_srli_epi16(qh, 2)), m2), 4);
                    let h1 = _mm512_slli_epi16(_mm512_and_si512(_mm512_mask_blend_epi32(0xFF00, _mm512_srli_epi16(qh, 4), _mm512_srli_epi16(qh, 6)), m2), 4);
                    let g = 8 * n;
                    let s0 = _mm512_set_epi32(sc(g + 3), sc(g + 3), sc(g + 3), sc(g + 3), sc(g + 2), sc(g + 2), sc(g + 2), sc(g + 2),
                                              sc(g + 1), sc(g + 1), sc(g + 1), sc(g + 1), sc(g), sc(g), sc(g), sc(g));
                    let s1 = _mm512_set_epi32(sc(g + 7), sc(g + 7), sc(g + 7), sc(g + 7), sc(g + 6), sc(g + 6), sc(g + 6), sc(g + 6),
                                              sc(g + 5), sc(g + 5), sc(g + 5), sc(g + 5), sc(g + 4), sc(g + 4), sc(g + 4), sc(g + 4));
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(_mm512_dpbusd_epi32(zero, _mm512_or_si512(lo_h, h0), q8_512(y, 128 * n)), s0));
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(_mm512_dpbusd_epi32(zero, _mm512_or_si512(hi_h, h1), q8_512(y, 128 * n + 64)), s1));
                }
                let m: i32 = (0..16).map(|g| sc(g) * y.bs[g] as i32).sum();
                let d = y.d * d16(&b[208..]);
                sumf += d * (_mm512_reduce_add_epi32(sumi) as f32 - 32.0 * m as f32);
            }
            sumf
        }
    }

    /// TQ1_0: all 256 trits per block evaluated in 4 512-bit `vpdpbusd` instructions.
    pub fn dot_tq1_0(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let th85 = _mm512_set1_epi8(85u8 as i8);
            let th170 = _mm512_set1_epi8(170u8 as i8);
            let (one8, zero) = (_mm512_set1_epi8(1), _mm512_setzero_si512());
            let trits_u = |q: __m512i| -> __m512i {
                let m1 = _mm512_cmpgt_epu8_mask(q, th85);
                let m2 = _mm512_cmpgt_epu8_mask(q, th170);
                let u = _mm512_maskz_set1_epi8(m1, 1);
                _mm512_mask_add_epi8(u, m2, u, one8)
            };
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(54).zip(xq) {
                let b0 = _mm256_loadu_si256(b.as_ptr() as *const __m256i);
                let raw0 = _mm512_inserti64x4(_mm512_castsi256_si512(b0), mul3_256(b0), 1);
                let raw1 = mul3_512(mul3_512(raw0));
                let b0_4 = mul3_256(_mm512_extracti64x4_epi64::<1>(raw1));
                let b1_0 = _mm_loadu_si128(b.as_ptr().add(32) as *const __m128i);
                let b1_01 = _mm256_set_m128i(mul3_128(b1_0), b1_0);
                let raw2 = _mm512_inserti64x4(_mm512_castsi256_si512(b0_4), b1_01, 1);
                let b1_23 = mul3_256(mul3_256(b1_01));
                let b1_4 = mul3_128(_mm256_extracti128_si256::<1>(b1_23));
                let bm = |m: u8| u32::from_le_bytes([b[48].wrapping_mul(m), b[49].wrapping_mul(m), b[50].wrapping_mul(m), b[51].wrapping_mul(m)]) as i32;
                let qh = _mm_set_epi32(bm(27), bm(9), bm(3), bm(1));
                let b14_qh = _mm256_set_m128i(qh, b1_4);
                let raw3 = _mm512_inserti64x4(_mm512_castsi256_si512(b1_23), b14_qh, 1);

                let mut s32 = zero;
                s32 = _mm512_dpbusd_epi32(s32, trits_u(raw0), q8_512(y, 0));
                s32 = _mm512_dpbusd_epi32(s32, trits_u(raw1), q8_512(y, 64));
                s32 = _mm512_dpbusd_epi32(s32, trits_u(raw2), q8_512(y, 128));
                s32 = _mm512_dpbusd_epi32(s32, trits_u(raw3), q8_512(y, 192));
                let bs: i32 = y.bs.iter().map(|&v| v as i32).sum();
                sumf += d16(&b[52..]) * y.d * (_mm512_reduce_add_epi32(s32) - bs) as f32;
            }
            sumf
        }
    }

    /// IQ1_S: two 32-value blocks per 512-bit `vpdpbusd` + delta bs term.
    pub fn dot_iq1s(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let zero = _mm512_setzero_si512();
            let (m7, ones, lo1) = (_mm_set1_epi16(7), _mm_set1_epi16(1), _mm_set1_epi16(1));
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(50).zip(xq) {
                let mut sumi = zero;
                for p in 0..4 {
                    let (ib0, ib1) = (2 * p, 2 * p + 1);
                    let qh0 = u16::from_le_bytes([b[34 + 2 * ib0], b[35 + 2 * ib0]]) as usize;
                    let qh1 = u16::from_le_bytes([b[34 + 2 * ib1], b[35 + 2 * ib1]]) as usize;
                    let g64 = |ib: usize, qh: usize, l: usize| IQ1S_GRID[b[2 + 4 * ib + l] as usize | ((qh >> (3 * l)) & 7) << 8] as i64;
                    let g0 = _mm256_set_epi64x(g64(ib0, qh0, 3), g64(ib0, qh0, 2), g64(ib0, qh0, 1), g64(ib0, qh0, 0));
                    let g1 = _mm256_set_epi64x(g64(ib1, qh1, 3), g64(ib1, qh1, 2), g64(ib1, qh1, 1), g64(ib1, qh1, 0));
                    let g = _mm512_inserti64x4(_mm512_castsi256_si512(g0), g1, 1);
                    let q = q8_512(y, 64 * p);
                    let neg = _mm512_movepi8_mask(g);
                    let p32 = _mm512_dpbusd_epi32(zero, _mm512_abs_epi8(g), _mm512_mask_sub_epi8(q, neg, zero, q));
                    let ls0 = 2 * ((qh0 >> 12) & 7) as i32 + 1;
                    let ls1 = 2 * ((qh1 >> 12) & 7) as i32 + 1;
                    let sv = _mm512_mask_set1_epi32(_mm512_set1_epi32(ls0), 0xFF00, ls1);
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(p32, sv));
                }
                let qh = _mm_loadu_si128(b[34..50].as_ptr() as *const __m128i);
                let ls = _mm_or_si128(_mm_slli_epi16(_mm_and_si128(_mm_srli_epi16(qh, 12), m7), 1), ones);
                let ls = _mm_sign_epi16(ls, _mm_or_si128(qh, lo1));
                let bs = |i: usize| _mm_loadu_si128(y.bs[8 * i..].as_ptr() as *const __m128i);
                let s = _mm_add_epi32(_mm_madd_epi16(_mm_unpacklo_epi16(ls, ls), bs(0)), _mm_madd_epi16(_mm_unpackhi_epi16(ls, ls), bs(1)));
                let s = _mm_add_epi32(s, _mm_shuffle_epi32(s, 0b01_00_11_10));
                let s1 = _mm_cvtsi128_si32(_mm_add_epi32(s, _mm_shuffle_epi32(s, 0b10_11_00_01)));
                let dy = d16(b) * y.d;
                sumf += dy * _mm512_reduce_add_epi32(sumi) as f32 + dy * 0.125 * s1 as f32;
            }
            sumf
        }
    }

    /// IQ2_XXS: two 32-value blocks per 512-bit `vpdpbusd` using 64-bit sign masks from `KSIGNS_IQ2XS`.
    pub fn dot_iq2xxs(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let zero = _mm512_setzero_si512();
            let ks = |aux: u32| -> u64 {
                (KSIGNS_IQ2XS[(aux & 127) as usize] as u64)
                    | ((KSIGNS_IQ2XS[((aux >> 7) & 127) as usize] as u64) << 8)
                    | ((KSIGNS_IQ2XS[((aux >> 14) & 127) as usize] as u64) << 16)
                    | ((KSIGNS_IQ2XS[((aux >> 21) & 127) as usize] as u64) << 24)
            };
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(66).zip(xq) {
                let mut sumi = zero;
                for p in 0..4 {
                    let (ib0, ib1) = (2 * p, 2 * p + 1);
                    let g = |ib: usize, l: usize| IQ2XXS_GRID[b[2 + 8 * ib + l] as usize] as i64;
                    let g0 = _mm256_set_epi64x(g(ib0, 3), g(ib0, 2), g(ib0, 1), g(ib0, 0));
                    let g1 = _mm256_set_epi64x(g(ib1, 3), g(ib1, 2), g(ib1, 1), g(ib1, 0));
                    let grid = _mm512_inserti64x4(_mm512_castsi256_si512(g0), g1, 1);
                    let aux0 = u32::from_le_bytes(b[6 + 8 * ib0..10 + 8 * ib0].try_into().unwrap());
                    let aux1 = u32::from_le_bytes(b[6 + 8 * ib1..10 + 8 * ib1].try_into().unwrap());
                    let neg = ks(aux0) | (ks(aux1) << 32);
                    let q = q8_512(y, 64 * p);
                    let p32 = _mm512_dpbusd_epi32(zero, grid, _mm512_mask_sub_epi8(q, neg, zero, q));
                    let (ls0, ls1) = (2 * (aux0 >> 28) as i32 + 1, 2 * (aux1 >> 28) as i32 + 1);
                    let sv = _mm512_mask_set1_epi32(_mm512_set1_epi32(ls0), 0xFF00, ls1);
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(p32, sv));
                }
                sumf += 0.125 * d16(b) * y.d * _mm512_reduce_add_epi32(sumi) as f32;
            }
            sumf
        }
    }

    /// IQ3_S: two 32-value blocks per 512-bit `vpdpbusd` with zero-cost 64-bit sign mask.
    pub fn dot_iq3s(row: &[u8], xq: &[Q8K]) -> f32 {
        unsafe {
            let zero = _mm512_setzero_si512();
            let shl = _mm256_set_epi32(1, 2, 3, 4, 5, 6, 7, 8);
            let h256 = _mm256_set1_epi32(256);
            let mut sumf = 0.0f32;
            for (b, y) in row.chunks_exact(110).zip(xq) {
                let (qs, qh) = (&b[2..66], &b[66..74]);
                let g256 = |ib: usize| -> __m256i {
                    let lo = _mm256_cvtepu8_epi32(_mm_loadl_epi64(qs[8 * ib..].as_ptr() as *const __m128i));
                    let hi = _mm256_and_si256(_mm256_sllv_epi32(_mm256_set1_epi32(qh[ib] as i32), shl), h256);
                    _mm256_i32gather_epi32(IQ3S_GRID.as_ptr() as *const i32, _mm256_or_si256(lo, hi), 4)
                };
                let mut sumi = zero;
                for p in 0..4 {
                    let (ib0, ib1) = (2 * p, 2 * p + 1);
                    let grid = _mm512_inserti64x4(_mm512_castsi256_si512(g256(ib0)), g256(ib1), 1);
                    let sb0 = u32::from_le_bytes(b[74 + 4 * ib0..78 + 4 * ib0].try_into().unwrap()) as u64;
                    let sb1 = u32::from_le_bytes(b[74 + 4 * ib1..78 + 4 * ib1].try_into().unwrap()) as u64;
                    let neg = sb0 | (sb1 << 32);
                    let q = q8_512(y, 64 * p);
                    let p32 = _mm512_dpbusd_epi32(zero, grid, _mm512_mask_sub_epi8(q, neg, zero, q));
                    let ls0 = 2 * ((b[106 + ib0 / 2] >> (4 * (ib0 % 2))) & 0xF) as i32 + 1;
                    let ls1 = 2 * ((b[106 + ib1 / 2] >> (4 * (ib1 % 2))) & 0xF) as i32 + 1;
                    let sv = _mm512_mask_set1_epi32(_mm512_set1_epi32(ls0), 0xFF00, ls1);
                    sumi = _mm512_add_epi32(sumi, _mm512_mullo_epi32(p32, sv));
                }
                sumf += d16(b) * y.d * _mm512_reduce_add_epi32(sumi) as f32;
            }
            sumf
        }
    }
}

/// A trellis row (blocks of B bytes for 128 values, `quant::tq_points`) times x, in f32: the codebook is not on an
/// int8 grid, so these types have no Q8_K kernel. Eight partial sums per block, where `Tensor::matvec` adds in order.
fn dot_tq<const B: usize>(row: &[u8], x: &[f32]) -> f32 {
    use crate::quant::{TQ_LUT as L, tq_points};
    let mut sum = 0f32;
    for (b, xb) in row.chunks_exact(B).zip(x.chunks_exact(128)) {
        let (qs, mut acc) = (&b[2..B], [0f32; 8]);
        for (t, xs) in xb.chunks_exact(8).enumerate() {
            let ((p0, p1), (p2, p3)) = (tq_points(qs, 2 * t), tq_points(qs, 2 * t + 1));
            let w = [L[p0][0], L[p0][1], L[p1][0], L[p1][1], L[p2][0], L[p2][1], L[p3][0], L[p3][1]];
            for i in 0..8 { acc[i] += w[i] * xs[i]; }
        }
        sum += d16(b) * acc.iter().sum::<f32>();
    }
    sum
}

/// Signed OCP FP8 E4M3 row times x via the 256-entry `F8_E4M3_LUT` (no intermediate dequant buffer).
fn dot_f8_e4m3(row: &[u8], x: &[f32]) -> f32 {
    let lut = &crate::quant::F8_E4M3_LUT;
    let mut acc = [0.0f32; 8];
    let mut r = row.chunks_exact(8);
    let mut xv = x.chunks_exact(8);
    for (rb, xb) in r.by_ref().zip(xv.by_ref()) {
        for i in 0..8 { acc[i] += lut[rb[i] as usize] * xb[i]; }
    }
    let mut sum: f32 = acc.iter().sum();
    for (&b, &v) in r.remainder().iter().zip(xv.remainder()) {
        sum += lut[b as usize] * v;
    }
    sum
}

/// The f32 kernel of a type that has no integer one but is too slow through `Tensor::row`.
pub fn float_dot(ty: GgmlType) -> Option<fn(&[u8], &[f32]) -> f32> {
    match ty {
        GgmlType::F8E4M3 => Some(dot_f8_e4m3),
        GgmlType::TQ2T => Some(dot_tq::<34>),
        GgmlType::TQK6 => Some(dot_tq::<26>),
        GgmlType::TQK7 => Some(dot_tq::<30>),
        _ => None,
    }
}

pub fn int_dot(ty: GgmlType) -> Option<fn(&[u8], &[Q8K]) -> f32> {
    #[cfg(all(target_arch = "x86_64", target_feature = "avx512f", target_feature = "avx512bw", target_feature = "avx512vnni", target_feature = "avx512vbmi"))]
    // Opt-in: these kernels add the block sums in a different order than the AVX2 ones, so their results differ in the
    // last bits, and that is enough to make a drafted run and a plain greedy run pick different tokens at a near tie.
    if std::env::var_os("STRATA_SCALAR").is_none() && std::env::var_os("STRATA_AVX512").is_some() {
        match ty {
            GgmlType::Q8_0 => return Some(avx512::dot_q8_0),
            GgmlType::Q4_0 => return Some(avx512::dot_q4::<2>),
            GgmlType::Q4_1 => return Some(avx512::dot_q4::<3>),
            GgmlType::Q5_0 => return Some(avx512::dot_q4::<6>),
            GgmlType::Q5_1 => return Some(avx512::dot_q4::<7>),
            GgmlType::IQ4NL => return Some(avx512::dot_iq4nl),
            GgmlType::IQ4XS => return Some(avx512::dot_iq4xs),
            GgmlType::Q4K => return Some(avx512::dot_q4k),
            GgmlType::Q6K => return Some(avx512::dot_q6k),
            GgmlType::TQ1_0 => return Some(avx512::dot_tq1_0),
            GgmlType::IQ1S => return Some(avx512::dot_iq1s),
            GgmlType::IQ2XXS => return Some(avx512::dot_iq2xxs),
            GgmlType::IQ3S => return Some(avx512::dot_iq3s),
            _ => {}
        }
    }
    #[cfg(target_feature = "avx2")]
    if std::env::var_os("STRATA_SCALAR").is_none() {
        return match ty { GgmlType::IQ2XS => Some(avx2::dot_iq2xs), GgmlType::IQ3XXS => Some(avx2::dot_iq3xxs), GgmlType::IQ4XS => Some(avx2::dot_iq4xs), GgmlType::IQ4NL => Some(avx2::dot_iq4nl), GgmlType::IQ3S => Some(avx2::dot_iq3s), GgmlType::Q6K => Some(avx2::dot_q6k), GgmlType::Q5K => Some(avx2::dot_q5k), GgmlType::Q8_0 => Some(avx2::dot_q8_0), GgmlType::IQ2XXS => Some(avx2::dot_iq2xxs), GgmlType::IQ2S => Some(avx2::dot_iq2s), GgmlType::IQ1S => Some(avx2::dot_iq1s), GgmlType::IQ1M => Some(avx2::dot_iq1m), GgmlType::Q2K => Some(avx2::dot_q2k), GgmlType::Q3K => Some(avx2::dot_q3k), GgmlType::Q4K => Some(avx2::dot_q4k), GgmlType::Q2_0 => Some(avx2::dot_q2_0), GgmlType::TQ1_0 => Some(avx2::dot_tq1_0), GgmlType::TQ2_0 => Some(dot_tq2_0), GgmlType::NVFP4 => Some(avx2::dot_nvfp4), GgmlType::Q4_0 => Some(avx2::dot_q4::<2>), GgmlType::Q4_1 => Some(avx2::dot_q4::<3>), GgmlType::Q5_0 => Some(avx2::dot_q4::<6>), GgmlType::Q5_1 => Some(avx2::dot_q4::<7>), _ => None };
    }
    match ty { GgmlType::IQ2XS => Some(dot_iq2xs), GgmlType::IQ3XXS => Some(dot_iq3xxs), GgmlType::IQ4XS => Some(dot_iq4xs), GgmlType::IQ4NL => Some(dot_iq4nl), GgmlType::IQ3S => Some(dot_iq3s), GgmlType::Q6K => Some(dot_q6k), GgmlType::Q5K => Some(dot_q5k), GgmlType::Q8_0 => Some(dot_q8_0), GgmlType::IQ2XXS => Some(dot_iq2xxs), GgmlType::IQ2S => Some(dot_iq2s), GgmlType::IQ1S => Some(dot_iq1s), GgmlType::IQ1M => Some(dot_iq1m), GgmlType::Q2K => Some(dot_q2k), GgmlType::Q3K => Some(dot_q3k), GgmlType::Q4K => Some(dot_q4k), GgmlType::Q2_0 => Some(dot_q2_0), GgmlType::TQ1_0 => Some(dot_tq1_0), GgmlType::TQ2_0 => Some(dot_tq2_0), GgmlType::NVFP4 => Some(dot_nvfp4), GgmlType::Q4_0 => Some(dot_q4::<2>), GgmlType::Q4_1 => Some(dot_q4::<3>), GgmlType::Q5_0 => Some(dot_q4::<6>), GgmlType::Q5_1 => Some(dot_q4::<7>), _ => None }
}

/// Four dots of one row at once, with the bits of `int_dot`'s kernel: the row is read and decoded once, and the
/// four sums do not wait for each other. None where no such kernel exists; the caller then uses `int_dot`.
pub fn int_dot4(ty: GgmlType) -> Option<fn(&[u8], [&[Q8K]; 4]) -> [f32; 4]> {
    #[cfg(target_feature = "avx2")]
    if std::env::var_os("STRATA_SCALAR").is_none() && std::env::var_os("STRATA_AVX512").is_none() {
        return match ty { GgmlType::Q8_0 => Some(avx2::dot4_q8_0), GgmlType::IQ4NL => Some(avx2::dot4_iq4nl), _ => None };
    }
    let _ = ty;
    None
}

/// y = t · x for every row, using the integer kernel when the type has one.
pub fn matvec(t: &Tensor, x: &[f32], y: &mut [f32]) {
    let rb = t.data.len() / t.rows;
    if let Some(dot) = float_dot(t.ty) {
        return y.par_chunks_mut(32).enumerate().for_each(|(c, ys)| {
            for (k, v) in ys.iter_mut().enumerate() { let r = c * 32 + k; *v = dot(&t.data[r * rb..(r + 1) * rb], x) * t.scale; }
        });
    }
    let Some(dot) = int_dot(t.ty) else { return t.matvec(x, y) };
    let xq = quantize_q8k(x);
    y.par_chunks_mut(32).enumerate().for_each(|(c, ys)| {
        for (k, v) in ys.iter_mut().enumerate() {
            let r = c * 32 + k;
            *v = dot(&t.data[r * rb..(r + 1) * rb], &xq) * t.scale;
        }
    });
}

/// Rows `out.len()` of a batch of matrices: out[i] = mats[i / rows] row (i % rows) · x of that matrix, where
/// matrix j uses xs[j * xs.len() / mats.len()] (one x for all, or one each), so a shared x is quantized once.
fn batch_rows(mats: &[Tensor], xs: &[&[f32]], out: &mut [f32]) {
    let rows = out.len() / mats.len();
    let (dot, fdot) = (int_dot(mats[0].ty), float_dot(mats[0].ty));
    let xq: Vec<Vec<Q8K>> = if dot.is_some() { xs.iter().map(|x| quantize_q8k(x)).collect() } else { vec![] };
    // Q2_0 experts benefit from fewer Rayon tasks on AVX2. Keep enough chunks to occupy the pool,
    // including when only one expert misses; other formats keep their existing granularity.
    let batch = if mats[0].ty == GgmlType::Q2_0 && cfg!(target_feature = "avx2")
        && out.len() / 64 >= rayon::current_num_threads() { 64 } else { 16 };
    out.par_chunks_mut(batch).enumerate().for_each_init(|| if dot.is_none() { vec![0f32; mats[0].cols] } else { Vec::new() }, |buf, (c, ys)| {
        for (k, v) in ys.iter_mut().enumerate() {
            let i = c * batch + k;
            let (t, r, x) = (&mats[i / rows], i % rows, i / rows * xs.len() / mats.len());
            let rb = t.data.len() / t.rows;
            *v = match (dot, fdot) {
                (Some(d), _) => d(&t.data[r * rb..(r + 1) * rb], &xq[x]) * t.scale,
                (_, Some(d)) => d(&t.data[r * rb..(r + 1) * rb], xs[x]) * t.scale,
                _ => { t.row(r, buf); buf.iter().zip(xs[x]).map(|(a, b)| a * b).sum() }
            };
        }
    });
}

/// Weighted sum of the given routed experts of layer `il` applied to `h` (no router, no shared expert).
pub fn experts(m: &Model, il: usize, h: &[f32], sel: &[(usize, f32)]) -> Vec<f32> {
    let c = &m.cfg;
    let mut y = vec![0f32; c.n_embd];
    if sel.is_empty() { return y; }
    let t = |s: &str| m.w.layer(il, s).unwrap_or_else(|e| panic!("{e:#}"));
    let (gate, up, down) = (t("ffn_gate_exps.weight"), t("ffn_up_exps.weight"), t("ffn_down_exps.weight"));
    let (n, ff, k) = (c.n_expert, c.n_ff_exp, sel.len());
    // Gate and up of every expert in one parallel pass (gate and up share a type in all supported files).
    let gu: Vec<Tensor> = sel.iter().flat_map(|&(e, _)| [gate.expert(e, n), up.expert(e, n)]).collect();
    let mut a = vec![0f32; 2 * k * ff];
    // Gyro files: the routed experts read rotated inputs (`config::Hadamard`); the LoRA delta below reads `acts`.
    let rot = c.hadamard.as_ref();
    let hr = rot.map(|r| r.rotated(h));
    let h = hr.as_deref().unwrap_or(h);
    if gate.ty == up.ty {
        batch_rows(&gu, &[h], &mut a);
    } else {
        for (i, t) in gu.iter().enumerate() { matvec(t, h, &mut a[i * ff..(i + 1) * ff]); }
    }
    let acts: Vec<Vec<f32>> = a.chunks_exact(2 * ff).map(|g| (0..ff).map(|i| ops::silu(g[i]) * g[ff + i]).collect()).collect();
    let racts: Option<Vec<Vec<f32>>> = rot.map(|r| acts.iter().map(|a| r.rotated(a)).collect());
    let ds: Vec<Tensor> = sel.iter().map(|&(e, _)| down.expert(e, n)).collect();
    let mut o = vec![0f32; k * c.n_embd];
    batch_rows(&ds, &racts.as_ref().unwrap_or(&acts).iter().map(|v| v.as_slice()).collect::<Vec<_>>(), &mut o);
    for (j, &(e, _)) in sel.iter().enumerate() {
        m.lora_delta(&format!("blk.{il}.ffn_down_exps.weight"), &acts[j], &mut o[j * c.n_embd..(j + 1) * c.n_embd], Some(e));
    }
    for (oj, &(_, w)) in o.chunks_exact(c.n_embd).zip(sel) {
        for (yy, oo) in y.iter_mut().zip(oj) { *yy += w * oo; }
    }
    y
}

/// `experts` for several tokens of one layer (a verify batch): ys[t] = `experts(m, il, &hs[t], &sels[t])`, bit for bit.
/// All tokens share the two parallel passes of `experts` (gate and up, then down), so a batch costs no more passes
/// than one token: a pass per expert costs more in thread wake-ups than the experts two tokens share save.
pub fn experts_multi(m: &Model, il: usize, hs: &[Vec<f32>], sels: &[Vec<(usize, f32)>]) -> Vec<Vec<f32>> {
    let c = &m.cfg;
    let mut ys = vec![vec![0f32; c.n_embd]; hs.len()];
    // (token, expert, weight) of every routed expert, in token and router order.
    let pairs: Vec<(usize, usize, f32)> = sels.iter().enumerate().flat_map(|(t, s)| s.iter().map(move |&(e, w)| (t, e, w))).collect();
    if pairs.is_empty() { return ys; }
    let t = |s: &str| m.w.layer(il, s).unwrap_or_else(|e| panic!("{e:#}"));
    let (gate, up, down) = (t("ffn_gate_exps.weight"), t("ffn_up_exps.weight"), t("ffn_down_exps.weight"));
    let (n, ff, k) = (c.n_expert, c.n_ff_exp, pairs.len());
    let gu: Vec<Tensor> = pairs.iter().flat_map(|&(_, e, _)| [gate.expert(e, n), up.expert(e, n)]).collect();
    // Gyro files: the routed experts read rotated inputs (`config::Hadamard`); the LoRA delta below reads `acts`.
    let rot = c.hadamard.as_ref();
    let hr: Option<Vec<Vec<f32>>> = rot.map(|r| hs.iter().map(|h| r.rotated(h)).collect());
    let hs = hr.as_deref().unwrap_or(hs);
    let xs: Vec<&[f32]> = pairs.iter().map(|&(t, ..)| hs[t].as_slice()).collect();
    let mut a = vec![0f32; 2 * k * ff];
    if gate.ty == up.ty {
        batch_rows(&gu, &xs, &mut a);
    } else {
        for (i, t) in gu.iter().enumerate() { matvec(t, xs[i / 2], &mut a[i * ff..(i + 1) * ff]); }
    }
    let acts: Vec<Vec<f32>> = a.chunks_exact(2 * ff).map(|g| (0..ff).map(|i| ops::silu(g[i]) * g[ff + i]).collect()).collect();
    let racts: Option<Vec<Vec<f32>>> = rot.map(|r| acts.iter().map(|a| r.rotated(a)).collect());
    let ds: Vec<Tensor> = pairs.iter().map(|&(_, e, _)| down.expert(e, n)).collect();
    let mut o = vec![0f32; k * c.n_embd];
    batch_rows(&ds, &racts.as_ref().unwrap_or(&acts).iter().map(|v| v.as_slice()).collect::<Vec<_>>(), &mut o);
    for (j, (oj, &(t, e, w))) in o.chunks_exact_mut(c.n_embd).zip(&pairs).enumerate() {
        m.lora_delta(&format!("blk.{il}.ffn_down_exps.weight"), &acts[j], oj, Some(e));
        for (yy, oo) in ys[t].iter_mut().zip(oj.iter()) { *yy += w * oo; }
    }
    ys
}

    /// Batch boundaries can cross matrices; both Q2_0 and the f32 fallback must use the right input row.
    #[test]
    fn batch_rows_match_individual_matvecs() {

        let (cols, rows, count) = (640, 257, 3);
        let inputs: Vec<Vec<f32>> = (0..count).map(|m| (0..cols).map(|i| ((i * 13 + m * 7) % 101) as f32 / 50.0 - 1.0).collect()).collect();
        for ty in [GgmlType::Q2_0, GgmlType::F32] {
            let rb = ty.row_bytes(cols as u64);
            let data: Vec<Vec<u8>> = (0..count).map(|m| {
                let mut v = vec![0u8; rows * rb];
                if ty == GgmlType::Q2_0 {
                    for (i, b) in v.chunks_exact_mut(18).enumerate() {
                        b[..2].copy_from_slice(&f16::from_f32(0.01).to_le_bytes());
                        for (j, q) in b[2..].iter_mut().enumerate() { *q = ((i * 17 + j * 11 + m) % 256) as u8; }
                    }
                } else {
                    for (i, b) in v.chunks_exact_mut(4).enumerate() { b.copy_from_slice(&(((i + m * 3) % 113) as f32 / 57.0 - 1.0).to_le_bytes()); }
                }
                v
            }).collect();
            let mats: Vec<Tensor> = data.iter().map(|d| Tensor { ty, cols, rows, data: d, scale: 1.0 }).collect();
            let xs: Vec<&[f32]> = inputs.iter().map(Vec::as_slice).collect();

// ==== /z/ai-models/rs-v131wt/src/cpu_worker.rs
//! CPU side of the expert split, without per-layer stream syncs. For each MoE layer the GPU writes routing ids,
//! weights, cache slots and (when some expert missed) the normed activation into host-mapped pinned memory, then
//! raises the layer's request flag (on CUDA a layer with no miss raises only its skip flag). This worker spins on the
//! flag, computes the missed experts with `cpu_moe`, writes their weighted sum back and raises the done flag, which the
//! GPU combine kernel waits on only if the layer missed.

use crate::model::Model;
use crate::backend::{Backend, Buf};
use std::sync::atomic::{AtomicBool, AtomicU64, Ordering, fence};
use std::sync::{Arc, mpsc};
use std::thread::JoinHandle;
use std::time::{Duration, Instant};

/// Word offsets inside one layer's mailbox. Flags sit on separate cache lines.
pub const REQ: usize = 0;
/// Tokens in the request (1 for decode, up to `TMAX` for a multi-token verify).
pub const NTOK: usize = 1;
/// Raised instead of REQ, with no data, when no expert of the layer missed (CUDA only; Vulkan raises REQ).
pub const SKIP: usize = 2;
/// Vulkan: the combine's verdict on its wait (0 open, 1 the worker answered, 2 gave up; see mb_wait.glsl).
pub const VERDICT: usize = 8;
pub const DONE: usize = 16;
pub const IDS: usize = 32;

/// Most tokens one request may carry (1 current + up to 16 speculative draft tokens).
pub const TMAX: usize = 17;

/// Per layer: ids, weights and slots for up to TMAX tokens of `k` entries (token t at t * k; the three arrays are
/// each NTOK * k long), then TMAX activations of `n` floats at `h` and TMAX results at `y`.
pub struct Mailbox { pub k: usize, pub n: usize, pub h: usize, pub y: usize, pub stride: usize }

impl Mailbox {
    pub fn new(k: usize, n: usize) -> Mailbox {
        let h = (IDS + 3 * TMAX * k).next_multiple_of(16);
        Mailbox { k, n, h, y: h + TMAX * n, stride: (h + 2 * TMAX * n).next_multiple_of(16) }
    }
}

pub struct CpuWorker {
    _mem: Buf,
    host: usize,
    /// Device address of the mailboxes (layer `il` at `dev + 4 * il * layout.stride`).
    pub dev: u64,
    pub layout: Mailbox,
    seq: u32,
    layers: Vec<usize>,
    tx: Option<mpsc::Sender<u32>>,
    stop: Arc<AtomicBool>,
    busy_ns: Arc<AtomicU64>,
    thread: Option<JoinHandle<()>>,
}

impl CpuWorker {
    /// Mailboxes cover layers up to the largest of `moe_layers`. Safety: `m` must outlive the worker (it is joined on drop).
    pub fn new(m: &Model, gpu: &dyn Backend, moe_layers: Vec<usize>) -> anyhow::Result<CpuWorker> {
        let c = &m.cfg;
        let layout = Mailbox::new(c.n_expert_used, c.n_embd);
        let len = layout.stride * moe_layers.iter().max().map_or(0, |&il| il + 1);
        let (mem, host) = gpu.host_mailbox(len)?;
        let (host, dev) = (host as usize, mem.addr());

        let (tx, rx) = mpsc::channel::<u32>();
        let (stop, busy_ns) = (Arc::new(AtomicBool::new(false)), Arc::new(AtomicU64::new(0)));
        let (st, bn) = (stop.clone(), busy_ns.clone());
        let mp = m as *const Model as usize;
        let lay = Mailbox::new(c.n_expert_used, c.n_embd);
        let layers = moe_layers.clone();
        // STRATA_TEST_WORKER_DELAY=ms: answer some requests (layer and seq picked by a hash) that much later, as a
        // worker reading experts from a cold drive would (tests the combines that give up).
        let delay = std::env::var("STRATA_TEST_WORKER_DELAY").ok().and_then(|v| v.parse::<u64>().ok()).map(Duration::from_millis);
        // The loop runs on a thread of its own pool, so the expert passes start on it at once and it takes part
        // in them instead of sleeping until a woken pool finishes (~12 us less per layer).
        let pool = rayon::ThreadPoolBuilder::new().num_threads(rayon::current_num_threads())
            .thread_name(|i| format!("strata-experts-{i}")).build()?;
        let thread = std::thread::Builder::new().name("strata-cpu-experts".into()).spawn(move || pool.install(move || {
            let m = unsafe { &*(mp as *const Model) };
            // Missed experts not in the page cache are read from the drive: reading all of a layer's ahead keeps
            // the reads in flight together instead of one page fault at a time per compute thread.
            let pre = crate::weights::Prefetch::start();
            let w = |i: usize| host as *mut u32 as usize + 4 * i;
            while let Ok(seq) = rx.recv() {
                for &il in &moe_layers {
                    let b = il * lay.stride;
                    let at = |o: usize| (w(b + o)) as *mut u32;
                    // Skip the layer on its skip flag, or on a newer seq anywhere (the GPU already moved on to the
                    // next token, so it did not wait for this one). Waiting for the old seq would hang.
                    let ahead = |v: u32| v != 0 && (v.wrapping_sub(seq) as i32) >= 0;
                    let mut r;
                    while { r = unsafe { at(REQ).read_volatile() }; r != seq } {
                        if st.load(Ordering::Relaxed) { return; }
                        if ahead(r) || ahead(unsafe { at(SKIP).read_volatile() }) { break; }
                        std::hint::spin_loop();
                    }
                    if r != seq { continue; }
                    // mfence: loads of write-combining memory (the Vulkan BAR mailbox) are weakly ordered, so an
                    // acquire fence (no instruction on x86) would not keep them after the REQ load.
                    fence(Ordering::SeqCst);
                    let t0 = Instant::now();
                    let rd = |o: usize| unsafe { at(o).read_volatile() };
                    let nt = (rd(NTOK) as usize).clamp(1, TMAX);
                    let (k, kt, n) = (lay.k, lay.k * nt, lay.n);
                    // Bulk streaming reads, as in `routing`: plain loads of the Vulkan BAR mailbox are each an
                    // uncached read (~130 us for one activation).
                    let mut v = vec![0u32; (3 * kt).next_multiple_of(4)];
                    unsafe { read_wc(at(IDS), &mut v) };
                    let missed: Vec<usize> = (0..kt).filter(|&j| (v[2 * kt + j] as i32) < 0).map(|j| v[j] as usize).collect();
                    if !missed.is_empty() {
                        // Gate and up first: they are read first.
                        pre.send(["ffn_gate_exps.weight", "ffn_up_exps.weight", "ffn_down_exps.weight"].iter()
                            .filter_map(|s| m.w.layer(il, s).ok())
                            .flat_map(|t| missed.iter().map(move |&e| { let d = t.expert(e, m.cfg.n_expert).data; (d.as_ptr() as usize, d.len()) }).collect::<Vec<_>>())
                            .collect());
                    }
                    if nt > 1 {
                        // A verify batch: one pass over each missed expert's weights for all of its tokens.
                        let (mut hs, mut sels, mut rows) = (Vec::new(), Vec::new(), Vec::new());
                        for t in 0..nt {
                            let sel: Vec<(usize, f32)> = (t * k..(t + 1) * k).filter(|&j| (v[2 * kt + j] as i32) < 0)
                                .map(|j| (v[j] as usize, f32::from_bits(v[kt + j]))).collect();
                            if sel.is_empty() { continue; }
                            let mut h = vec![0u32; n];
                            unsafe { read_wc(at(lay.h + t * n), &mut h) };
                            hs.push(h.into_iter().map(f32::from_bits).collect::<Vec<f32>>());
                            sels.push(sel);
                            rows.push(t);
                        }
                        if !hs.is_empty() {
                            let ys = crate::cpu_moe::experts_multi(m, il, &hs, &sels);
                            for (y, &t) in ys.iter().zip(&rows) {
                                unsafe { std::ptr::copy_nonoverlapping(y.as_ptr(), at(lay.y + t * n) as *mut f32, n) };
                            }
                        }
                    } else {
                        for t in 0..nt {
                            let sel: Vec<(usize, f32)> = (t * k..(t + 1) * k).filter(|&j| (v[2 * kt + j] as i32) < 0)
                                .map(|j| (v[j] as usize, f32::from_bits(v[kt + j]))).collect();
                            if sel.is_empty() { continue; }
                            let mut h = vec![0u32; n];
                            unsafe { read_wc(at(lay.h + t * n), &mut h) };
                            let h: Vec<f32> = h.into_iter().map(f32::from_bits).collect();
                            let y = crate::cpu_moe::experts(m, il, &h, &sel);
                            unsafe { std::ptr::copy_nonoverlapping(y.as_ptr(), at(lay.y + t * n) as *mut f32, n) };
                        }
                    }
                    // SeqCst fences (mfence on x86) also order and flush write-combining stores (Vulkan maps the
                    // mailbox through the BAR): y before DONE, and DONE out now.
                    if let Some(d) = delay { if (seq as usize * 31 + il) % 17 == 0 { std::thread::sleep(d) } }
                    fence(Ordering::SeqCst);
                    unsafe { at(DONE).write_volatile(seq) };
                    fence(Ordering::SeqCst);
                    bn.fetch_add(t0.elapsed().as_nanos() as u64, Ordering::Relaxed);
                }
            }
        }))?;
        Ok(CpuWorker { _mem: mem, host, dev, layout, seq: 0, layers, tx: Some(tx), stop, busy_ns, thread: Some(thread) })
    }

    /// Start a new token: returns the sequence number the GPU must publish with.
    pub fn begin(&mut self) -> u32 {
        self.seq = self.seq.wrapping_add(1).max(1);
        let _ = self.tx.as_ref().unwrap().send(self.seq);
        self.seq
    }

    /// Routing of layer `il` from the last finished request, all its tokens: (expert id, routed weight, slot or -1).
    pub fn routing(&self, il: usize) -> Vec<(i32, f32, i32)> {
        let b = il * self.layout.stride;
        let rd = |o: usize| unsafe { (self.host as *const u32).add(b + o).read_volatile() };
        let k = self.layout.k * (rd(NTOK) as usize).clamp(1, TMAX);
        // One bulk read: on Vulkan the mailbox is BAR memory, where each host read is a slow uncached access.
        let mut v = vec![0u32; (3 * k).next_multiple_of(4)];
        unsafe { read_wc((self.host as *const u32).add(b + IDS), &mut v) };
        (0..k).map(|j| (v[j] as i32, f32::from_bits(v[k + j]), v[2 * k + j] as i32)).collect()
    }

    /// The first layer whose combine gave up waiting (verdict 2) in the last request: the layers before it ran.
    pub fn stalled_layer(&self) -> Option<usize> {
        self.layers.iter().copied().find(|&il| unsafe { (self.host as *const u32).add(il * self.layout.stride + VERDICT).read_volatile() } == 2)
    }

    /// Wait until the worker has answered layer `il` of the current request, then reopen the layer's verdict so its
    /// combine can run again.
    pub fn wait_done(&self, il: usize) -> anyhow::Result<()> {
        let at = |o: usize| unsafe { (self.host as *mut u32).add(il * self.layout.stride + o) };
        while unsafe { at(DONE).read_volatile() } != self.seq {
            anyhow::ensure!(!self.thread.as_ref().is_some_and(|t| t.is_finished()), "the CPU expert worker stopped");
            std::thread::sleep(Duration::from_micros(50));
        }
        unsafe { at(VERDICT).write_volatile(0) };
        fence(Ordering::SeqCst);
        Ok(())
    }

    /// Total time the worker spent computing experts.
    pub fn busy(&self) -> Duration { Duration::from_nanos(self.busy_ns.load(Ordering::Relaxed)) }
}

/// Copies `out.len()` (a multiple of 4) words from 16-byte aligned `p`. On write-combining memory (the Vulkan BAR
/// mailbox) streaming loads fetch a whole 64-byte line per read instead of one uncached read per load.
unsafe fn read_wc(p: *const u32, out: &mut [u32]) {
    #[cfg(target_feature = "sse4.1")]
    for (i, c) in out.chunks_exact_mut(4).enumerate() {
        use std::arch::x86_64::*;
        unsafe { _mm_storeu_si128(c.as_mut_ptr() as *mut __m128i, _mm_stream_load_si128(p.add(4 * i) as *const __m128i)) };
    }
    #[cfg(not(target_feature = "sse4.1"))]
    unsafe { std::ptr::copy_nonoverlapping(p, out.as_mut_ptr(), out.len()) };
}

impl Drop for CpuWorker {
    fn drop(&mut self) {
        self.stop.store(true, Ordering::Relaxed);
        self.tx = None;
        if let Some(t) = self.thread.take() { let _ = t.join(); }
    }
}

pub struct Gpu {
    pub ctx: Arc<CudaContext>,
    pub stream: Arc<CudaStream>,
    /// Takes every second launch inside `batch`, so a captured graph runs them as parallel branches.
    side: Arc<CudaStream>,
    /// Launches so far in the current `batch` (usize::MAX outside one).
    nb: AtomicUsize,
    fns: HashMap<String, CudaFunction>,

    i8_gemm: bool,
    /// q8 rows of the last integer GEMM's X (see quant_q8 in kernels.cu), grown on demand, and which x they hold
    /// (x, x_stride, n_x, cols) for a `SAME_X` call; None after a call without it that did not quantize.
    q8: std::sync::Mutex<(CudaSlice<u8>, Option<[u64; 4]>)>,
    /// `gemmd_*` scratch for a GEMM that two blocks share the columns of: a slot per stream (see `GEMMD_SLOT`),
    /// allocated by the first such GEMM.
    gd: std::sync::OnceLock<CudaSlice<u8>>,
    /// The kernels were compiled for KV cache mode (0 = f16, 1 = q8, 2 = q4).
    pub kv_mode: u32,
    /// The GPU has int8 tensor-core mma (sm_80+). Turing runs the prefill GEMM on the f16 path instead.
    pub int8_mma: bool,
    /// sm_75: m16n8k8 mma, 64 KB of shared memory a block (no int8 mma, unlike sm_80+; unlike sm_70 it has no m8n8k4).
    turing: bool,
    /// The device's opt-in shared memory per block (96 KB on Ampere, 64 KB on Turing).
    max_smem: u32,
    /// `argmax_rows` partials, AMAX_PARTS per row.
    am: CudaSlice<[i32; 2]>,
    /// Recorded graphs by `Backend::graph` key.
    graphs: std::sync::Mutex<HashMap<u64, Graph>>,
    /// Pinned staging for large `htod`, used as a ring: the copy returns before its DMA, so the host fills the next
    /// one while the GPU works (expert streaming in prefill). Head offset and the regions queued copies still read.
    ring: std::sync::Mutex<(PinnedHostSlice<u8>, usize, VecDeque<(usize, usize, CudaEvent)>)>,
    /// Recorded by `wait`.
    done: CudaEvent,
    /// The side stream after the last `beside`'s copies; the next `beside` makes the stream wait for it.
    lag: std::sync::Mutex<Option<CudaEvent>>,
    /// rope_freq tables by (base bits, n_rot), and the ones a graph capture asked for (see `rope_fr`).
    fr: std::sync::Mutex<(HashMap<(u32, usize), CudaSlice<f64>>, Vec<(f32, usize)>)>,
}

const RING: usize = 64 << 20;

/// Smaller copies take the driver's pageable path (staged before the call returns).
const RING_MIN: usize = 256 << 10;
/// `nb` inside `beside`: launches stay on the main stream, copies go to the side stream.
const BESIDE: usize = usize::MAX - 1;
/// `gemmd_*` splits the columns of a matrix of at most this many rows between two blocks. Measured on an RTX 3060
/// with Qwen3-4B: the 9728-row FFN matrices gain at 8 and 16 positions and are level at 64; the output head (151936
/// rows) fills the card without the split.
const GEMMD_SPLIT_ROWS: usize = 16384;
/// A scratch slot of `gemmd_*`: a count per 128 rows, then the partial sums of both blocks for 64 positions.
const GEMMD_SLOT: usize = GEMMD_SPLIT_ROWS / 128 * 4 + 2 * 64 * GEMMD_SPLIT_ROWS * 4;

/// A recorded graph. Its handles are plain CUDA driver handles, which any thread may launch once the context is
/// bound (cudarc leaves `CudaGraph` without `Send` only because it holds raw pointers).
struct Graph(CudaGraph);

pub(crate) const AMAX_PARTS: usize = 64;
unsafe impl Send for Graph {}

/// `--llama-numerics`: quantize activations like llama.cpp's CPU q8_K (one scale per 256) for every integer matmul,
/// and run attention as its CPU flash attention does (LLAMA_Q8 in kernels.cu).
pub fn llama_numerics() -> bool { std::env::var_os("STRATA_LLAMA_NUMERICS").is_some() }

/// KV cache mode (`0 = f16`, `1 = q8`, `2 = q4`, `3 = q2`): read once by Gpu::new and Vk::new.
pub fn kv_mode() -> u32 {
    if let Ok(v) = std::env::var("STRATA_KV") {
        if v == "q2" { return 3; }
        if v == "q4" { return 2; }
        if v == "q8" { return 1; }
        return 0;
    }
    u32::from(std::env::var_os("STRATA_KV_Q8").is_some())
}

/// `--kv q8`: KV cache rows in ggml q8_0 form (`kv_mode() == 1`).
pub fn kv_q8() -> bool { kv_mode() == 1 }

/// `STRATA_GEMM_FMA`: force the CUDA-core FMA fallback in the f16 prefill kernels instead of the sm_70 tensor-core
/// path (`GEMM_TC` in kernels.cu). It covers both the f16 prefill GEMM and the prefill attention (`attn_prefill_t`):
/// the attention's P staging tile is part of every request, so the shared-memory size does not depend on it.
/// Ignored on sm_80+, which always uses `mma.m16n8k16`.
pub fn gemm_fma() -> bool { std::env::var_os("STRATA_GEMM_FMA").is_some() }

/// Prefill attention query / key tile sizes in positions. `FA_BQ` is one warp's 16 query rows times the four warps
/// of the 128-thread block; `FA_BK` is the K/V tile depth and must stay a multiple of 32 for the sm_70 `m8n8k4`
/// and sm_80+ `m16n8k16` fragment loops. Injected into kernels.cu by `source()`, never edited there.
pub(crate) const FA_BQ: usize = 64;
pub(crate) const FA_BK: usize = 32;
/// Prefill GEMM K tile in halves (default, candidate C3 overrides it with `STRATA_GEMM_BK=32`). `GEMM_BM` lives in
/// `backend.rs` because the prefill tile-list builder shares it with Vulkan.
pub(crate) const GEMM_BK: usize = 64;
/// Row padding of the attention shared-memory tiles (`PQ = HD + 8`, `FA_BK + 8` in `attn_prefill_t`).
const ATTN_SMEM_PAD: usize = 8;

/// The kernel tunables `source()` injects as `#define`s. Kept together so a Rust-only edit cannot desync the
/// kernel, and so the C2/C3 candidates can be selected for a measurement window without a code change.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Tune {
    gemm_bk: usize,
    gemm_min_blocks: usize,
    /// M2-T5 candidate: widen the Q4_K header/scale read of the GEMV path to one 16-byte load (`GEMV_LOADW`).
    gemv_loadw: usize,
}

/// Graphs kept by `Gpu::graph`.
const GRAPHS_MAX: usize = 32;

impl Tune {
    /// The shipped values: `GEMM_BK = 64`, no minimum-block hint, the byte-for-byte `QI` load widths.
    #[cfg(test)]
    const fn def() -> Tune { Tune { gemm_bk: GEMM_BK, gemm_min_blocks: 1, gemv_loadw: 0 } }

    /// The window-selected candidates: `STRATA_GEMM_BK` (M2-T3 C3, a multiple of 8), `STRATA_GEMM_MIN_BLOCKS`
    /// (M2-T3 C2, `__launch_bounds__(·, n)` on `gemm_*`/`gemms_*`) and `STRATA_GEMV_LOADW` (M2-T5, 0 or 1).
    /// Invalid values fall back to the defaults.
    fn from_env() -> Tune {
        let gemm_bk = tuned(std::env::var("STRATA_GEMM_BK").ok().as_deref(), GEMM_BK, |v| v > 0 && v % 8 == 0);
        let gemm_min_blocks = tuned(std::env::var("STRATA_GEMM_MIN_BLOCKS").ok().as_deref(), 1, |v| (1..=8).contains(&v));
        let gemv_loadw = tuned(std::env::var("STRATA_GEMV_LOADW").ok().as_deref(), 0, |v| v <= 1);
        Tune { gemm_bk, gemm_min_blocks, gemv_loadw }
    }
}

/// `default` unless `v` parses to a value accepted by `valid`.
pub(crate) fn tuned(v: Option<&str>, default: usize, valid: impl Fn(usize) -> bool) -> usize {
    v.and_then(|s| s.trim().parse::<usize>().ok()).filter(|&x| valid(x)).unwrap_or(default)
}

/// Rows per warp: the family override (`STRATA_RPW_MOE`/`_DENSE`) first, then `STRATA_RPW`, then the shape default.
/// 1..=64 keeps a block's row loop bounded; anything else falls back one level, as `tuned` does.
pub(crate) fn rpw_from(global: Option<&str>, family: Option<&str>, default: usize) -> usize {
    let ok = |v: usize| (1..=64).contains(&v);
    family.and_then(|s| s.trim().parse().ok()).filter(|&v| ok(v))
        .or_else(|| global.and_then(|s| s.trim().parse().ok()).filter(|&v| ok(v)))
        .unwrap_or(default)
}

/// Threads per block of the integer GEMV: the family override first, then the global value, then 256. Only the
/// three warp counts the kernel's row mapping can tile are accepted (4/8/16 warps per block).
pub(crate) fn gemv_block_from(global: Option<&str>, family: Option<&str>) -> u32 {
    let ok = |v: u32| matches!(v, 128 | 256 | 512);
    family.and_then(|s| s.trim().parse::<u32>().ok()).filter(|&v| ok(v))
        .or_else(|| global.and_then(|s| s.trim().parse::<u32>().ok()).filter(|&v| ok(v)))
        .unwrap_or(256)
}

/// Shared memory one `attn_prefill_*` launch requests: `FA_BQ` query rows, `FA_BK` K and `FA_BK` V rows at
/// `hd + 8` padded halves, plus the sm_70 P staging tile of `FA_BQ` x (`FA_BK` + 8) halves. The single place that
/// knows the `qs/ks/vs/ps` layout; both `dyn_shmem_need` and `attn_prefill_m` call it.
fn attn_prefill_smem(hd: usize) -> usize { attn_prefill_smem_bk(hd, FA_BK) }

/// `attn_prefill_smem` for a K/V tile depth of `bk` keys (16 on Turing at head_dim 256, else `FA_BK`).
pub(crate) fn attn_prefill_smem_bk(hd: usize, bk: usize) -> usize {
    let pq = hd + ATTN_SMEM_PAD;
    2 * (FA_BQ + 2 * bk) * pq + 2 * FA_BQ * (bk + ATTN_SMEM_PAD)
}

/// The user-facing error for a device `cc` with an NVRTC that cannot target it, `None` when the pair is fine.
/// CUDA 13 dropped Volta (and older), not Turing, so an sm_70 device must be compiled by NVRTC 12.x; saying it here keeps NVRTC's own
/// `invalid value for --gpu-architecture` from being the first thing the user sees.
fn arch_support_error(cc: (i32, i32), nvrtc: (i32, i32)) -> Option<String> {
    (cc < (7, 5) && nvrtc.0 >= 13).then(|| format!(
        "NVRTC {}.{} cannot compile for sm_{}{}: 13.x removed Volta (sm_70) support; install toolkit/NVRTC 12.x",
        nvrtc.0, nvrtc.1, cc.0, cc.1))
}

/// The user-facing error for a device older than Volta, `None` for sm_70 and newer. The kernels need sm_70
/// instructions (`__nanosleep`, `mma`): without this check NVRTC fails deep in the source on Pascal (sm_6x) and older
/// (`identifier "__nanosleep" is undefined`). Checked before anything is compiled.
fn min_arch_error(name: &str, cc: (i32, i32)) -> Option<String> {
    (cc < (7, 0)).then(|| format!(
        "{name} is sm_{}{}: the CUDA backend requires sm_70 or newer (Volta, Turing, Ampere and later); \
         Pascal and older cards have no tensor cores and are not supported",
        cc.0, cc.1))
}

/// The startup line for the device: arch always, plus the sm_70 prefill GEMM path (tensor cores vs FMA fallback).
/// `gemm_fma` is passed explicitly so the log reflects the cubin actually compiled, not the env at print time.
fn device_log(device: usize, name: &str, arch: &str, cc: (i32, i32), gemm_fma: bool) -> String {
    let path = match (cc, gemm_fma) {
        (c, _) if c < (7, 0) => " (pre-Volta, not supported)",  // not reached from `Gpu::new`: `min_arch_error`
        (c, false) if c < (7, 5) => " (Volta, m8n8k4)",
        (c, true) if c < (7, 5) => " (Volta fallback)",
        (c, _) if c < (8, 0) => " (Turing, m16n8k8)",
        _ => "",  // sm_80+ always uses mma.m16n8k16
    };
    format!("cuda device {device}: {name}, {arch}{path}")
}

/// Whether the device has int8 tensor cores, i.e. whether the `gemmi_*`/`gemmis_*` prefill GEMM exists in its cubin
/// and may be launched. Volta (sm_70) does not: the s8 `mma` shapes need sm_80, so the kernels are compiled out of
/// the sm_70 build (see the `#if __CUDA_ARCH__ >= 800` guards in kernels.cu) and dropped from the loaded name list.
/// This is the CUDA launch gate only: `Gpu::mmq` stays the pure type policy shared with Vulkan and the tests.
fn int8_gemm_arch(cc: (i32, i32)) -> bool { cc.0 >= 8 }

/// The dynamic shared memory a launch of kernel `name` can request, with its family for the startup log: the fixed
/// maximum of the attention prefill (`attn_prefill_smem`, the only place that knows the tile layout) and of the
/// decode GQA kernels (8 query slices per head group plus the cp.async K/V chunks at head_dim 128/256, `attn`),
/// `usize::MAX` for `gemvn_*` whose request depends on the tensor shape (`xq_smem * nt`, `gemv_cfg`), and `None`
/// for every kernel that never asks for more than the 48 KiB default (`attn_decode_*`, `attn_merge`, the rest).
pub(crate) fn dyn_shmem_need(name: &str) -> Option<(&'static str, usize)> {
    let family = if name.starts_with("attn_prefill") { "attn_prefill_*" }
        else if name.starts_with("attn_decode_gqa") { "attn_decode_gqa*" }
        else if name.starts_with("gemvn") { "gemvn_*" }
        else { return None };
    let need = match family {
        "attn_prefill_*" => attn_prefill_smem(name["attn_prefill_".len()..].parse::<usize>().ok()?),
        "attn_decode_gqa*" => {
            let hd = name["attn_decode_gqa".len()..].trim_start_matches('4');
            let hd = hd.strip_prefix('_')?.parse::<usize>().ok()?;
            // attn_decode_gqa4_128 copies one K/V chunk, the generic kernel two: the larger value covers both, and
            // both are below the 48 KiB default at head_dim 64 and 128.
            let cp = if hd == 128 || hd == 256 { 2 * 2 * 32 * hd * 2 } else { 0 };
            8 * hd * 4 + cp
        }
        // The only shape-dependent request: picking the device ceiling keeps every legal launch possible.
        _ => usize::MAX,
    };
    Some((family, need))
}

/// The dynamic shared memory to opt in to for a function with `stat` bytes of static shared memory, or `None` to
/// leave the attribute alone because the 48 KiB default already covers the launch. `need` is clamped to the device
/// opt-in minus the static part: `CU_FUNC_ATTRIBUTE_MAX_DYNAMIC_SHARED_SIZE_BYTES` cannot exceed what the
/// device allows for the block, and on Volta the opt-in ceiling is shared with the static usage (up to 2112 B in
/// `attn_decode_gqa_*`). A clamp below `need` makes the load succeed and a genuinely infeasible launch fail where
/// it is attempted, instead of the whole module failing to load.
fn dyn_shmem_optin(need: usize, stat: usize, optin: usize) -> Option<usize> {
    const DEFAULT: usize = 48 << 10;
    (need > DEFAULT).then(|| need.min(optin.saturating_sub(stat)))
}

/// `STRATA_SMEM_CARVEOUT`: the default `off` leaves the driver default, which the M2-T3 window measured as the
/// fastest (C1 was rejected); `policy` enables the R3 decision, `min` additionally picks the minimal fitting
/// carveout for the static-only GEMM families (C1's gemm half).
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum CarveoutMode { Off, Policy, Min }

fn carveout_mode_from(v: Option<&str>) -> CarveoutMode {
    match v.map(str::trim) {
        Some("policy") => CarveoutMode::Policy,
        Some("min") => CarveoutMode::Min,
        _ => CarveoutMode::Off,
    }
}

fn carveout_mode() -> CarveoutMode { carveout_mode_from(std::env::var("STRATA_SMEM_CARVEOUT").ok().as_deref()) }

/// The preferred shared-memory carveout percentage for a function, or `None` to leave the driver's default.
/// `dyn`/`stat` are one block's dynamic/static usage, `blocks_wanted` the number of blocks per SM the carveout is
/// meant to hold, `optin` the device's per-SM shared-memory maximum (96 KiB on the V100). Returns `None` when the
/// 48 KiB default already holds every wanted block (nothing to gain, e.g. the register-bound `gemm_*`) or when the
/// wanted blocks cannot fit at all (hd 256 attention: two blocks need 142 KiB). Otherwise the percentage is rounded
/// up so it never requests less than the launch needs, and clamped to `1..=100`.
fn carveout(dynamic: usize, stat: usize, blocks_wanted: usize, optin: usize) -> Option<i32> {
    const DEFAULT: usize = 48 << 10;
    if blocks_wanted == 0 || optin == 0 { return None; }
    let need = dynamic.checked_add(stat)?;
    if need == 0 { return None; }
    let total = need.checked_mul(blocks_wanted)?;
    if total <= DEFAULT { return None; }  // the default allocation already fits: request it nowhere
    (total <= optin).then(|| (total * 100).div_ceil(optin).clamp(1, 100) as i32)
}

/// The smallest carveout that still fits `need` bytes, used only by C1's `min` mode for the static-only GEMM
/// families: the driver's default allocation may reserve more shared memory than 27 KiB, and every KiB not carved
/// out is L1 for the weight stream.
fn carveout_min(need: usize, optin: usize) -> Option<i32> {
    (need > 0 && need <= optin).then(|| (need * 100).div_ceil(optin).clamp(1, 100) as i32)
}

/// The carveout to set on kernel `name`: `need` is its `dyn_shmem_need` (the launch's dynamic request, `None` for
/// kernels that use only static shared memory).
fn carveout_for(name: &str, need: Option<usize>, stat: usize, optin: usize, mode: CarveoutMode) -> Option<i32> {
    if mode == CarveoutMode::Off { return None; }
    let (dynamic, blocks) = match need {
        // hd 128 gains a second resident block at 96 KiB; hd 64 already fits two in the default; hd 256 cannot fit
        // two at all (`carveout` returns None), so the request leaves it unchanged.
        Some(v) if name.starts_with("attn_prefill") => (v, 2),
        Some(v) => (v, 1),
        None => (0, 1),
    };
    if let Some(p) = carveout(dynamic, stat, blocks, optin) { return Some(p); }
    let gemm = name.starts_with("gemm_") || name.starts_with("gemms_");
    (mode == CarveoutMode::Min && gemm).then(|| carveout_min(dynamic + stat, optin)).flatten()
}

/// One startup line per kernel family: the device's shared-memory opt-in (not a table), what the load set for the
/// family's largest static usage, and the preferred carveout. Quotable in an issue or PR as the numbers actually
/// observed on the card.
fn smem_log(family: &str, optin: usize, stat: usize, set: Option<usize>, carveout: Option<i32>) -> String {
    let dynamic = set.map_or_else(|| "default (48 KiB)".to_string(), |v| format!("{v} B"));
    let carveout = carveout.map_or_else(|| "default".to_string(), |p| format!("{p}%"));
    format!("cuda shared memory: opt-in {optin} B per block, {family}: {stat} B static, dynamic max {dynamic}, carveout {carveout}")
}

/// The kernel names to load from the cubin: the full list on sm_80+, without the int8 GEMM (`gemmi_*`, `gemmis_*` and
/// `gemmd_*`, 23 non-float types x 3 fewer names) on a device where that code is not compiled in. Pure so a CPU test can pin
/// both lists (393 / 324 names).
pub(crate) fn kernel_names(sm80: bool) -> Vec<String> {
    let int8 = move |&&t: &&GgmlType| sm80 && !t.float_x();
    KERNELS.iter().map(|s| s.to_string())
        .chain(QTYPES.iter().flat_map(|&t| ["dequant", "gemv", "gemvn", "moe_gemv", "gemm", "gemms"].map(|p| kname(p, t))))
        .chain(QTYPES.iter().flat_map(|&t| ["gemv", "gemvn", "moe_gemv"].map(|p| kname(p, t) + "_p")))
        .chain(QTYPES.iter().filter(int8).flat_map(|&t| [kname("gemmi", t), kname("gemmis", t), kname("gemmd", t)]))
        .chain([64, 128, 256, 512].iter().flat_map(|h| [format!("attn_decode_{h}"), format!("attn_decode_gqa_{h}")]))
        .chain(["attn_decode_gqa4_128".to_string()])
        .collect()
}

/// Cubin cache key: the compiled source, the target arch and the NVRTC version that produced it. The version is part
/// of it so a cubin written by another toolkit is never loaded (a 13.x sm_70 cubin must not hide the unsupported
/// toolkit, and machine code from an older compiler must not survive an upgrade). Detected NVRTC absent -> tag
/// `"unverified"`: any cached entry is then only reused without a version check, never in place of the missing-
/// library error below.
fn cache_key(src: &str, arch: &str, nvrtc: &str) -> u64 {
    let mut h = std::collections::hash_map::DefaultHasher::new();
    (src, arch, nvrtc).hash(&mut h);
    h.finish()
}

/// Compile `src` to a cubin for `arch`, reusing `$XDG_CACHE_HOME/rusting_strata/<hash>.cubin` when present. A cubin,
/// unlike PTX, needs no driver JIT at load (~6 s per start whenever the driver's own JIT cache missed). The hash
/// covers `(src, arch, NVRTC version)`; when NVRTC cannot be loaded the tag is `"unverified"`, so a cache entry is
/// no longer reusable and the run reports the missing library instead of silently reusing a cubin.
fn ptx(src: &str, arch: &'static str, cc: (i32, i32)) -> anyhow::Result<Ptx> {
    let nvrtc = nvrtc::version();
    // A device/NVRTC mismatch is fatal: no "CUDA init" context, or `backend::open` and the tests swallow it as a skip.
    if let Some(nv) = nvrtc {
        if let Some(msg) = arch_support_error(cc, nv) {
            return Err(anyhow::anyhow!("{msg}"));
        }
    }
    let tag = nvrtc.map(|(maj, min)| format!("{maj}.{min}")).unwrap_or_else(|| "unverified".to_string());
    let tag = format!("{tag} {}", nvrtc::extra_opts()).trim_end().to_string();
    let dir = std::env::var_os("XDG_CACHE_HOME").map(Into::into)
        .unwrap_or_else(|| std::path::PathBuf::from(std::env::var_os("HOME").unwrap_or_default()).join(".cache"))
        .join("rusting_strata");
    let path = dir.join(format!("{:016x}.cubin", cache_key(src, arch, &tag)));
    if let Ok(b) = std::fs::read(&path) {
        // Refresh mtime so pruning below keeps a cubin that is still in use.
        let _ = std::fs::File::options().append(true).open(&path).and_then(|f| f.set_modified(std::time::SystemTime::now()));
        return Ok(Ptx::from_binary(b));
    }
    // "CUDA init" lets `backend::open` fall back to Vulkan.
    if !nvrtc::present() { return Err(nvrtc::missing()).context("CUDA init"); }
    prune_ptx(&dir, std::time::Duration::from_secs(7 * 86400));
    crate::say!("compiling CUDA kernels for {arch} with NVRTC {tag} (first run only, ~15 s)");
    let b = nvrtc::cubin(src, arch)?;
    let _ = std::fs::create_dir_all(&dir).and_then(|_| std::fs::write(&path, &b));
    Ok(Ptx::from_binary(b))
}

/// Delete cached `.ptx`/`.cubin` files in `dir` not used for `age`; every kernel edit leaves a stale file behind.
fn prune_ptx(dir: &std::path::Path, age: std::time::Duration) {
    let Ok(rd) = std::fs::read_dir(dir) else { return };
    for e in rd.flatten() {
        let p = e.path();
        let old = e.metadata().and_then(|m| m.modified()).ok().and_then(|t| t.elapsed().ok()).is_some_and(|d| d > age);
        if old && p.extension().is_some_and(|x| x == "ptx" || x == "cubin") { let _ = std::fs::remove_file(p); }
    }
}

pub fn cfg(grid: (u32, u32), block: u32, smem: u32) -> LaunchConfig {
    LaunchConfig { grid_dim: (grid.0, grid.1, 1), block_dim: (block, 1, 1), shared_mem_bytes: smem }
}

/// Launch kernel `$name` with `cfg`; every argument is evaluated into a local so the builder can borrow it.
/// `STRATA_CUDA_SYNC`: wait for every kernel right after its launch, so a device error (reported at the next sync
/// otherwise) names the kernel that raised it. Debugging only: no graphs, and every launch pays a stream sync.
static SYNC_EACH: std::sync::LazyLock<bool> = std::sync::LazyLock::new(|| std::env::var_os("STRATA_CUDA_SYNC").is_some());

macro_rules! launch {
    ($g:expr, $name:expr, $cfg:expr, $($a:expr),* $(,)?) => {{
        let f = $g.f($name);
        let mut b = $g.st().launch_builder(f);
        $( let v = $a; b.arg(&v); )*
        unsafe { b.launch($cfg) }.with_context(|| format!("launch {}", $name))?;
        if *SYNC_EACH {
            $g.stream.synchronize().and_then(|_| $g.side.synchronize()).with_context(|| format!("kernel {} failed", $name))?;
        }
    }};
}

pub(crate) fn kname(prefix: &str, ty: GgmlType) -> String { format!("{prefix}_{}", ty.name().to_lowercase()) }

/// The UUID of every CUDA device in CUDA's order, the order `Gpu::new` counts in: `None` for a device the driver
/// gives none for, empty when CUDA does not load. No context is created. NVIDIA's Vulkan driver reports the same 16
/// bytes as `deviceUUID` (NVIDIA's simpleVulkan sample picks its CUDA device by comparing the two), so `Vk::new`
/// finds the card of a CUDA index with them.
pub fn device_uuids() -> Vec<Option<[u8; 16]>> {
    if !unsafe { sys::is_culib_present() } || result::init().is_err() { return vec![]; }
    // Looked up by hand: cudarc panics on a symbol the driver lacks, and this one came with CUDA 11.4.
    type GetUuid = unsafe extern "C" fn(*mut sys::CUuuid, sys::CUdevice) -> sys::CUresult;
    let get = unsafe { sys::culib().get::<GetUuid>(b"cuDeviceGetUuid_v2\0") }.ok().map(|f| *f);
    (0..result::device::get_count().unwrap_or(0)).map(|i| {
        let mut u = sys::CUuuid { bytes: [0; 16] };
        let ok = get.is_some_and(|f| result::device::get(i).is_ok_and(|d| unsafe { f(&mut u, d) } == sys::CUresult::CUDA_SUCCESS));
        ok.then(|| u.bytes.map(|b| b as u8))
    }).collect()
}
