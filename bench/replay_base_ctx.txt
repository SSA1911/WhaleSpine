fn set_kv_env(kv: &str) {
    match kv {
        "q8" | "q4" | "q2" => {
            unsafe { std::env::set_var("STRATA_KV", kv) };
            if kv == "q8" { unsafe { std::env::set_var("STRATA_KV_Q8", "1") }; }
            else { unsafe { std::env::remove_var("STRATA_KV_Q8") }; }
        }
        _ => {
            unsafe { std::env::remove_var("STRATA_KV") };
            unsafe { std::env::remove_var("STRATA_KV_Q8") };
        }
    }
}

/// `Own`: a model outside the main engine (`config::OWN_ARCHS`). Its chat answers a request whole, so the token
/// methods below do not apply to it.
enum Runner<'a> { Cpu(&'a Model<'a>, State), Gpu(Box<GpuModel<'a>>), Slot(Client), Own(serve::Own<'a>) }

impl Runner<'_> {
    fn step(&mut self, t: u32) -> anyhow::Result<Vec<f32>> {
        match self { Runner::Cpu(m, st) => Ok(m.step(st, t)), Runner::Gpu(g) => g.step(t), Runner::Slot(c) => Ok(c.step(t, true)?.1), Runner::Own(_) => anyhow::bail!("step on a model outside the main engine") }
    }
    /// Feed a prompt; returns the logits after its last token.
    fn prefill(&mut self, ts: &[u32]) -> anyhow::Result<Vec<f32>> {
        match self {
            Runner::Cpu(m, st) => Ok(ts.iter().map(|&t| m.step(st, t)).last().unwrap_or_default()),
            Runner::Gpu(g) => g.prefill(ts),
            Runner::Slot(_) | Runner::Own(_) => anyhow::bail!("prefill on a slot or a model outside the main engine"),
        }
    }
    /// Feed a chat prompt as a fresh sequence, reusing the earlier request's state for the shared message history
    /// (GPU; `history` is the prompt without the assistant header, `system` its system and tools block, kept for
    /// later requests that rewrite the history). Returns the logits and the number of tokens reused.
    fn chat(&mut self, tok: &Tokenizer, prompt: &str, history: &str, system: &str) -> anyhow::Result<(Vec<u32>, Vec<f32>, usize)> {
        let ids = tok.encode(prompt)?;
        let h = tok.encode(history)?;
        let keep = if ids.starts_with(&h) { h.len() } else { 0 };
        let s = if system.is_empty() { vec![] } else { tok.encode(system)? };
        let sys = if ids.starts_with(&s) && s.len() <= keep { s.len() } else { 0 };
        let (logits, reused) = match self {
            Runner::Gpu(g) => g.prefill_cached(&ids, sys, keep)?,
            Runner::Slot(c) => c.chat(&ids, sys, keep)?,
            _ => { self.reset()?; (self.prefill(&ids)?, 0) }
        };
        Ok((ids, logits, reused))
    }
    /// Feed a prompt that contains image embeddings as a fresh sequence.
    fn chat_image(&mut self, p: &vision::Prompt, embs: &[Vec<Vec<f32>>]) -> anyhow::Result<Vec<f32>> {
        match self {
            Runner::Cpu(m, st) => { *st = State::new(&m.cfg); vision::feed_prompt(m, st, p, embs) }
            Runner::Gpu(g) => g.prefill_image(p, embs),
            Runner::Slot(_) | Runner::Own(_) => anyhow::bail!("image input is not supported on a serve slot"),
        }
    }
    /// Forget the sequence so far.
    fn reset(&mut self) -> anyhow::Result<()> {
        match self { Runner::Cpu(m, st) => { *st = State::new(&m.cfg); Ok(()) } Runner::Gpu(g) => g.reset(), Runner::Slot(_) | Runner::Own(_) => Ok(()) }
    }
    /// Seconds spent waiting on CPU experts so far (GPU only).
    fn cpu_secs(&self) -> f64 { if let Runner::Gpu(g) = self { g.parts().map(|p| p.cpu_time.as_secs_f64()).sum() } else { 0.0 } }
    /// Bytes of routed-expert weights one decode token reads from the cache (0 without an expert cache).
    fn expert_bytes_per_token(&self) -> usize {
        if let Runner::Gpu(g) = self { g.parts().map(|p| p.expert_bytes_per_token()).sum() } else { 0 }
    }
    /// Split model: seconds per part but the last (forward until its GPU is done, copy to the next part).
    fn split_secs(&self) -> Vec<(f64, f64)> {
        let Runner::Gpu(g) = self else { return vec![] };
        let mut v: Vec<_> = g.parts().map(|p| (p.split_time.0.as_secs_f64(), p.split_time.1.as_secs_f64())).collect();
        v.pop();
        v
    }
    /// Backend name of the first part (`"cpu"` on the CPU, `"slot"` for a serve slot).
    fn backend_name(&self) -> &'static str {
        match self {
            Runner::Cpu(..) => "cpu",
            Runner::Gpu(g) => g.parts().next().map_or("cpu", |p| p.gpu.name()),
            Runner::Slot(_) => "slot",
            Runner::Own(_) => "own",
        }
    }
    /// Expert-cache counters summed over the parts, for `finish()` and the benchmark JSON (`None` off the GPU).
    fn expert_stats(&self) -> Option<ExpertStats> {
        let Runner::Gpu(g) = self else { return None };
        let sum = |f: &dyn Fn(&GpuModel) -> usize| g.parts().map(f).sum::<usize>();
        let parts: Vec<_> = g.parts().collect();
        let cached: Vec<_> = parts.iter().filter(|p| p.cache.is_some()).collect();
        Some(ExpertStats {
            hits: sum(&|p| p.hits),
            misses: sum(&|p| p.misses),
            swaps: sum(&|p| p.cache.as_ref().map_or(0, |c| c.swaps)),
            resident: sum(&|p| p.cache.as_ref().map_or(0, |c| c.resident_count())),
            total: sum(&|p| p.cache.as_ref().map_or(0, |c| c.total_experts())),
            bytes_per_token: sum(&|p| p.expert_bytes_per_token()),
            cached: !cached.is_empty(),
            all_fast: cached.iter().all(|p| p.expert_fast_path()),
        })
    }
    /// The same counters, but only when an expert cache exists (the benchmark record's top-level `expert`).
    fn expert_summary(&self) -> Option<ExpertStats> {
        self.expert_stats().filter(|s| s.cached)
    }
    fn finish(&self) -> anyhow::Result<()> {
        if let Runner::Gpu(g) = self {
            let sum = |f: &dyn Fn(&GpuModel) -> usize| g.parts().map(f).sum::<usize>();
            let st = self.expert_stats().expect("Runner::Gpu has expert stats");
            if g.drafts.0 > 0 {
                let kind = if g.has_dflash() && g.ngram.is_some() {
                    "FlashCascade (DFlash + suffix/ngram)"
                } else if g.has_dflash() {
                    "DFlash"
                } else if g.has_ext_draft() && g.ngram.is_some() {
                    "FlashCascade (external draft + suffix/ngram)"
                } else if g.has_ext_draft() {
                    "external draft"
                } else if g.ngram.is_some() {
                    "speculative ngram"
                } else {
                    "MTP"
                };
                say!("{kind} drafts accepted {}/{} ({:.1}%)", g.drafts.1, g.drafts.0, 100.0 * g.drafts.1 as f64 / g.drafts.0 as f64);
            }
            say!("expert hit rate {:.1}% ({} misses), cache swaps {}", 100.0 * st.hits as f64 / (st.hits + st.misses).max(1) as f64, st.misses, st.swaps);
            // Issue #34 R1: printed on every run, so a fully resident run reads as a measured 0 in the log instead of
            // an absent line. `tokens` counts the token forwards the worker could have been asked about.
            let tokens = sum(&|p| p.tokens);
            let busy: f64 = g.parts().map(|p| p.cpu_time.as_secs_f64()).sum();
            let stalls = sum(&|p| p.resumed.get());
            say!("worker busy {:.1} µs/token, worker stalls {stalls}", if tokens > 0 { 1e6 * busy / tokens as f64 } else { 0.0 });
            let mode = if !st.cached { "no expert cache".to_string() }
                else if st.all_fast { format!("fast path on, {}/{} experts resident", st.resident, st.total) }
                else { format!("fast path off, {}/{} experts resident", st.resident, st.total) };
            say!("mailbox traffic: {} GPU writes, {} host routing reads ({mode})", sum(&|p| p.mailbox_writes.get()), sum(&|p| p.mailbox_reads.get()));
            for p in g.parts() { if let Some(c) = &p.cache { c.save_profile()?; } }
        }
        Ok(())
    }
}

/// Chunk variables an `own_calib` call set, with the value each had before it. `serve` loads several models in one
/// process; without restoring them, a model with no calibration runs with the previous model's chunk.
static CALIB_ENV: std::sync::Mutex<Vec<(&'static str, Option<std::ffi::OsString>)>> = std::sync::Mutex::new(Vec::new());

/// Undo the chunk variables set by the previous `own_calib` call, before a new model applies its own calibration.
fn restore_calib_env() {
    let saved = std::mem::take(&mut *CALIB_ENV.lock().unwrap_or_else(|e| e.into_inner()));
    for (key, old) in saved {
        // SAFETY: called before a model is loaded, before any thread of that model reads these variables.
        unsafe { match old { Some(v) => std::env::set_var(key, v), None => std::env::remove_var(key) } }
    }
}

impl Engine {
    /// Hand `--backend`, `--device`, and `--kv` to the DeepSeek, GLM, Gemma 4 and Inkling 2 paths.
    fn ds_env(&self) {
        // SAFETY: called at the start of a command, before any thread reads these variables.
        if let Some(b) = &self.backend { unsafe { std::env::set_var("STRATA_DS_BACKEND", b) } }
        if self.device != [0] { unsafe { std::env::set_var("STRATA_DS_DEVICES", self.device.iter().map(|d| d.to_string()).collect::<Vec<_>>().join(",")) } }
        if self.fixed_experts { unsafe { std::env::set_var("STRATA_FIXED_EXPERTS", "1") } }  // Hy4: every expert on the GPU
        if self.kv != "f16" && std::env::var_os("STRATA_KV").is_none() { set_kv_env(&self.kv); }
    }

    /// Models on their own code path never call `Engine::runner`, the only place that reads a calibration file.
    /// Apply the measured chunk through the environment those engines actually read. Context and threads are not
    /// searched (they are memory choices), so this never touches the global CTX: doing that once made an explicit
    /// --ctx look implicit to the main engine's runner.
    fn own_calib(&self, model: &Path) {
        restore_calib_env();
        let Ok(g) = Gguf::open(model) else { return };
        let arch = g.get("general.architecture").ok().and_then(|v| v.as_str()).unwrap_or("");
        if !rusting_strata::config::OWN_ARCHS.contains(&arch) { return; }
        // Like `runner`, look up the file under the backend that really opened: with no --backend, CUDA may fall
        // back to Vulkan, and `calibrate` writes the file under that name.
        let Some(&d) = self.device.first() else { return };
        let Ok(b) = backend::open(self.backend.as_deref(), d) else { return };
        let path = calibration_path(model, b.name(), &self.device);
        let Ok(bytes) = std::fs::read(&path) else { return };
        let Ok(v) = serde_json::from_slice::<serde_json::Value>(&bytes) else { return };
        let gpu = b.device();
        if !calibration_fits(&v, &gpu) { return; }
        if v["bytes"].as_u64().is_some_and(|n| std::fs::metadata(model).is_ok_and(|m| m.len() != n)) { return; }
        // GLM reads its own variable, Gemma 4 its own, the others read STRATA_DS_CHUNK. Explicit values win.
        if let Some(n) = v["chunk"].as_u64() {
            let mut saved = CALIB_ENV.lock().unwrap_or_else(|e| e.into_inner());
            for (k, value) in chunk_env_updates(n as usize, |k| std::env::var_os(k).is_some()) {
                saved.push((k, std::env::var_os(k)));
                unsafe { std::env::set_var(k, value); }
            }
        }
    }

    /// Fills settings not given on the command line from the model's calibration file, else built-in defaults.
    fn ram_cap(&self) -> Option<u64> { memory::ram_cap((self.ram_margin as u64) << 20, (self.max_ram as u64) << 20) }
    /// A copy with the load options of an `/api/load` body applied (`ctx`, `kv`, `cache_ram`, `vram_margin`, `mtp`,

impl DFlash2Selector {
    fn dequant_codebook_row(raw: &[u8], ty: GgmlType, tok: u32, rank: usize, dst: &mut [f32]) {
        let rb = ty.row_bytes(rank as u64);
        let off = tok as usize * rb;
        if off + rb <= raw.len() {
            crate::quant::dequant_row(ty, &raw[off..off + rb], dst);
        } else {
            dst.fill(0.0);
        }
    }

    fn select_path(
        &self,
        anchor_tok: u32,
        k_eff: usize,
        logits: &[f32],
        row_len: usize,
        short_ids: Option<&[u32]>,
        h_proj: &[f32],
        out_mult: f32,
        smp: &mut Sampler,
    ) -> anyhow::Result<(Vec<u32>, Vec<Vec<(usize, f32)>>)> {
        let k_top = self.top_k.min(row_len).max(1);
        let rank = self.rank;
        let mut pred = anchor_tok;
        let mut toks = Vec::with_capacity(k_eff);
        let mut q_dists = Vec::new();
        let mut pred_vec = vec![0f32; rank];
        let mut succ_vec = vec![0f32; rank];
        let mut q_vec = vec![0f32; rank];
        let mut row_cands: Vec<(usize, f32)> = Vec::with_capacity(row_len);
        let mut scores = vec![0f32; k_top];

        for pos in 0..k_eff {
            let row = &logits[pos * row_len..(pos + 1) * row_len];
            row_cands.clear();
            for (idx, &v) in row.iter().enumerate() {
                row_cands.push((idx, v));
            }
            if row_cands.len() > k_top {
                row_cands.select_nth_unstable_by(k_top - 1, |a, b| b.1.total_cmp(&a.1));
                row_cands.truncate(k_top);
            }
            Self::dequant_codebook_row(&self.pred_raw, self.pred_ty, pred, rank, &mut pred_vec);
            let hp = &h_proj[pos * rank..(pos + 1) * rank];
            for r in 0..rank {
                q_vec[r] = pred_vec[r] * hp[r];
            }
            let mut best_i = 0usize;
            let mut best_s = f32::NEG_INFINITY;
            for (j, &(idx, u)) in row_cands.iter().enumerate() {
                let tok_id = short_ids.map_or(idx as u32, |ids| ids[idx]);
                Self::dequant_codebook_row(&self.succ_raw, self.succ_ty, tok_id, rank, &mut succ_vec);
                let dot: f32 = q_vec.iter().zip(&succ_vec).map(|(&a, &b)| a * b).sum();
                let s = u * out_mult + dot;
                scores[j] = s;
                if s > best_s {
                    best_s = s;
                    best_i = j;
                }
            }
            if smp.temp <= 0.0 {
                let chosen_idx = row_cands[best_i].0;
                pred = short_ids.map_or(chosen_idx as u32, |ids| ids[chosen_idx]);
                toks.push(pred);
            } else {
                let (loc_i, q_local) = smp.draft_dist(&scores[..row_cands.len()])?;
                let chosen_idx = row_cands[loc_i as usize].0;
                pred = short_ids.map_or(chosen_idx as u32, |ids| ids[chosen_idx]);
                toks.push(pred);
                let q_mapped = q_local
                    .into_iter()
                    .map(|(j, p)| {
                        let idx = row_cands[j].0;
                        (short_ids.map_or(idx, |ids| ids[idx] as usize), p)
                    })
                    .collect();
                q_dists.push(q_mapped);
            }
        }
        Ok((toks, q_dists))
    }
}

/// The draft file's metadata sizes the DFlash buffers and the kernels write its tensors' outputs through raw
/// addresses, so the two must agree (and fit the target's shared buffers) before anything is allocated.
pub(crate) fn dflash_check(m: &Model, dc: &crate::config::DFlashConfig) -> anyhow::Result<()> {
    let c = &m.cfg;
    let (ne, q, kv, hd, ff) = (dc.n_embd, dc.n_head * dc.head_dim, dc.n_head_kv * dc.head_dim, dc.head_dim, dc.n_ff);
    ensure!(dc.n_layer > 0 && ne == c.n_embd && hd > 0 && dc.n_head_kv > 0 && dc.n_head.is_multiple_of(dc.n_head_kv) && dc.n_head > 0
        && dc.n_rot <= hd && ff > 0 && (dc.mask_token_id as usize) < c.n_vocab,
        "draft model: bad DFlash metadata (layers {}, embd {ne} for a {} target, heads {}/{}x{hd}, rope {}, ff {ff}, mask token {})",
        dc.n_layer, c.n_embd, dc.n_head, dc.n_head_kv, dc.n_rot, dc.mask_token_id);
    let dim = |name: String, rows: usize, cols: usize| -> anyhow::Result<()> {
        let t = m.w.get(&name)?;
        ensure!(t.rows == rows && t.cols == cols, "draft model: {name} is {}x{}, its metadata needs {rows}x{cols}", t.rows, t.cols);
        Ok(())
    };
    dim("dflash.fc.weight".into(), ne, dc.target_layer_ids.len() * c.n_embd)?;
    for n in ["hidden_norm", "norm"] { dim(format!("dflash.{n}.weight"), 1, ne)?; }
    let (k, g) = (dc.conv_kernel_size, dc.conv_group_size);
    ensure!(k == 0 || (g > 0 && ne.is_multiple_of(g)), "draft model: conv group size {g} does not divide {ne}");
    for il in 0..dc.n_layer {
        let l = |s: &str, rows, cols| dim(format!("dflash.blk.{il}.{s}.weight"), rows, cols);
        for (s, rows, cols) in [("attn_norm", 1, ne), ("ffn_norm", 1, ne), ("attn_q_norm", 1, hd), ("attn_k_norm", 1, hd), ("attn_q", q, ne),
            ("attn_k", kv, ne), ("attn_v", kv, ne), ("attn_output", ne, q), ("ffn_gate", ff, ne), ("ffn_up", ff, ne), ("ffn_down", ne, ff)] {
            l(s, rows, cols)?;
        }
        if k > 0 {
            for s in ["attn_conv", "mlp_conv"] {
                l(&format!("{s}_proj"), 2 * k * (ne / g), ne)?;
                let b = m.w.get(&format!("dflash.blk.{il}.{s}_base.weight"))?;  // [2, K, H], indexed flat
                ensure!(b.rows * b.cols == 2 * k * ne, "draft model: layer {il} {s}_base has {} values, its metadata needs {}", b.rows * b.cols, 2 * k * ne);
            }
        }
    }
    if dc.selector_rank > 0 { dim("dflash.selector.hidden_proj.weight".into(), dc.selector_rank, ne)?; }
    Ok(())
}

fn build_shortlist(m: &Model, gpu: &dyn Backend, output: Option<&GpuTensor>) -> anyhow::Result<Option<Shortlist>> {
    if std::env::var_os("STRATA_MTP_FULL_VOCAB").is_some() { return Ok(None); }
    let c = &m.cfg;
    if c.n_vocab <= 65536 || !gpu.supports_gemv_rows() { return Ok(None); }
    let Some(out) = output else { return Ok(None); };
    if out.ty.float_x() { return Ok(None); }
    let Ok(crate::gguf::Value::Arr(arr)) = m.w.g.get("tokenizer.ggml.tokens") else { return Ok(None); };
    if arr.len() != c.n_vocab { return Ok(None); }

    let mut u2b = [0xFFFFu16; 512];
    for ch in (b'!'..=b'~').chain(0xA1..=0xAC).chain(0xAE..=0xFF) {
        u2b[ch as usize] = ch as u16;
    }
    let mut next_cp = 256usize;
    for b in 0..=255u8 {
        if u2b[b as usize] == 0xFFFF { u2b[next_cp] = b as u16; next_cp += 1; }
    }
    let decode_bpe = |s: &str| -> Vec<u8> {
        let mut raw = Vec::with_capacity(s.len());
        for ch in s.chars() {
            let cu = ch as usize;
            if cu < 512 && u2b[cu] != 0xFFFF { raw.push(u2b[cu] as u8); }
            else { return s.as_bytes().to_vec(); }
        }
        raw
    };

    let mut ids = Vec::with_capacity(c.n_vocab);
    let mut in_set = vec![false; c.n_vocab];
    for (i, v) in arr.iter().enumerate() {
        let Some(s) = v.as_str() else { return Ok(None); };
        let keep = i < 4096
            || s.starts_with("<|")
            || matches!(s, "<think>" | "</think>")
            || {
                let raw = decode_bpe(s);
                match std::str::from_utf8(&raw) {
                    Ok(txt) => txt.chars().all(|ch| (ch as u32) <= 0x03FF || (0x2000..=0x2BFF).contains(&(ch as u32))),
                    Err(_) => raw.len() <= 2,
                }
            };
        if keep {
            ids.push(i as u32);
            in_set[i] = true;
        }
    }
    if ids.is_empty() || ids.len() >= c.n_vocab * 4 / 5 { return Ok(None); }
    let gpu_ids: Vec<i32> = ids.iter().map(|&x| x as i32).collect();
    let buf = gpu.zeros::<i32>(gpu_ids.len())?;
    gpu.put(&gpu_ids, &buf, 0)?;
    Ok(Some(Shortlist { buf, ids, in_set }))
}

/// Why `pd` cannot run the kernels, if it cannot.
fn unusable(instance: &ash::Instance, pd: vk::PhysicalDevice) -> Option<String> {
    let mut sg = vk::PhysicalDeviceSubgroupSizeControlProperties::default();
    let mut p11 = vk::PhysicalDeviceVulkan11Properties::default();
    let mut p2 = vk::PhysicalDeviceProperties2::default().push_next(&mut sg).push_next(&mut p11);
    unsafe { instance.get_physical_device_properties2(pd, &mut p2) };
    let p = p2.properties;
    if p.device_type == vk::PhysicalDeviceType::CPU { return Some("a CPU device".into()); }
    if p.api_version < vk::API_VERSION_1_3 { return Some("Vulkan 1.3 is needed".into()); }
    if !((sg.min_subgroup_size..=sg.max_subgroup_size).contains(&32) && sg.required_subgroup_size_stages.contains(vk::ShaderStageFlags::COMPUTE))
        && !(sg.min_subgroup_size == 64 && sg.max_subgroup_size == 64 && p11.subgroup_size == 64) {
        return Some(format!("subgroup size 32 or fixed size 64 is needed (it has {}-{})", sg.min_subgroup_size, sg.max_subgroup_size));
    }
    let ops = vk::SubgroupFeatureFlags::ARITHMETIC | vk::SubgroupFeatureFlags::SHUFFLE;
    if !p11.subgroup_supported_operations.contains(ops) || !p11.subgroup_supported_stages.contains(vk::ShaderStageFlags::COMPUTE) {
        return Some("subgroup arithmetic and shuffle in compute shaders are needed".into());
    }
    let (mut f11, mut f12, mut f13) = Default::default();
    let mut f = features(&mut f11, &mut f12, &mut f13);
    unsafe { instance.get_physical_device_features2(pd, &mut f) };
    let (int16, int64, f64) = (f.features.shader_int16, f.features.shader_int64, f.features.shader_float64);
    let missing: Vec<_> = [
        ("shaderInt16", int16), ("shaderInt64", int64), ("shaderFloat64", f64), ("storageBuffer16BitAccess", f11.storage_buffer16_bit_access),
        ("bufferDeviceAddress", f12.buffer_device_address), ("storageBuffer8BitAccess", f12.storage_buffer8_bit_access),
        ("shaderInt8", f12.shader_int8), ("shaderFloat16", f12.shader_float16), ("vulkanMemoryModel", f12.vulkan_memory_model),

        ("shaderIntegerDotProduct", f13.shader_integer_dot_product),
    ].into_iter().filter(|f| f.1 == vk::FALSE).map(|f| f.0).collect();
    (!missing.is_empty()).then(|| format!("missing {}", missing.join(", ")))
}

/// Shared memory one `attn_prefill_*` launch requests: `FA_BQ` query rows, `FA_BK` K and `FA_BK` V rows at

fn attn_prefill_smem(hd: usize) -> usize { attn_prefill_smem_bk(hd, FA_BK) }

/// `attn_prefill_smem` for a K/V tile depth of `bk` keys (16 on Turing at head_dim 256, else `FA_BK`).
pub(crate) fn attn_prefill_smem_bk(hd: usize, bk: usize) -> usize {
    let pq = hd + ATTN_SMEM_PAD;
    2 * (FA_BQ + 2 * bk) * pq + 2 * FA_BQ * (bk + ATTN_SMEM_PAD)
}

/// The user-facing error for a device `cc` with an NVRTC that cannot target it, `None` when the pair is fine.

        (c, true) if c < (7, 5) => " (Volta fallback)",
        (c, _) if c < (8, 0) => " (Turing, m16n8k8)",
        _ => "",  // sm_80+ always uses mma.m16n8k16

/// Whether the device has int8 tensor cores, i.e. whether the `gemmi_*`/`gemmis_*` prefill GEMM exists in its cubin
/// and may be launched. Volta (sm_70) does not: the s8 `mma` shapes need sm_80, so the kernels are compiled out of

fn int8_gemm_arch(cc: (i32, i32)) -> bool { cc.0 >= 8 }

/// The dynamic shared memory a launch of kernel `name` can request, with its family for the startup log: the fixed
/// maximum of the attention prefill (`attn_prefill_smem`, the only place that knows the tile layout) and of the
/// decode GQA kernels (8 query slices per head group plus the cp.async K/V chunks at head_dim 128/256, `attn`),
/// `usize::MAX` for `gemvn_*` whose request depends on the tensor shape (`xq_smem * nt`, `gemv_cfg`), and `None`
/// for every kernel that never asks for more than the 48 KiB default (`attn_decode_*`, `attn_merge`, the rest).
pub(crate) fn dyn_shmem_need(name: &str) -> Option<(&'static str, usize)> {
    let family = if name.starts_with("attn_prefill") { "attn_prefill_*" }
        else if name.starts_with("attn_decode_gqa") { "attn_decode_gqa*" }
        else if name.starts_with("gemvn") { "gemvn_*" }
        else { return None };
    let need = match family {
        "attn_prefill_*" => attn_prefill_smem(name["attn_prefill_".len()..].parse::<usize>().ok()?),
        "attn_decode_gqa*" => {
            let hd = name["attn_decode_gqa".len()..].trim_start_matches('4');
            let hd = hd.strip_prefix('_')?.parse::<usize>().ok()?;
            // attn_decode_gqa4_128 copies one K/V chunk, the generic kernel two: the larger value covers both, and
            // both are below the 48 KiB default at head_dim 64 and 128.
            let cp = if hd == 128 || hd == 256 { 2 * 2 * 32 * hd * 2 } else { 0 };
            8 * hd * 4 + cp
        }
        // The only shape-dependent request: picking the device ceiling keeps every legal launch possible.
        _ => usize::MAX,
    };
    Some((family, need))
}

/// The dynamic shared memory to opt in to for a function with `stat` bytes of static shared memory, or `None` to
/// leave the attribute alone because the 48 KiB default already covers the launch. `need` is clamped to the device
/// opt-in minus the static part: `CU_FUNC_ATTRIBUTE_MAX_DYNAMIC_SHARED_SIZE_BYTES` cannot exceed what the
/// device allows for the block, and on Volta the opt-in ceiling is shared with the static usage (up to 2112 B in
/// `attn_decode_gqa_*`). A clamp below `need` makes the load succeed and a genuinely infeasible launch fail where
/// it is attempted, instead of the whole module failing to load.
fn dyn_shmem_optin(need: usize, stat: usize, optin: usize) -> Option<usize> {
    const DEFAULT: usize = 48 << 10;
    (need > DEFAULT).then(|| need.min(optin.saturating_sub(stat)))
}

/// `STRATA_SMEM_CARVEOUT`: the default `off` leaves the driver default, which the M2-T3 window measured as the
/// fastest (C1 was rejected); `policy` enables the R3 decision, `min` additionally picks the minimal fitting
/// carveout for the static-only GEMM families (C1's gemm half).
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum CarveoutMode { Off, Policy, Min }

fn carveout_mode_from(v: Option<&str>) -> CarveoutMode {
    match v.map(str::trim) {
        Some("policy") => CarveoutMode::Policy,
        Some("min") => CarveoutMode::Min,
        _ => CarveoutMode::Off,
    }
}

fn carveout_mode() -> CarveoutMode { carveout_mode_from(std::env::var("STRATA_SMEM_CARVEOUT").ok().as_deref()) }

/// The preferred shared-memory carveout percentage for a function, or `None` to leave the driver's default.
/// `dyn`/`stat` are one block's dynamic/static usage, `blocks_wanted` the number of blocks per SM the carveout is
/// meant to hold, `optin` the device's per-SM shared-memory maximum (96 KiB on the V100). Returns `None` when the
/// 48 KiB default already holds every wanted block (nothing to gain, e.g. the register-bound `gemm_*`) or when the
/// wanted blocks cannot fit at all (hd 256 attention: two blocks need 142 KiB). Otherwise the percentage is rounded
/// up so it never requests less than the launch needs, and clamped to `1..=100`.
fn carveout(dynamic: usize, stat: usize, blocks_wanted: usize, optin: usize) -> Option<i32> {
    const DEFAULT: usize = 48 << 10;
    if blocks_wanted == 0 || optin == 0 { return None; }
    let need = dynamic.checked_add(stat)?;
    if need == 0 { return None; }
    let total = need.checked_mul(blocks_wanted)?;
    if total <= DEFAULT { return None; }  // the default allocation already fits: request it nowhere
    (total <= optin).then(|| (total * 100).div_ceil(optin).clamp(1, 100) as i32)
}

/// The smallest carveout that still fits `need` bytes, used only by C1's `min` mode for the static-only GEMM
/// families: the driver's default allocation may reserve more shared memory than 27 KiB, and every KiB not carved
/// out is L1 for the weight stream.
fn carveout_min(need: usize, optin: usize) -> Option<i32> {
    (need > 0 && need <= optin).then(|| (need * 100).div_ceil(optin).clamp(1, 100) as i32)
}

/// The carveout to set on kernel `name`: `need` is its `dyn_shmem_need` (the launch's dynamic request, `None` for
/// kernels that use only static shared memory).
fn carveout_for(name: &str, need: Option<usize>, stat: usize, optin: usize, mode: CarveoutMode) -> Option<i32> {
    if mode == CarveoutMode::Off { return None; }
    let (dynamic, blocks) = match need {
        // hd 128 gains a second resident block at 96 KiB; hd 64 already fits two in the default; hd 256 cannot fit
        // two at all (`carveout` returns None), so the request leaves it unchanged.
        Some(v) if name.starts_with("attn_prefill") => (v, 2),
        Some(v) => (v, 1),
        None => (0, 1),
    };
    if let Some(p) = carveout(dynamic, stat, blocks, optin) { return Some(p); }
    let gemm = name.starts_with("gemm_") || name.starts_with("gemms_");
    (mode == CarveoutMode::Min && gemm).then(|| carveout_min(dynamic + stat, optin)).flatten()
}

/// One startup line per kernel family: the device's shared-memory opt-in (not a table), what the load set for the

fn smem_log(family: &str, optin: usize, stat: usize, set: Option<usize>, carveout: Option<i32>) -> String {

    format!("cuda shared memory: opt-in {optin} B per block, {family}: {stat} B static, dynamic max {dynamic}, carveout {carveout}")
}

/// `bs` holds the sums of the 16 groups of 16 `qs`, for the min terms of Q2_K and Q4_K.
pub struct Q8K { d: f32, qs: [i8; 256], bs: [i16; 16] }

/// Signs of IQ2/IQ3 sub-blocks as ±1 bytes, indexed by the 7-bit sign field.
static SIGNS: [[i8; 8]; 128] = {
    let mut t = [[1i8; 8]; 128];
    let mut i = 0;
    while i < 128 {
        let mut j = 0;
        while j < 8 { if KSIGNS_IQ2XS[i] & (1 << j) != 0 { t[i][j] = -1; } j += 1; }
        i += 1;
    }
    t
};

/// Quantizes `x` in blocks of 256; a short last block is zero-padded (Q2_0 rows can hold 64 * k values).
pub fn quantize_q8k(x: &[f32]) -> Vec<Q8K> {
    x.chunks(256).map(|b| {
        let mut pad = [0f32; 256];
        let b = if b.len() == 256 { b } else { pad[..b.len()].copy_from_slice(b); &pad[..] };
        // The first value of largest magnitude (a max over |v| first, which vectorizes).
        let amax = b.iter().fold(0f32, |m, &v| m.max(v.abs()));
        if amax == 0.0 { return Q8K { d: 0.0, qs: [0; 256], bs: [0; 16] }; }
        let max = *b.iter().find(|v| v.abs() == amax).unwrap();
        let is = -127.0 / max;
        let mut qs = [0i8; 256];
        #[cfg(target_feature = "avx2")]
        avx2::quant_block(is, b, &mut qs);
        #[cfg(not(target_feature = "avx2"))]
        for (q, &v) in qs.iter_mut().zip(b) { *q = ((is * v).round_ties_even() as i32).min(127) as i8; }
        let mut bs = [0i16; 16];
        for (s, c) in bs.iter_mut().zip(qs.chunks_exact(16)) { *s = c.iter().map(|&v| v as i16).sum(); }
        Q8K { d: 1.0 / is, qs, bs }
    }).collect()
}

                let q = b[4 + 8 * s + j];
                KVALUES_FP4[(q & 0xF) as usize] as i32 * q8[j] as i32 + KVALUES_FP4[(q >> 4) as usize] as i32 * q8[j + 8] as i32
            }).sum();
            crate::quant::ue4m3(b[s]) * y.d * v as f32
        }).sum::<f32>()
    }).sum()
}

/// Sum over the 16 groups of 16 of `sc(group) * (a · q)`, written so the compiler vectorizes it.
#[inline(always)]
fn grouped(a: &[i8; 256], q: &[i8; 256], sc: impl Fn(usize) -> i32) -> i32 {
    (0..16).map(|j| sc(j) * (16 * j..16 * j + 16).map(|i| a[i] as i32 * q[i] as i32).sum::<i32>()).sum()
}

fn dot_q2k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(84).zip(xq) {
        let sc = &b[0..16];
        let mut a = [0i8; 256];
        for i in 0..256 { a[i] = (b[16 + 32 * (i / 128) + i % 32] >> (2 * (i / 32 % 4)) & 3) as i8; }
        let s = grouped(&a, &y.qs, |j| (sc[j] & 0xF) as i32);
        let m: i32 = (0..16).map(|j| (sc[j] >> 4) as i32 * y.bs[j] as i32).sum();
        sumf += y.d * d16(&b[80..]) * s as f32 - y.d * d16(&b[82..]) * m as f32;
    }
    sumf
}

fn dot_q3k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(110).zip(xq) {
        let (hm, sc) = (&b[0..32], crate::quant::q3k_scales(b));
        let mut a = [0i8; 256];
        for i in 0..256 {
            let k = i / 32 % 4 + 4 * (i / 128);
            a[i] = (b[32 + 32 * (i / 128) + i % 32] >> (2 * (k % 4)) & 3) as i8 - if hm[i % 32] >> k & 1 != 0 { 0 } else { 4 };
        }
        sumf += d16(&b[108..]) * y.d * grouped(&a, &y.qs, |j| sc[j] as i32 - 32) as f32;
    }
    sumf
}

/// Q2_0: 64 values (q - 1) * d with q in 0..4, four to a byte; four blocks per Q8_K block.
fn dot_q2_0(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(18).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 4], i % 4 * 64);
        let s: i32 = (0..64).map(|j| (b[2 + j / 4] >> (2 * (j % 4)) & 3) as i32 * y.qs[o + j] as i32).sum();
        let m: i32 = y.bs[o / 16..o / 16 + 4].iter().map(|&v| v as i32).sum();
        d16(b) * y.d * (s - m) as f32
    }).sum()
}

/// TQ1_0: five base-3 trits per byte in qs, four in qh; values -1, 0, 1 times d.
fn dot_tq1_0(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(54).zip(xq) {
        let trit = |v: u8, n: usize| { const POW3: [u8; 5] = [1, 3, 9, 27, 81]; ((v.wrapping_mul(POW3[n]) as u16 * 3) >> 8) as i32 - 1 };
        let mut s = 0i32;
        for n in 0..5 { for m in 0..32 { s += trit(b[m], n) * y.qs[32 * n + m] as i32; } }
        for n in 0..5 { for m in 0..16 { s += trit(b[32 + m], n) * y.qs[160 + 16 * n + m] as i32; } }
        for n in 0..4 { for j in 0..4 { s += trit(b[48 + j], n) * y.qs[240 + 4 * n + j] as i32; } }
        sumf += d16(&b[52..]) * y.d * s as f32;
    }
    sumf
}

/// TQ2_0: 64 bytes of 2-bit codes (code - 1 in -1..2), then f16 d; 256 values a block.
fn dot_tq2_0(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(66).zip(xq) {
        let mut s = 0i32;
        for j in 0..2 { for l in 0..4 { for m in 0..32 {
            let code = (b[32 * j + m] >> (2 * l)) & 3;
            s += (code as i32 - 1) * y.qs[128 * j + 32 * l + m] as i32;
        } } }
        sumf += d16(&b[64..]) * y.d * s as f32;
    }
    sumf
}

/// Q4_0 (`T` = 2, 18 bytes), Q4_1 (3, 20), Q5_0 (6, 22), Q5_1 (7, 24): block bytes, offset of the 32 high bits
/// (0 = none) and of the 16 bytes of nibbles.
const fn q4_layout(t: u8) -> (usize, usize, usize) { match t { 2 => (18, 0, 2), 3 => (20, 0, 4), 6 => (22, 2, 6), _ => (24, 4, 8) } }
/// The block's m in value = d * q + m (q the unsigned 4- or 5-bit code): its own for Q4_1 / Q5_1, -8 d or -16 d for
/// Q4_0 / Q5_0.
fn q4_min(t: u8, b: &[u8]) -> f32 { match t { 3 | 7 => d16(&b[2..]), 2 => -8.0 * d16(b), _ => -16.0 * d16(b) } }

/// Q4_0, Q4_1, Q5_0, Q5_1 (`q4_layout`): 32 values d * q + m per block, eight blocks per Q8_K block.
fn dot_q4<const T: u8>(row: &[u8], xq: &[Q8K]) -> f32 {
    let (bb, oh, oq) = q4_layout(T);
    row.chunks_exact(bb).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 8], i % 8 * 32);
        let (qh, qs) = if oh > 0 { (u32::from_le_bytes(b[oh..oh + 4].try_into().unwrap()), &b[oq..oq + 16]) } else { (0, &b[oq..oq + 16]) };
        let s: i32 = (0..16).map(|j| {
            let (l, h) = ((qs[j] & 0xF) as i32 | ((qh >> j) & 1) as i32 * 16, (qs[j] >> 4) as i32 | ((qh >> (j + 16)) & 1) as i32 * 16);
            l * y.qs[o + j] as i32 + h * y.qs[o + j + 16] as i32
        }).sum();
        y.d * (d16(b) * s as f32 + q4_min(T, b) * (y.bs[o / 16] as i32 + y.bs[o / 16 + 1] as i32) as f32)
    }).sum()
}

/// Q8_0: 32 values d * q per block, eight blocks per Q8_K block.
fn dot_q8_0(row: &[u8], xq: &[Q8K]) -> f32 {
    row.chunks_exact(34).enumerate().map(|(i, b)| {
        let (y, o) = (&xq[i / 8], i % 8 * 32);
        let s: i32 = (0..32).map(|j| b[2 + j] as i8 as i32 * y.qs[o + j] as i32).sum();
        d16(b) * y.d * s as f32
    }).sum()
}

fn dot_q4k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(144).zip(xq) {
        let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
        let mut a = [0i8; 256];
        for i in 0..256 { a[i] = (b[16 + 32 * (i / 64) + i % 32] >> (4 * (i / 32 % 2)) & 0xF) as i8; }
        let s = grouped(&a, &y.qs, |j| sm[j / 2].0 as i32);
        let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
        sumf += d16(b) * y.d * s as f32 - d16(&b[2..]) * y.d * m as f32;
    }
    sumf
}

/// Q5_K: Q4_K plus a fifth bit per value from `qh` (bit 2 * (i / 64) + i / 32 % 2 of byte i % 32).
fn dot_q5k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(176).zip(xq) {
        let sm: [(u8, u8); 8] = std::array::from_fn(|j| crate::quant::scale_min_k4(j, &b[4..16]));
        let mut a = [0i8; 256];
        for j in 0..4 {
            for l in 0..32 {
                let (q, h) = (b[48 + 32 * j + l], b[16 + l] >> (2 * j));
                a[64 * j + l] = (q & 0xF | (h & 1) << 4) as i8;
                a[64 * j + 32 + l] = (q >> 4 | (h & 2) << 3) as i8;
            }
        }
        let s = grouped(&a, &y.qs, |j| sm[j / 2].0 as i32);
        let m: i32 = (0..8).map(|j| sm[j].1 as i32 * (y.bs[2 * j] + y.bs[2 * j + 1]) as i32).sum();
        sumf += d16(b) * y.d * s as f32 - d16(&b[2..]) * y.d * m as f32;
    }
    sumf
}

fn dot_q6k(row: &[u8], xq: &[Q8K]) -> f32 {
    let mut sumf = 0.0;
    for (b, y) in row.chunks_exact(210).zip(xq) {
        let mut a = [0i8; 256];
        for n in 0..2 {
            let (ql, qh, a) = (&b[64 * n..], &b[128 + 32 * n..], &mut a[128 * n..]);
            for l in 0..32 {
                a[l] = ((ql[l] & 0xF) | ((qh[l] & 3) << 4)) as i8 - 32;
                a[l + 32] = ((ql[l + 32] & 0xF) | (((qh[l] >> 2) & 3) << 4)) as i8 - 32;
                a[l + 64] = ((ql[l] >> 4) | (((qh[l] >> 4) & 3) << 4)) as i8 - 32;
                a[l + 96] = ((ql[l + 32] >> 4) | ((qh[l] >> 6) << 4)) as i8 - 32;
            }
        }
        let s: i32 = (0..16).map(|j| b[192 + j] as i8 as i32 * (16 * j..16 * j + 16).map(|i| a[i] as i32 * y.qs[i] as i32).sum::<i32>()).sum();
        sumf += d16(&b[208..]) * y.d * s as f32;
    }
    sumf
}

    fn alloc(be: &dyn Backend, c: &Inkling2Config, rank: usize, vocab_rows: usize, max_nt: usize) -> Result<Self> {
        let lay0 = PrefillLayout::at(0, c, rank, max_nt);
        let buf = be.alloc(lay0.end as usize)?;
        let p = PrefillLayout::at(buf.addr(), c, rank, max_nt);
        let acc_buf = be.alloc(4 * max_nt * c.n_embd)?;
        let kw = c.n_head_kv * c.head_dim;
        let rb = 2 * be.kv_words(kw);
        let kv_bytes = max_nt * rb;
        let (kc_tmp, vc_tmp) = (be.alloc(kv_bytes)?, be.alloc(kv_bytes)?);
        let max_np = max_nt * c.n_used;
        let meta = be.alloc((40 * max_np).max(1024))?;
        let logits_buf = be.alloc(4 * VERIFY_MAX * vocab_rows)?;
        let dspark_buf = be.alloc(4 * 4 * max_nt * c.n_embd)?;
        let n_slot = c.n_slot();
        let hist_bak = be.alloc(4 * n_slot * hists(c).1)?;
        let conv_log = be.alloc(4 * n_slot * VERIFY_MAX * (2 * kw + 2 * c.n_embd))?;
        let swa_log_bytes = (n_slot * VERIFY_MAX * rb).max(1024);
        let (swa_kc_log, swa_vc_log) = (be.alloc(swa_log_bytes)?, be.alloc(swa_log_bytes)?);
        Ok(Self {
            max_nt, buf, acc_buf, kc_tmp, vc_tmp, meta, logits_buf, dspark_buf,
            hist_bak, conv_log, swa_kc_log, swa_vc_log, p,
        })
    }
}

fn up<T: crate::backend::Plain>(be: &dyn Backend, v: &[T]) -> Result<Buf> {
    let b = be.alloc(size_of_val(v))?;
    be.put(v, &b, 0)?;
    Ok(b)
}

fn f(floats: usize) -> u64 { 4 * floats as u64 }

/// Rows of a K/V buffer that grows to hold `need`: a quarter more, so that a reply does not grow it at every token
/// and a long prompt does not take twice its rows from the experts.
fn grown(need: usize, max_ctx: usize) -> usize { (need + need / 4).next_multiple_of(1024).min(max_ctx) }

/// Free expert slots until `need` bytes of VRAM are free.
fn room(d: &mut dev::Dev, need: usize) {
    let free = d.parts[0].be.mem_info().map_or(0, |m| m.0);
    if free < need { d.release(0, need - free); }
}

/// Bytes of the attention partials for `rows` full-attention rows.
fn sc_bytes(c: &Inkling2Config, rows: usize) -> usize { 4 * c.n_head * rows.div_ceil(64) * (c.head_dim + 2) }

/// `moe_gemv` operands for matrices at `ptrs`: the lowest address, a step every address is a multiple of from it, and
/// each one's index in steps. None when an index does not fit an i32.
fn slab(ptrs: &[u64]) -> Option<(u64, usize, Vec<i32>)> {
    let base = *ptrs.iter().min()?;
    let gcd = |mut a: u64, mut b: u64| { while b != 0 { (a, b) = (b, a % b); } a };
    let step = ptrs.iter().fold(0, |g, &p| gcd(g, p - base)).max(1);
    Some((base, step as usize, ptrs.iter().map(|&p| i32::try_from((p - base) / step).ok()).collect::<Option<_>>()?))
}

/// The kernels index the scratch buffer and the small tensors with the config's sizes and do not check bounds: a file
/// whose tensors have other lengths runs on the host, where a wrong length panics instead of reading or writing past a
/// buffer on the card.
fn shapes_fit(m: &Inkling2Model, rank: usize) -> bool {
    let c = &m.cfg;
    let (n, q, kw, nh, nc, nf) = (c.n_embd, c.n_head * c.head_dim, c.n_head_kv * c.head_dim, c.n_head, c.n_conv, c.n_ff_exp);
    let mat = |x: &Mat, rows: usize, cols: usize| x.rows == rows && x.cols == cols;
    // Residual blocks held at once: those of the prelude, one pass of the core and the coda (the `blocks` region).
    let starts = |r: std::ops::Range<usize>| r.filter(|i| i % c.block_size == 0).count();
    let blocks = starts(0..c.loop_start) + starts(c.loop_start..c.loop_end) + starts(c.loop_end..c.n_layer);
    blocks <= 4 && rank > 0 && mat(&m.output, m.output.rows, n) && m.out_query.len() == n && m.out_norm.len() == n
        && m.loops.iter().all(|l| {
            l.msg_query.len() == 4 * n && [(&l.down, 2 * n), (&l.gain, n), (&l.forget, n)].iter().all(|(w, cols)| w.len() == 4 && w.iter().all(|x| x.len() == rank * cols))
        })
        && m.layers.iter().all(|l| {
            let (gu, dn) = (l.gate_up.expert_host(0, c.n_expert), l.down.expert_host(0, c.n_expert));
            [(&l.wq, q), (&l.gate, q), (&l.wk, kw), (&l.wv, kw), (&l.rel, nh * c.n_rel), (&l.sink, nh)].iter().all(|(x, r)| mat(x, *r, n))
                && mat(&l.wo, n, q) && mat(&gu, 2 * nf, n) && mat(&dn, n, nf)
                && l.conv_proj.as_ref().is_none_or(|p| p.iter().all(|x| mat(x, c.n_head_kv * nc, n)))
                && l.res_query.as_ref().is_none_or(|r| !r.is_empty() && r.len() % n == 0 && r.len() <= 3 * n)
                && (1..=3).contains(&l.norms.len()) && l.norms.iter().all(|x| x.len() == n)
                && l.q_clip.len() == q && l.rel_clip.len() == nh * c.n_rel && l.rel_bias.len() == c.n_rel * c.rel_extent
                && [kw, kw, n, n].iter().zip(&l.convs).all(|(ch, w)| w.len() == ch * nc)
                && l.ffn_query.len() == n && l.ffn_norm.len() == n
        })
}

/// The layer whose weights slot `il` runs.
fn layer_of(c: &Inkling2Config, il: usize) -> usize {
    if il < c.n_layer { il } else { c.loop_start + (il - c.n_layer) % (c.loop_end - c.loop_start) }
}

/// Floats in the histories of a slot's K, V, attention-output and FFN-output convolutions, and where each starts.
fn hists(c: &Inkling2Config) -> ([usize; 4], usize) {
    let (kw, n, m) = (c.n_head_kv * c.head_dim, c.n_embd, c.n_conv - 1);
    ([0, m * kw, 2 * m * kw, 2 * m * kw + m * n], 2 * m * (kw + n))
}

/// The runs of the positions `from..to` that are contiguous rows of a slot: `(first position, first row, rows)`.
fn runs(from: usize, to: usize, ring: usize) -> impl Iterator<Item = (usize, usize, usize)> {
    let mut a = from.max(to.saturating_sub(ring));
    std::iter::from_fn(move || (a < to).then(|| {
        let b = to.min((a / ring + 1) * ring);
        let r = (a, a % ring, b - a);
        a = b;
        r
    }))
}

impl Device {
    /// None when a token cannot run on the device: no GPU or more than one, a dense matrix that is not in VRAM, a
    /// shape the kernels do not take, a tensor trace, no room, or `STRATA_DS_DEVICE=0`.
    pub(super) fn new(m: &Inkling2Model) -> Option<Self> {
        let (c, pos) = (&m.cfg, m.toks.len());
        let off = std::env::var("STRATA_DS_DEVICE").is_ok_and(|v| v == "0") || m.trace || std::env::var_os("STRATA_DS_DUMP").is_some();
        if off || c.n_conv > 8 || 128 % c.head_dim != 0 || m.loops.is_empty() { return None; }
        let rank = m.loops[0].down[0].len() / (2 * c.n_embd);
        if !shapes_fit(m, rank) { return None; }
        let (kw, nr, ext) = (c.n_head_kv * c.head_dim, c.n_rel, c.rel_extent);
        let full = (2 * (pos + 1)).clamp(4096.min(m.max_ctx), m.max_ctx);
        {
            let mut d = dev::get()?.lock().unwrap();
            if d.parts.len() != 1 { return None; }
            let dense = |x: &Mat| !x.host_only();
            if !dense(&m.output) || !m.layers.iter().all(|l| {
                [l.wq, l.wk, l.wv, l.wo, l.gate, l.rel, l.sink].iter().chain(l.conv_proj.iter().flatten()).all(dense)
            }) { return None; }
            let rows: usize = (0..c.n_slot()).map(|il| if c.swa[layer_of(c, il)] { c.n_swa } else { full }).sum();
            let row_words = d.parts[0].be.kv_words(kw);
            room(&mut d, 4 * rows * row_words + sc_bytes(c, full) + Scratch::at(0, c, rank, m.output.rows).end as usize + (512 << 20));
        }
        let made = (|| -> Result<Self> {
            let (mut layers, loops, slots, out_query, out_norm, scr, sc_buf) = {
                let d = dev::get().unwrap().lock().unwrap();
                let be = &*d.parts[0].be;
                let layers = m.layers.iter().map(|l| Ok(DLayer {
                    res_query: l.res_query.as_ref().map(|q| up(be, q)).transpose()?,
                    norms: up(be, &l.norms.concat())?,
                    q_clip: up(be, &l.q_clip)?,
                    // The file's order, `[distance][channel]`.
                    rel_bias: up(be, &(0..nr * ext).map(|i| l.rel_bias[i % nr * ext + ext - 1 - i / nr]).collect::<Vec<_>>())?,
                    rel_clip: up(be, &l.rel_clip)?,
                    convs: [up(be, &l.convs[0])?, up(be, &l.convs[1])?, up(be, &l.convs[2])?, up(be, &l.convs[3])?],
                    ffn_query: up(be, &l.ffn_query)?,
                    ffn_norm: up(be, &l.ffn_norm)?,
                    full_exps: None,
                })).collect::<Result<Vec<_>>>()?;
                let half = |w: &[Vec<f16>]| up(be, w.concat().reinterpret_cast());
                let loops = m.loops.iter().map(|l| {
                    Ok(DLoop { msg_query: up(be, &l.msg_query)?, down: half(&l.down)?, gain: half(&l.gain)?, forget: half(&l.forget)? })
                }).collect::<Result<Vec<_>>>()?;
                let row_words = be.kv_words(kw);
                let slots = (0..c.n_slot()).map(|il| {
                    let rows = if c.swa[layer_of(c, il)] { c.n_swa } else { full };
                    Ok(DSlot { k: be.alloc(2 * rows * row_words)?, v: be.alloc(2 * rows * row_words)?, hist: be.zeros::<f32>(hists(c).1)?, cap: rows })
                }).collect::<Result<Vec<_>>>()?;
                let sc_buf = be.alloc(sc_bytes(c, (pos + 1).max(64)))?;
                let (out_query, out_norm, scr) = (up(be, &m.out_query)?, up(be, &m.out_norm)?, be.alloc(Scratch::at(0, c, rank, m.output.rows).end as usize)?);
                (layers, loops, slots, out_query, out_norm, scr, sc_buf)
            };
            let s = Scratch::at(scr.addr(), c, rank, m.output.rows);

            // Preload full-layer expert slabs so selected layers skip CPU routing, uploads, and host syncs.
            // Core loop layers run 3x per token (75% of FFN compute), so prioritize them first.
            let cpu_share = std::env::var("STRATA_DS_CPU_SHARE").ok().and_then(|v| v.parse::<f32>().ok()).unwrap_or(0.0);
            let (mut n_full, mut slab_bytes) = (0usize, 0usize);
            if cpu_share <= 0.0 && c.n_expert <= 512 && c.n_used <= 32 {
                let (b_res, f_res) = (64 << 20, 384 << 20);
                let order = (c.loop_start..c.loop_end).chain(0..c.loop_start).chain(c.loop_end..c.n_layer);
                let key = |s: &[u8]| Some((s.as_ptr() as usize, s.len()));
                for ip in order {
                    let ml = &m.layers[ip];
                    let gi_bytes: Vec<u8> = ml.gate_inp.iter().flat_map(|x| x.to_le_bytes()).collect();
                    let bias_bytes: Vec<u8> = ml.bias.iter().flat_map(|x| x.to_le_bytes()).collect();
                    let gs_bytes: Vec<u8> = ml.gu_scale.iter().flat_map(|x| x.to_le_bytes()).collect();
                    let ds_bytes: Vec<u8> = ml.down_scale.iter().flat_map(|x| x.to_le_bytes()).collect();
                    let Some((gu, dn, gi_slab, bias, gu_scale, dn_scale)) = (|| -> Option<_> {
                        let mut d = dev::get().unwrap().lock().unwrap();
                        let gu = d.alloc_slab(ml.gate_up.data, key(ml.gate_up.data), b_res, f_res)?;
                        let dn = d.alloc_slab(ml.down.data, key(ml.down.data), b_res, f_res)?;
                        let gi = d.alloc_slab(&gi_bytes, None, b_res, f_res)?;
                        let bi = d.alloc_slab(&bias_bytes, None, b_res, f_res)?;
                        let gs = if gs_bytes.is_empty() { None } else { Some(d.alloc_slab(&gs_bytes, None, b_res, f_res)?) };
                        let ds = if ds_bytes.is_empty() { None } else { Some(d.alloc_slab(&ds_bytes, None, b_res, f_res)?) };
                        Some((gu, dn, gi, bi, gs, ds))
                    })() else {
                        break;
                    };
                    slab_bytes += gu.len() + dn.len() + gi_slab.len() + bias.len()
                        + gu_scale.as_ref().map_or(0, |b| b.len())
                        + dn_scale.as_ref().map_or(0, |b| b.len());
                    n_full += 1;
                    let (gu_rows, gu_cols) = (ml.gate_up.rows / c.n_expert, ml.gate_up.cols);
                    let (dn_rows, dn_cols) = (ml.down.rows / c.n_expert, ml.down.cols);
                    layers[ip].full_exps = Some(FullExps {
                        gate_inp: GpuTensor::new(GgmlType::F32, c.n_expert, c.n_embd, Buf::new(gi_slab.addr(), gi_slab.len(), ())),
                        _gi: gi_slab,
                        bias,
                        gu, gu_ty: ml.gate_up.ty, gu_rows, gu_cols,
                        gu_stride: gu_rows * ml.gate_up.ty.row_bytes(gu_cols as u64),
                        dn, dn_ty: ml.down.ty, dn_rows, dn_cols,
                        dn_stride: dn_rows * ml.down.ty.row_bytes(dn_cols as u64),
                        gu_scale,
                        dn_scale,
                    });
                }
            }
            if std::env::var_os("STRATA_DS_PROF").is_some() {
                eprintln!("inkling2: preloaded {n_full}/{} expert layers ({:.2} GB) as contiguous slabs", c.n_layer, slab_bytes as f64 / 1e9);
            }
            let all_full = layers.iter().all(|l| l.full_exps.is_some());
            Ok(Device { layers, loops, slots, out_query, out_norm, scr, sc_buf, pf: None, s, rank, all_full, graph_keys: [0; 2], synced: 0, ahead: false })
        })();
        made.inspect_err(|e| eprintln!("inkling2: no room in VRAM for the decode state ({e:#}); the tokens run as prompt chunks do")).ok()
    }

    /// The model starts over: nothing is shared with the host's rows.
    pub(super) fn forget(&mut self) {
        (self.synced, self.ahead) = (0, false);
    }

    /// Save conv histories and SWA ring buffers for fast prefix rollback (full-attention KV rows `< pos` are untouched by later tokens).
    pub(super) fn save_checkpoint(&self, c: &Inkling2Config) -> (Vec<Vec<f32>>, Vec<Option<(Vec<u16>, Vec<u16>)>>, usize, bool) {
        if !self.ahead { return (vec![], vec![], self.synced, false); }
        let d = dev::get().unwrap().lock().unwrap();
        let be = &*d.parts[0].be;
        let len = hists(c).1;
        let kw = c.n_head_kv * c.head_dim;
        let swa_words = c.n_swa * be.kv_words(kw);
        let mut dev_hist = Vec::with_capacity(self.slots.len());
        let mut dev_swa = Vec::with_capacity(self.slots.len());
        for (il, ds) in self.slots.iter().enumerate() {
            dev_hist.push(be.get(&ds.hist, 0, len).expect("inkling2 save_checkpoint hist"));
            if c.swa[layer_of(c, il)] {
                let k: Vec<u16> = be.get(&ds.k, 0, swa_words).expect("inkling2 save_checkpoint swa k");
                let v: Vec<u16> = be.get(&ds.v, 0, swa_words).expect("inkling2 save_checkpoint swa v");
                dev_swa.push(Some((k, v)));
            } else {
                dev_swa.push(None);
            }
        }
        (dev_hist, dev_swa, self.synced, true)
    }

    /// Restore conv histories and SWA ring buffers from a `Checkpoint`.
    pub(super) fn restore_checkpoint(
        &mut self,
        hist: &[Vec<f32>],
        swa: &[Option<(Vec<u16>, Vec<u16>)>],
        synced: usize,
        ahead: bool,
    ) {
        (self.synced, self.ahead) = (synced, ahead);
        if !ahead || hist.is_empty() { return; }
        let d = dev::get().unwrap().lock().unwrap();
        let be = &*d.parts[0].be;
        for ((ds, h), kv) in self.slots.iter().zip(hist).zip(swa) {
            be.put(h, &ds.hist, 0).expect("inkling2 restore_checkpoint hist");
            if let Some((k, v)) = kv {
                be.put(k, &ds.k, 0).expect("inkling2 restore_checkpoint swa k");
                be.put(v, &ds.v, 0).expect("inkling2 restore_checkpoint swa v");
            }
        }
    }

    /// Room for the rows of position `need_pos` in the slots that grow and `sc_buf`.
    pub(super) fn prepare(&mut self, m: &Inkling2Model, need_pos: usize) -> Result<()> {
        let (c, kw) = (&m.cfg, m.cfg.n_head_kv * m.cfg.head_dim);
        let need = need_pos + 1;
        ensure!(need <= m.max_ctx, "context is limited to {} tokens", m.max_ctx);
        let mut d = dev::get().unwrap().lock().unwrap();
        let row_words = d.parts[0].be.kv_words(kw);
        let full_cap = self.slots.iter().enumerate().find(|&(il, _)| !c.swa[layer_of(c, il)]).map(|(_, s)| s.cap);
        if let Some(cap) = full_cap && need > cap {
            let new_cap = grown(need, m.max_ctx);
            let n = (0..c.n_slot()).filter(|&il| !c.swa[layer_of(c, il)]).count();
            // The margin is the one of `new`: the prompt scratch and the graphs of the reply come after this.
            let bytes = 4 * n * new_cap * row_words + sc_bytes(c, new_cap) + (512 << 20);
            room(&mut d, bytes);
            // The expert cache was not enough: whole layers of experts go, the last loaded first. Their experts
            // then run through the expert cache or on the CPU, which costs less than a model without its device state.
            let free = |d: &dev::Dev| d.parts[0].be.mem_info().map_or(0, |m| m.0);
            if free(&d) < bytes {
                d.parts[0].be.sync()?;
                let order: Vec<usize> = (c.loop_start..c.loop_end).chain(0..c.loop_start).chain(c.loop_end..c.n_layer).collect();
                // Counted here: the driver may report the freed bytes late.
                let mut short = bytes - free(&d);
                for &ip in order.iter().rev() {
                    if short == 0 { break; }
                    if let Some(fe) = self.layers[ip].full_exps.take() {
                        // The budget falls too, so the expert cache does not take the room back.
                        d.parts[0].budget = d.parts[0].budget.saturating_sub(fe.gu.len() + fe.dn.len());
                        short = short.saturating_sub(fe.gu.len() + fe.dn.len());
                        self.all_full = false;
                    }
                }
            }
        }
        let be = &*d.parts[0].be;
        // The score buffer follows the K/V rows: it is in the room made above, and a reply does not grow it every 64 tokens.
        let need_sc = sc_bytes(c, full_cap.map_or(need, |cap| if need > cap { grown(need, m.max_ctx) } else { cap }));
        if self.sc_buf.len() < need_sc {
            be.sync()?;
            self.sc_buf = be.alloc(need_sc)?;
        }
        for (il, s) in self.slots.iter_mut().enumerate() {
            if c.swa[layer_of(c, il)] || need <= s.cap { continue; }
            let new_cap = grown(need, m.max_ctx);
            for b in [&mut s.k, &mut s.v] {
                let new = be.alloc(2 * new_cap * row_words)?;
                be.dtod(b, 0, &new, 0, b.len())?;
                be.sync()?;  // the copy reads the buffer that `*b = new` frees
                *b = new;
            }
            s.cap = new_cap;
        }
        Ok(())
    }

    Recurrent { conv: Vec<f32>, s: Vec<f32> },
    /// k, v: pos x n_head_kv x head_dim; ik: pos x idx_head_dim raw QSA indexer keys (empty without an indexer).
    Attn { k: Vec<f32>, v: Vec<f32>, ik: Vec<f32> },
}

pub struct State {
    pub layers: Vec<LayerState>,
    pub pos: usize,
    /// M-RoPE positions for the next token: `[t, y, x, z]`. Text tokens use `pos` in all four slots.
    pub rope_pos: [i32; 4],
    /// qwen4exp PLE: the last PLE_HIST normalized rows (hc * n_embd each, oldest first), and the two previous
    /// tokens (newest first, -1 for none) for the n-gram hash.
    pub ple_hist: Vec<f32>,
    pub ple_prev: [i64; 2],
}

impl State {
    pub fn new(c: &ModelConfig) -> State {
        let channels = c.ssm_d_inner + 2 * c.ssm_n_group * c.ssm_d_state;
        let layers = c.layers.iter()
            .map(|k| match k {
                LayerKind::Delta => LayerState::Recurrent {
                    conv: vec![0.0; (c.ssm_conv - 1) * channels],
                    s: vec![0.0; c.ssm_dt_rank * c.ssm_dv() * c.ssm_d_state],
                },
                LayerKind::Attn => LayerState::Attn { k: Vec::new(), v: Vec::new(), ik: Vec::new() },
            })
            .collect();
        let ple = if c.n_embd_ple > 0 { PLE_HIST * c.hc_n * c.n_embd } else { 0 };
        State { layers, pos: 0, rope_pos: [0; 4], ple_hist: vec![0.0; ple], ple_prev: [-1, -1] }
    }
}

/// MTP draft state: the KV cache of the nextn block.
pub struct Mtp { pub kv: LayerState, pub pos: usize }

impl Default for Mtp {
    fn default() -> Mtp { Mtp { kv: LayerState::Attn { k: Vec::new(), v: Vec::new(), ik: Vec::new() }, pos: 0 } }
}

fn mv(t: &Tensor, x: &[f32]) -> Vec<f32> {
    let mut y = vec![0f32; t.rows];
    t.matvec(x, &mut y);
    y
}

impl<'a> Model<'a> {
    pub fn load(g: &'a Gguf) -> anyhow::Result<Model<'a>> { Self::load_with_lora(g, &[]) }
    pub fn load_with_lora(g: &'a Gguf, loras: &[(String, f32)]) -> anyhow::Result<Model<'a>> {
        Self::load_with_aux(g, None, loras)
    }
    pub fn load_with_aux(g: &'a Gguf, aux: Option<&'a Gguf>, loras: &[(String, f32)]) -> anyhow::Result<Model<'a>> {
        let mut cfg = ModelConfig::from_gguf(g)?;
        // A DFlash draft given on the command line replaces the target's own MTP head; without this the draft was
        // ignored without a word for a target GGUF that carries one.
        if cfg.n_nextn > 0 && aux.is_some_and(|ag| ag.tensors.contains_key("dflash.fc.weight")) {
            crate::say!("note: DFlash draft model given: the target's own MTP head is not used");
            cfg.n_nextn = 0;
        }
        let is_standalone_draft = aux.is_some_and(|ag| {
            !ag.tensors.contains_key("fc.weight")
                && !ag.tensors.contains_key("dflash.fc.weight")
                && !ag.tensors.contains_key(&format!("blk.{}.nextn.eh_proj.weight", cfg.n_layer))
                && ag.tensors.contains_key("token_embd.weight")
                && ModelConfig::from_gguf(ag).is_ok()
        });
        let w = Weights::new_with_aux(g, if is_standalone_draft { None } else { aux })?;
        let mut draft_model = None;
        if is_standalone_draft {
            let ag = aux.unwrap();
            let dm = Model::load(ag)?;
            anyhow::ensure!(
                dm.cfg.n_vocab == cfg.n_vocab,
                "standalone draft model vocab ({}) != target vocab ({})",
                dm.cfg.n_vocab,
                cfg.n_vocab
            );
            crate::say!(
                "note: loaded standalone external draft model (arch={}, layers={}, embd={}, vocab={})",
                dm.cfg.arch, dm.cfg.n_layer, dm.cfg.n_embd, dm.cfg.n_vocab
            );
            draft_model = Some(Box::new(dm));
        } else if cfg.n_nextn == 0 {
            if let Some(ag) = aux.filter(|_| w.layer(cfg.n_layer, "eagle3.fc.weight").is_ok()) {
                let il = cfg.n_layer;
                let wq = w.layer(il, "attn_q.weight")?;
                let wk = w.layer(il, "attn_k.weight")?;
                let gate = w.layer(il, "ffn_gate.weight")?;
                let out = w.layer(il, "eagle3.output.weight")?;
                let n_head = ag.meta.get("eagle3.attention.head_count").and_then(|v| v.as_u64()).map(|v| v as usize).unwrap_or(cfg.n_head);
                let head_dim = wq.rows / n_head.max(1);
                let n_head_kv = (wk.rows / head_dim.max(1)).max(1);
                let n_ff = gate.rows;
                let draft_vocab = out.rows;
                let n_rot = ag.meta.get("eagle3.rope.dimension_count").and_then(|v| v.as_u64()).map(|v| v as usize).unwrap_or(head_dim);
                let rope_base = ag.meta.get("eagle3.rope.freq_base").and_then(|v| v.as_f32()).unwrap_or(cfg.rope_base);
                let eps = ag.meta.get("eagle3.attention.layer_norm_rms_epsilon").and_then(|v| v.as_f32()).unwrap_or(cfg.eps);
                let norm_before_residual = matches!(ag.meta.get("eagle3.norm_before_residual"), Some(crate::gguf::Value::Bool(true)));
                let rope_neox = !matches!(ag.meta.get("eagle3.rope.neox"), Some(crate::gguf::Value::Bool(false)));
                let extract_layers = match ag.meta.get("eagle3.target_layers").or_else(|| ag.meta.get("eagle3.extract_layers")).and_then(|v| v.as_arr()) {
                    Some(arr) if arr.len() == 3 => {
                        let mut ls = [0usize; 3];
                        for (i, v) in arr.iter().enumerate() {
                            ls[i] = v.as_u64().context("bad eagle3 extract layer")? as usize;
                        }
                        ls
                    }
                    _ => [2, cfg.n_layer / 2, cfg.n_layer.saturating_sub(3)],
                };
                let d2t = if let Some(t) = ag.tensors.get("d2t") {
                    let n = t.dims[0] as usize;
                    anyhow::ensure!(n == draft_vocab, "d2t len {n} != draft_vocab {draft_vocab}");
                    let mut ids: Vec<i32> = match t.ty {
                        26 => ag.raw_bytes(t, 4 * n)?.chunks_exact(4).map(|b| i32::from_le_bytes(b.try_into().unwrap())).collect(),
                        27 => ag.raw_bytes(t, 8 * n)?.chunks_exact(8).map(|b| i64::from_le_bytes(b.try_into().unwrap()) as i32).collect(),
                        ty => anyhow::bail!("unsupported EAGLE3 d2t tensor type {ty}"),
                    };
                    if ids.windows(2).any(|w| w[0] >= w[1]) {
                        for (i, v) in ids.iter_mut().enumerate() { *v += i as i32; }
                    }
                    ids
                } else {
                    (0..draft_vocab as i32).collect()
                };
                cfg.n_nextn = 1;
                cfg.eagle3 = Some(Eagle3Config {
                    extract_layers, n_ff, n_head, n_head_kv, head_dim, n_rot, rope_base, eps,
                    draft_vocab, norm_before_residual, rope_neox, d2t,
                });
                crate::say!(
                    "note: loaded standalone EAGLE3 draft head (extract_layers={extract_layers:?}, heads={n_head}x{head_dim}, kv={n_head_kv}, ff={n_ff}, draft_vocab={draft_vocab})"
                );
            } else if aux.is_some_and(|ag| ag.tensors.contains_key("dflash.fc.weight")) {
                let ag = aux.unwrap();
                let u = |k: &str| ag.meta.get(k).and_then(|v| v.as_u64()).map(|v| v as usize);
                let f = |k: &str| ag.meta.get(k).and_then(|v| v.as_f32());
                let n_layer = u("dflash.block_count").context("dflash.block_count")?;
                let block_size = u("dflash.block_size").unwrap_or(16).max(2);
                let mask_token_id = u("dflash.mask_token_id").context("dflash.mask_token_id")? as u32;
                let n_embd = u("dflash.embedding_length").unwrap_or(cfg.n_embd);
                let n_ff = u("dflash.feed_forward_length").context("dflash.feed_forward_length")?;
                let n_head = u("dflash.attention.head_count").unwrap_or(cfg.n_head);
                let n_head_kv = u("dflash.attention.head_count_kv").unwrap_or(cfg.n_head_kv);
                let head_dim = u("dflash.attention.key_length").unwrap_or(n_embd / n_head.max(1));
                let n_rot = u("dflash.rope.dimension_count").unwrap_or(head_dim);
                let rope_base = f("dflash.rope.freq_base").unwrap_or(cfg.rope_base);
                let eps = f("dflash.attention.layer_norm_rms_epsilon").unwrap_or(cfg.eps);
                let sliding_window = u("dflash.attention.sliding_window").unwrap_or(0);
                let input_embedding_scale = f("dflash.input_embedding_scale").unwrap_or(1.0);
                let output_multiplier = f("dflash.output_multiplier").unwrap_or(1.0);
                let conv_kernel_size = u("dflash.conv_kernel_size").unwrap_or(0);
                let conv_group_size = u("dflash.conv_group_size").unwrap_or(0);
                let selector_rank = u("dflash.selector_rank").unwrap_or(0);
                let selector_top_k = u("dflash.selector_top_k").unwrap_or(0);
                let target_layer_ids = ag.meta.get("dflash.target_layers").and_then(|v| v.as_arr())
                    .context("dflash.target_layers")?
                    .iter().map(|v| v.as_u64().map(|x| x as usize).context("bad dflash target layer")).collect::<anyhow::Result<Vec<_>>>()?;
                anyhow::ensure!(target_layer_ids.iter().all(|&id| id < cfg.n_layer), "dflash target_layer_ids out of range for {} layers: {target_layer_ids:?}", cfg.n_layer);
                let layer_swa = ag.meta.get("dflash.attention.sliding_window_pattern").and_then(|v| v.as_arr())
                    .filter(|a| a.len() >= n_layer)
                    .map(|a| a[..n_layer].iter().map(|v| v.as_u64().unwrap_or(0) != 0).collect())
                    .unwrap_or_else(|| vec![false; n_layer]);
                cfg.dflash = Some(DFlashConfig {
                    target_layer_ids: target_layer_ids.clone(),
                    block_size,
                    mask_token_id,
                    n_layer,
                    n_embd,
                    n_ff,
                    n_head,
                    n_head_kv,
                    head_dim,
                    n_rot,
                    rope_base,
                    eps,
                    sliding_window,
                    layer_swa,
                    input_embedding_scale,
                    output_multiplier,
                    conv_kernel_size,
                    conv_group_size,
                    selector_rank,
                    selector_top_k,
                });
                crate::say!(
                    "note: loaded standalone DFlash{} draft model (layers={n_layer}, block={block_size}, mask={mask_token_id}, target_layers={target_layer_ids:?}, heads={n_head}x{head_dim}, kv={n_head_kv}, ff={n_ff})",
                    if selector_rank > 0 { "2" } else { "" }
                );
            } else if aux.is_some_and(|ag| ag.tensors.contains_key(&format!("blk.{}.nextn.eh_proj.weight", cfg.n_layer))) {
                cfg.n_nextn = 1;
                crate::say!("note: loaded standalone MTP draft head from companion GGUF (layer {})", cfg.n_layer);
            }
        }
        if !w.validate(cfg.n_layer)? && cfg.n_nextn > 0 {
            crate::say!("note: the MTP (nextn) layer has a quant type this engine does not support; loading without MTP");
            cfg.n_nextn = 0;
        }
        let emb_t = w.get("token_embd.weight")?;
        // Pre-touch mmap pages of token_embd.weight so decode step put_embed() never stalls on SSD page faults.
        let mut _touch = 0u8;
        for off in (0..emb_t.data.len()).step_by(4096) {
            _touch ^= unsafe { std::ptr::read_volatile(&emb_t.data[off]) };
        }
        std::hint::black_box(_touch);
        let ple = if cfg.n_embd_ple > 0 {
            let p = |k: &str| format!("{}.ple.{k}", cfg.arch);
            let arr = |k: &str| -> anyhow::Result<Vec<u64>> {
                g.get(&p(k))?.as_arr().with_context(|| p(k))?.iter().map(|v| v.as_u64().with_context(|| p(k))).collect()
            };
            let int = |k: &str| -> anyhow::Result<u64> { g.get(&p(k))?.as_u64().with_context(|| p(k)) };
            let heads_per_ngram = int("heads_per_ngram")? as usize;
            anyhow::ensure!(int("ngram_size")? == PLE_NGRAM as u64 && heads_per_ngram > 0
                && int("conv_kernel")? == PLE_KERNEL as u64, "unsupported PLE geometry");
            let ple = Ple { mult: arr("layer_multipliers")?, off: arr("head_offsets")?, vocab: arr("head_vocab_sizes")?, eos: int("eos_token_id")? as i64, heads_per_ngram };
            let heads = (PLE_NGRAM - 1) * heads_per_ngram;
            anyhow::ensure!(ple.mult.len() == PLE_NGRAM && ple.off.len() == heads && ple.vocab.len() == heads
                && ple.vocab.iter().all(|&v| v > 0) && heads * cfg.n_embd_ple == cfg.n_embd, "bad PLE metadata");
            w.get("per_layer_token_embd.weight")?;
            Some(ple)
        } else {
            w.get("output_norm.weight")?;
            None
        };
        let mut m = Model { cfg, w, loras: Vec::new(), draft_model, ple };
        m.apply_lora(loras)?;
        Ok(m)
    }

    pub fn apply_lora(&mut self, specs: &[(String, f32)]) -> anyhow::Result<()> {
        // Load everything before touching the current overlays, so a bad adapter leaves the model unchanged.
        let loras: Vec<Lora> = specs.iter().map(|(path, scale)| load_lora(path, *scale)).collect::<anyhow::Result<_>>()?;
        // Routed-expert adapters are not baked (`bake_lora` checks the others): they index activations and the
        // residual row by their own dimensions at inference, so these must be the base tensor's.
        let ne = self.cfg.n_expert.max(1);
        for (name, e) in loras.iter().flat_map(|l| &l.entries).filter(|(n, _)| n.ends_with("ffn_down_exps.weight")) {
            let Ok(base) = self.w.get(name) else { continue };
            anyhow::ensure!(e.in_dim == base.cols && e.out_dim * ne == base.rows && (e.n_expert == 1 || e.n_expert == ne),
                "lora: target {name} is {}x{} with {ne} experts, adapter is {}x{} with {}", base.rows / ne, base.cols, e.out_dim, e.in_dim, e.n_expert);
        }
        self.w.clear_baked();
        self.loras = loras;
        self.w.bake_lora(&self.loras);
        Ok(())
    }

    pub fn lora_delta(&self, name: &str, x: &[f32], y: &mut [f32], expert: Option<usize>) {
        for lora in &self.loras {
            if let Some(e) = lora.entries.get(name) {
                let ei = if e.n_expert == 1 { 0 } else { expert.unwrap_or(0) };
                if ei >= e.n_expert && e.n_expert > 1 { continue; }
                let a = &e.a[ei];
                let b = &e.b[ei];
                let rank = e.rank;
                let in_dim = e.in_dim;
                let out_dim = e.out_dim;
                for r in 0..rank {
                    let mut s = 0f32;
                    for i in 0..in_dim {
                        if i < x.len() { s += a[r * in_dim + i] * x[i]; }
                    }
                    let coef = if lora.alpha > 0.0 { lora.scale * lora.alpha / rank as f32 * s } else { lora.scale * s };
                    for o in 0..out_dim {
                        if o < y.len() { y[o] += coef * b[o * rank + r]; }
                    }
                }
            }
        }
    }

    fn t(&self, il: usize, suffix: &str) -> Tensor<'a> {
        self.w.layer(il, suffix).unwrap_or_else(|e| panic!("{e:#}"))
    }
    fn f(&self, il: usize, suffix: &str) -> Vec<f32> { self.t(il, suffix).to_f32() }

    pub fn embed(&self, token: u32) -> Vec<f32> {
        let mut x = vec![0f32; self.cfg.n_embd];
        let idx = (token as usize).min(self.cfg.n_vocab.saturating_sub(1));
        self.w.get("token_embd.weight").unwrap().row(idx, &mut x);
        x
    }

    /// Run one token; returns logits over the vocabulary.
    pub fn step(&self, st: &mut State, token: u32) -> Vec<f32> { self.step_h(st, token).0 }

    /// Run one token; returns logits and the last hidden state (before output_norm).
    pub fn step_h(&self, st: &mut State, token: u32) -> (Vec<f32>, Vec<f32>) {
        let rope_pos = st.rope_pos;
        self.step_embed_h(st, self.embed(token), rope_pos, token)
    }

    /// Feed a precomputed input embedding at explicit M-RoPE positions. `token` is used only by qwen4exp's PLE;
    /// pass the placeholder id for image tokens. The KV row still advances by one.
    pub fn step_embed(&self, st: &mut State, x: Vec<f32>, rope_pos: [i32; 4], token: u32) -> Vec<f32> {
        self.step_embed_h(st, x, rope_pos, token).0
    }

    pub fn step_embed_h(&self, st: &mut State, mut x: Vec<f32>, rope_pos: [i32; 4], token: u32) -> (Vec<f32>, Vec<f32>) {
        if self.cfg.hc_n > 0 { return self.step_embed_hc(st, x, rope_pos, token); }
        if self.cfg.mhc_streams > 0 { return self.step_embed_mhc(st, x, rope_pos); }
        let kv_pos = st.pos;
        for il in 0..self.cfg.n_layer { self.block(il, &mut x, &mut st.layers[il], kv_pos, rope_pos); }
        st.pos += 1;
        st.rope_pos = [rope_pos[0] + 1; 4];
        (self.head(&x, &self.w.get("output_norm.weight").unwrap().to_f32()), x)
    }

    /// SenseNova-Loop (`sensenova_loop`): shared-weight looped layers with manifold-constrained hyper-connections (mHC)
    /// and 20-step log-space Sinkhorn-Knopp doubly-stochastic stream mixing.
    fn step_embed_mhc(&self, st: &mut State, mut x: Vec<f32>, rope_pos: [i32; 4]) -> (Vec<f32>, Vec<f32>) {
        let c = &self.cfg;
        let kv_pos = st.pos;
        let s = c.mhc_streams;
        let mut r = Vec::new();
        for il in 0..c.n_layer {
            let mk = c.mhc_layer_kinds.get(il).copied().unwrap_or(0);
            if mk == 0 {
                self.block(il, &mut x, &mut st.layers[il], kv_pos, rope_pos);
                continue;
            }
            if mk & 1 != 0 {
                r = x.repeat(s);
            }
            let (mut branch, post, mix) = self.mhc_route(il, "mhc_attn", &r, s);
            let mut h = vec![0f32; c.n_embd];
            ops::rms_norm(&branch, &self.f(il, "attn_norm.weight"), c.eps, &mut h);
            let attn_out = match &mut st.layers[il] {
                LayerState::Recurrent { conv, s: dstate } => self.deltanet(il, &h, conv, dstate),
                LayerState::Attn { k, v, ik } => self.attention(il, &h, k, v, ik, kv_pos, rope_pos),
            };
            Self::mhc_merge(&mut r, &attn_out, &post, &mix, s);
            let (branch_f, post_f, mix_f) = self.mhc_route(il, "mhc_ffn", &r, s);
            branch = branch_f;
            ops::rms_norm(&branch, &self.f(il, "post_attention_norm.weight"), c.eps, &mut h);
            let ffn_out = if c.is_moe_layer(il) { self.moe(il, &h) } else { self.dense_ffn(il, &h) };
            Self::mhc_merge(&mut r, &ffn_out, &post_f, &mix_f, s);
            if mk & 2 != 0 {
                let inv_s = 1.0 / s as f32;
                for i in 0..c.n_embd {
                    let mut sum = 0.0f32;
                    for sc in 0..s { sum += r[sc * c.n_embd + i]; }
                    x[i] = sum * inv_s;
                }
            }
        }
        st.pos += 1;
        st.rope_pos = [rope_pos[0] + 1; 4];
        (self.head(&x, &self.w.get("output_norm.weight").unwrap().to_f32()), x)
    }

    fn mhc_route(&self, il: usize, prefix: &str, r: &[f32], streams: usize) -> (Vec<f32>, Vec<f32>, Vec<f32>) {
        let n = self.cfg.n_embd;
        let sq_mean = r.iter().map(|&v| v * v).sum::<f32>() / (r.len().max(1) as f32);
        let scale = 1.0 / (sq_mean + 1e-6).sqrt();
        let norm: Vec<f32> = r.iter().map(|&v| v * scale).collect();
        let w = self.t(il, &format!("{prefix}.weight"));
        let b = self.f(il, &format!("{prefix}.bias"));
        let mut logits = mv(&w, &norm);
        for (l, &bi) in logits.iter_mut().zip(&b) { *l += bi; }
        let pre: Vec<f32> = logits[..streams].iter().map(|&v| ops::sigmoid(v)).collect();
        let post: Vec<f32> = logits[streams..2 * streams].iter().map(|&v| 2.0 * ops::sigmoid(v)).collect();
        let mix = if logits.len() <= 2 * streams {
            let mut id = vec![0.0f32; streams * streams];
            for i in 0..streams { id[i * streams + i] = 1.0; }
            id
        } else {
            let mut m = logits[2 * streams..2 * streams + streams * streams].to_vec();
            for _ in 0..20 {
                for r0 in 0..streams {
                    let row = &mut m[r0 * streams..(r0 + 1) * streams];
                    let mx = row.iter().copied().fold(f32::NEG_INFINITY, f32::max);
                    let lse = mx + row.iter().map(|&v| (v - mx).exp()).sum::<f32>().ln();
                    for v in row.iter_mut() { *v -= lse; }
                }
                for c0 in 0..streams {
                    let mut mx = f32::NEG_INFINITY;
                    for r0 in 0..streams { mx = mx.max(m[r0 * streams + c0]); }
                    let mut se = 0.0f32;
                    for r0 in 0..streams { se += (m[r0 * streams + c0] - mx).exp(); }
                    let lse = mx + se.ln();
                    for r0 in 0..streams { m[r0 * streams + c0] -= lse; }
                }
            }
            m.into_iter().map(f32::exp).collect()
        };
        let mut branch = vec![0.0f32; n];
        for (sc, &p) in pre.iter().enumerate() {
            for (bi, &rv) in branch.iter_mut().zip(&r[sc * n..(sc + 1) * n]) {
                *bi += p * rv;
            }
        }
        (branch, post, mix)
    }

    fn mhc_merge(r: &mut [f32], branch: &[f32], post: &[f32], mix: &[f32], streams: usize) {
        let n = branch.len();
        let old = r.to_vec();
        for r0 in 0..streams {
            let p = post[r0];
            let dst = &mut r[r0 * n..(r0 + 1) * n];
            for i in 0..n {
                let mut acc = p * branch[i];
                for c0 in 0..streams {
                    acc += mix[r0 * streams + c0] * old[c0 * n + i];
                }
                dst[i] = acc;
            }
        }
    }

    /// qwen4exp: the residual is a stack of hc_n streams (hyper-connections). Every half-layer reads the stack
    /// down to one vector, runs its block and writes the output back into every stream with a per-stream weight.

fn sigmoid(x: f32) -> f32 {
    1.0 / (1.0 + exp(-x))
}

/// e^x to 2 ulp (cephes `expf`) with no branch and no table, so a loop of it compiles to vector code. libm's `exp`
/// and `tanh` are calls: about 4 ms per token in the activation and the loop controllers.
#[inline(always)]
fn exp(x: f32) -> f32 {
    let x = x.clamp(-87.0, 88.0);
    let k = (x * std::f32::consts::LOG2_E).round_ties_even();
    let r = k.mul_add(2.121_944_4e-4, k.mul_add(-0.693_359_375, x));
    let mut p = 1.987_569_1e-4f32;
    for c in [1.398_2e-3, 8.333_452e-3, 4.166_579_6e-2, 1.666_666_5e-1, 0.5] { p = p.mul_add(r, c); }
    ((p * r).mul_add(r, r) + 1.0) * f32::from_bits(((k as i32 + 127) as u32) << 23)
}

/// tanh from `exp`: an absolute error of 1.2e-7.
#[inline(always)]
fn tanh(x: f32) -> f32 {
    let t = exp(2.0 * x.clamp(-10.0, 10.0));
    (t - 1.0) / (t + 1.0)
}

/// Scale each row of width `w` down to an L2 norm of at most `thr`, after the optional per-channel scale.
fn clip(x: &mut [f32], scale: Option<&[f32]>, thr: f32, w: usize) {
    if let Some(s) = scale {
        for (x, s) in x.iter_mut().zip(s) { *x *= s; }
    }
    for row in x.chunks_exact_mut(w) {
        let f = (thr / (dot(row, row).sqrt() + 1e-6)).clamp(0.0, 1.0);
        for v in row { *v *= f; }
    }
}

/// Attention over the depth: for each query row, the softmax-weighted sum of the normalised states. The states are
/// the earlier block outputs (the newest first, as llama.cpp stacks them) and the current stream.
fn mix(blocks: &[Vec<f32>], cur: &[f32], query: &[f32]) -> Vec<f32> {
    let n = cur.len();
    let states: Vec<Vec<f32>> = blocks.iter().rev().map(|b| b.as_slice()).chain([cur]).map(|s| {
        let mut o = vec![0f32; n];
        rms_norm(s, MIX_EPS, &mut o);
        o
    }).collect();
    let scale = 1.0 / (n as f32).sqrt();
    let mut out = vec![0f32; query.len()];
    for (q, o) in query.chunks_exact(n).zip(out.chunks_exact_mut(n)) {
        let mut sc: Vec<f32> = states.iter().map(|s| dot(s, q) * scale).collect();
        let mx = sc.iter().cloned().fold(f32::NEG_INFINITY, f32::max);
        for s in sc.iter_mut() { *s = (*s - mx).exp(); }
        let z: f32 = sc.iter().sum();
        for (s, p) in states.iter().zip(&sc) {
            for (o, s) in o.iter_mut().zip(s) { *o += p / z * s; }
        }
    }
    out
}

/// Causal short convolution with a residual. `hist` holds the last `n_conv - 1` inputs and is advanced by `cur`.
/// `taps` (`[head][lag]`, lag 0 is the current input) adds input-dependent taps shared by the channels of a head.
fn convolve(cur: &[f32], w: &[f32], taps: Option<&[f32]>, hist: &mut Vec<f32>, n_conv: usize) -> Vec<f32> {
    let n = cur.len();
    hist.resize((n_conv - 1) * n, 0.0);
    let x = |k: usize, c: usize| if k + 1 < n_conv { hist[k * n + c] } else { cur[c] };
    let per = taps.map_or(n, |t| n / (t.len() / n_conv));
    let s = 1.0 / ((if taps.is_some() { 2 * n_conv + 1 } else { n_conv + 1 }) as f32).sqrt();
    let out = (0..n).map(|c| {
        let mut o: f32 = (0..n_conv).map(|k| x(k, c) * w[c * n_conv + k]).sum();
        if let Some(t) = taps {
            for lag in 0..n_conv { o += x(n_conv - 1 - lag, c) * t[c / per * n_conv + lag]; }
        }
        (cur[c] + o) * s
    }).collect();
    hist.copy_within(n.., 0);
    hist[(n_conv - 2) * n..].copy_from_slice(cur);
    out
}

/// `convolve` of a chunk of tokens at once: token t reads the inputs of the tokens before it, those of the chunk
/// and (in `hist`) those before the chunk.
fn convolve_rows(rows: &[&[f32]], w: &[f32], taps: Option<&[&[f32]]>, hist: &mut Vec<f32>, n_conv: usize) -> Vec<Vec<f32>> {
    if let [row] = rows { return vec![convolve(row, w, taps.map(|t| t[0]), hist, n_conv)]; }
    let n = rows[0].len();
    hist.resize((n_conv - 1) * n, 0.0);
    let mut all = std::mem::take(hist);
    for r in rows { all.extend_from_slice(r); }
    let out = (0..rows.len()).into_par_iter().map(|t| {
        convolve(rows[t], w, taps.map(|x| x[t]), &mut all[t * n..(t + n_conv - 1) * n].to_vec(), n_conv)
    }).collect();
    *hist = all[rows.len() * n..].to_vec();
    out
}

/// Sum of the eight values, as a tree.
#[cfg(all(target_feature = "f16c", target_feature = "fma"))]
#[inline(always)]
fn hsum(v: std::arch::x86_64::__m256) -> f32 {
    use std::arch::x86_64::*;
    // SAFETY: the features are those of the build.
    unsafe {
        let s = _mm_add_ps(_mm256_castps256_ps128(v), _mm256_extractf128_ps(v, 1));
        let s = _mm_add_ps(s, _mm_movehl_ps(s, s));
        _mm_cvtss_f32(_mm_add_ss(s, _mm_shuffle_ps(s, s, 1)))
    }
}

/// `p[j * n + row]` += `scale` times the dot of f16 row `row` of `w` (`n` rows) with row j of `x`.
/// One to three rows of `x` take the same sums in the same order, so a token gets the same bits alone or in a chunk.
pub(crate) fn dots_h(p: &mut [f32], scale: f32, w: &[f16], x: &[f32], n: usize) {
    let hd = w.len() / n;
    #[cfg(all(target_feature = "f16c", target_feature = "fma"))]
    if hd % 32 == 0 && x.len() % hd == 0 && p.len() == x.len() / hd * n {
        match x.len() / hd {
            1 => return dots_k::<1>(p, scale, w, x, n),
            2 => return dots_k::<2>(p, scale, w, x, n),
            3 => return dots_k::<3>(p, scale, w, x, n),
            _ => {}
        }
    }
    for (a, w) in w.chunks_exact(hd).enumerate() {
        for (j, x) in x.chunks_exact(hd).enumerate() { p[j * n + a] += scale * dot_h(w, x); }
    }
}

/// `dots_h` for `K` rows of `x`: a weight is widened once for `K` products, and the sums stay in registers.
#[cfg(all(target_feature = "f16c", target_feature = "fma"))]
fn dots_k<const K: usize>(p: &mut [f32], scale: f32, w: &[f16], x: &[f32], n: usize) {
    use std::arch::x86_64::*;
    let hd = w.len() / n;
    assert!(x.len() == K * hd && p.len() == K * n && hd % 32 == 0);
    for a in 0..n {
        unsafe {
            let mut acc = [[_mm256_setzero_ps(); 4]; K];
            for c in (0..hd).step_by(32) {
                for b in 0..4 {
                    let k = _mm256_cvtph_ps(_mm_loadu_si128(w.as_ptr().add(a * hd + c + 8 * b).cast()));
                    for j in 0..K { acc[j][b] = _mm256_fmadd_ps(k, _mm256_loadu_ps(x.as_ptr().add(j * hd + c + 8 * b)), acc[j][b]); }
                }
            }
            for j in 0..K { p[j * n + a] += scale * hsum(_mm256_add_ps(_mm256_add_ps(acc[j][0], acc[j][1]), _mm256_add_ps(acc[j][2], acc[j][3]))); }
        }
    }
}

/// Dot of f16 weights and f32 values. The loop controllers are 113 MB of F32 in the file and every token reads them
/// twice, at the speed of the memory: as f16 they take half the time (their values stay within 5e-4 relative).
pub(crate) fn dot_h(w: &[f16], x: &[f32]) -> f32 {
    #[cfg(all(target_feature = "f16c", target_feature = "fma"))]
    {
        use std::arch::x86_64::*;
        let (cw, cx) = (w.chunks_exact(32), x.chunks_exact(32));
        let tail: f32 = cw.remainder().iter().zip(cx.remainder()).map(|(w, x)| w.to_f32() * x).sum();
        // Four sums: one would wait on its own multiply-add.
        let mut acc = [unsafe { _mm256_setzero_ps() }; 4];
        for (w, x) in cw.zip(cx) {
            for i in 0..4 {
                // SAFETY: both chunks hold 32 values, and the features are those of the build.
                acc[i] = unsafe { _mm256_fmadd_ps(_mm256_cvtph_ps(_mm_loadu_si128(w.as_ptr().add(8 * i).cast())), _mm256_loadu_ps(x.as_ptr().add(8 * i)), acc[i]) };
            }
        }
        let s: [f32; 8] = unsafe { std::mem::transmute(_mm256_add_ps(_mm256_add_ps(acc[0], acc[1]), _mm256_add_ps(acc[2], acc[3]))) };
        s.iter().sum::<f32>() + tail
    }
    #[cfg(not(all(target_feature = "f16c", target_feature = "fma")))]
    w.iter().zip(x).map(|(w, x)| w.to_f32() * x).sum()
}

/// o += p * w, for f16 w.
fn axpy_h(o: &mut [f32], p: f32, w: &[f16]) {
    #[cfg(all(target_feature = "f16c", target_feature = "fma"))]
    {
        use std::arch::x86_64::*;
        let (co, cw) = (o.chunks_exact_mut(8), w.chunks_exact(8));
        let tail = cw.remainder();
        for (o, w) in co.zip(cw) {
            // SAFETY: both chunks hold 8 values, and the features are those of the build.
            unsafe { _mm256_storeu_ps(o.as_mut_ptr(), _mm256_fmadd_ps(_mm256_cvtph_ps(_mm_loadu_si128(w.as_ptr().cast())), _mm256_set1_ps(p), _mm256_loadu_ps(o.as_ptr()))) };
        }
        let at = o.len() - tail.len();
        for (o, w) in o[at..].iter_mut().zip(tail) { *o += p * w.to_f32(); }
    }
    #[cfg(not(all(target_feature = "f16c", target_feature = "fma")))]
    for (o, w) in o.iter_mut().zip(w) { *o += p * w.to_f32(); }
}

/// Row j of `o` += the sum over the `n` f16 rows of `w` of `p[j * n + row]` times that row. One to three rows of
/// `o` take the same sums in the same order.
pub(crate) fn mix_h(o: &mut [f32], p: &[f32], w: &[f16], n: usize) {
    let hd = w.len() / n;
    #[cfg(all(target_feature = "f16c", target_feature = "fma"))]
    if hd % 32 == 0 && o.len() % hd == 0 && p.len() == o.len() / hd * n {
        match o.len() / hd {
            1 => return mix_k::<1>(o, p, w, n),
            2 => return mix_k::<2>(o, p, w, n),
            3 => return mix_k::<3>(o, p, w, n),
            _ => {}
        }
    }
    for (a, w) in w.chunks_exact(hd).enumerate() {
        for (j, o) in o.chunks_exact_mut(hd).enumerate() { axpy_h(o, p[j * n + a], w); }
    }
}

/// `mix_h` for `K` rows of `o`, 32 columns at a time in registers.
#[cfg(all(target_feature = "f16c", target_feature = "fma"))]
fn mix_k<const K: usize>(o: &mut [f32], p: &[f32], w: &[f16], n: usize) {
    use std::arch::x86_64::*;
    let hd = w.len() / n;
    assert!(o.len() == K * hd && p.len() == K * n && hd % 32 == 0);
    for c in (0..hd).step_by(32) {
        unsafe {
            let mut acc = [[_mm256_setzero_ps(); 4]; K];
            for j in 0..K { for b in 0..4 { acc[j][b] = _mm256_loadu_ps(o.as_ptr().add(j * hd + c + 8 * b)); } }
            for a in 0..n {
                let pj: [__m256; K] = std::array::from_fn(|j| _mm256_broadcast_ss(&p[j * n + a]));
                for b in 0..4 {
                    let v = _mm256_cvtph_ps(_mm_loadu_si128(w.as_ptr().add(a * hd + c + 8 * b).cast()));
                    for j in 0..K { acc[j][b] = _mm256_fmadd_ps(v, pj[j], acc[j][b]); }
                }
            }
            for j in 0..K { for b in 0..4 { _mm256_storeu_ps(o.as_mut_ptr().add(j * hd + c + 8 * b), acc[j][b]); } }
        }
    }
}

/// Mix a cross-pass message into a looped layer input: `(1 - forget)*cur + gain*message`, both gated on the two.
/// Controller `c` on a run of tokens, `jobs` = (state, message). Three tokens share a pass over the matrices
/// (`dots_h`, `mix_h`): a weight is widened from f16 once for three products.
fn control(jobs: &[(&[f32], &[f32])], lp: &Loop, c: usize) -> Vec<Vec<f32>> {
    let n = jobs[0].0.len();
    let rank = lp.down[c].len() / (2 * n);
    jobs.par_chunks(3).flat_map_iter(|js| {
        let mut cat = vec![0f32; js.len() * 2 * n];
        for (cat, &(cur, msg)) in cat.chunks_exact_mut(2 * n).zip(js) {
            rms_norm(cur, MIX_EPS, &mut cat[..n]);
            rms_norm(msg, MIX_EPS, &mut cat[n..]);
        }
        let (mut gate, mut g, mut f) = (vec![0f32; js.len() * rank], vec![0f32; js.len() * n], vec![0f32; js.len() * n]);
        dots_h(&mut gate, 1.0, &lp.down[c], &cat, rank);
        mix_h(&mut g, &gate, &lp.gain[c], rank);
        mix_h(&mut f, &gate, &lp.forget[c], rank);
        js.iter().zip(g.chunks_exact(n).zip(f.chunks_exact(n))).map(|(&(cur, msg), (g, f))| {
            (0..n).map(|i| cur[i] - tanh(f[i]) * cur[i] + tanh(g[i]) * msg[i]).collect::<Vec<f32>>()
        }).collect::<Vec<_>>()
    }).collect()
}

/// situ(gate, up) = beta*tanh(gate/beta)*sigmoid(gate) * lb*tanh(up/lb) on one output of the fused gate_up matrix
/// (the gate rows first), both times the expert's gate_up scale `s`.
fn situ(y: &[f32], s: f32, cfg: &Inkling2Config) -> Vec<f32> {
    let (nf, beta, lb) = (cfg.n_ff_exp, cfg.situ_beta, cfg.situ_linear_beta);
    let (g, u, mut o) = (&y[..nf], &y[nf..2 * nf], vec![0f32; nf]);
    for i in 0..nf {
        let (g, u) = (g[i] * s, u[i] * s);
        let up = if lb > 0.0 { lb * tanh(u / lb) } else { u };
        o[i] = beta * tanh(g / beta) * sigmoid(g) * up;
    }
    o
}

/// The experts of one token and their weights: sigmoid router, the selection bias only picks the experts. The
/// weight carries the expert's down scale.
fn route(l: &Layer, x: &[f32], cfg: &Inkling2Config) -> Vec<(usize, f32)> {
    let probs: Vec<f32> = l.gate_inp.chunks_exact(x.len()).map(|w| sigmoid(dot(w, x))).collect();
    let mut sel: Vec<usize> = (0..cfg.n_expert).collect();
    sel.sort_by(|&a, &b| (probs[b] + l.bias[b]).partial_cmp(&(probs[a] + l.bias[a])).unwrap().then(a.cmp(&b)));
    sel.truncate(cfg.n_used);
    let sum = sel.iter().map(|&e| probs[e]).sum::<f32>() + 1e-20;
    sel.into_iter().map(|e| (e, probs[e] / sum * cfg.exp_scale * l.down_scale[e])).collect()
}

/// Weighted sum of the listed experts' outputs, `(gate_up, down, weight, gate_up scale)` each. Experts that share a GPU run as two
/// batched round trips; otherwise each matvec runs where its matrix lives.
fn run_experts(list: &[(Mat, Mat, f32, f32)], x: &[f32], cfg: &Inkling2Config) -> Vec<f32> {
    let (nf, nl) = (cfg.n_ff_exp, x.len());
    let mut mix = vec![0f32; nl];
    let batched = (!list.is_empty() && list.iter().all(|e| e.0.gpu.is_some())).then(|| {
        let (gus, downs): (Vec<Mat>, Vec<Mat>) = list.iter().map(|e| (e.0, e.1)).unzip();
        let gs: Vec<f32> = list.iter().map(|e| e.3).collect();
        // One round trip with the activation on the device; two, with the activation here, on more than one GPU.
        situ_experts(&gus, &downs, &gs, x, cfg.situ_beta, cfg.situ_linear_beta).or_else(|| {
            let y = matvec_many_opt(&gus, x, 0)?;
            let a: Vec<f32> = y.chunks_exact(2 * nf).zip(&gs).flat_map(|(y, &s)| situ(y, s, cfg)).collect();
            matvec_many_opt(&downs, &a, nf)
        })
    }).flatten();
    match batched {
        Some(dy) => {
            for (e, dy) in list.iter().zip(dy.chunks_exact(nl)) {
                for (m, d) in mix.iter_mut().zip(dy) { *m += e.2 * d; }
            }
        }
        None => {
            let gus: Vec<Mat> = list.iter().map(|e| e.0).collect();
            let downs: Vec<Mat> = list.iter().map(|e| e.1).collect();
            let gs: Vec<f32> = list.iter().map(|e| e.3).collect();
            let gx: Vec<&[f32]> = vec![x; list.len()];
            if let Some(y) = matvec_many_cpu(&gus, &gx) {
                let a: Vec<f32> = y.chunks_exact(2 * nf).zip(&gs).flat_map(|(y, &s)| situ(y, s, cfg)).collect();
                let acts: Vec<&[f32]> = a.chunks_exact(nf).collect();
                if let Some(dy) = matvec_many_cpu(&downs, &acts) {
                    for (e, dy) in list.iter().zip(dy.chunks_exact(nl)) {
                        for (m, d) in mix.iter_mut().zip(dy) { *m += e.2 * d; }
                    }
                    return mix;
                }
            }
            let (mut y, mut dy) = (vec![0f32; 2 * nf], vec![0f32; nl]);
            for (gu, down, w, s) in list {
                gu.matvec(x, &mut y);
                down.matvec(&situ(&y, *s, cfg), &mut dy);
                for (m, d) in mix.iter_mut().zip(&dy) { *m += w * d; }
            }
        }
    }
    mix
}

    fn rope_fr(&self, base: f32, n_rot: usize) -> anyhow::Result<u64> {
        let mut fr = self.fr.lock().unwrap();

        if self.capturing() {
            fr.1.push((base, n_rot));

        }
        let buf = alloc_raw(&self.hip, 8 * (n_rot / 2).max(1))?;
        let p = buf.addr();
        let (bs, nr) = (base, n_rot as i32);
        launch!(self, "rope_freq", cfg((1, 1), 256, 0), p, bs, nr);
        self.wait()?;
        fr.0.insert((base.to_bits(), n_rot), buf);
        Ok(p)
    }

    /// `gemvq`; with z = 2 also the matrix of `pair.2` rows at byte offset `pair.0` into the output at byte offset
    /// `pair.1` (token stride `pair.2`).
    fn gemvq_pair(&self, w: &GpuTensor, x: u64, x_stride: usize, xq: u64, y: u64, y_stride: usize, nt: usize,
                  pair: (i64, i64, i32, f32), z: u32) -> anyhow::Result<()> {
        if !(1..=4).contains(&nt) { bail!("gemvn: {nt} inputs (1 to 4)"); }
        let (ncols, k1) = Self::gemv_cols("gemv", w.ty, w.cols)?;
        let (mut c, rpw) = Self::gemv_cfg(w.ty, w.rows, w.cols, 1);
        let per = c.shared_mem_bytes;
        c.shared_mem_bytes = if xq == 0 { per * nt as u32 } else { 0 };
        c.grid_dim.2 = z;
        // gemvn caches one x row per token in shared memory; when that exceeds the block's limit, run a token at a time.
        if xq == 0 && c.shared_mem_bytes as usize > self.max_smem {
            c.shared_mem_bytes = per;
            for t in 0..nt {
                let (xt, yt) = (x + 4 * t as u64 * x_stride as u64, y + 4 * t as u64 * y_stride as u64);
                launch!(self, &k1, c, w.ptr, xt, yt, w.rows as i32, ncols, rpw, xq, pair.0, pair.1, pair.2, w.scale, pair.3);
            }
            return Ok(());
        }
        if nt == 1 {
            launch!(self, &k1, c, w.ptr, x, y, w.rows as i32, ncols, rpw, xq, pair.0, pair.1, pair.2, w.scale, pair.3);
            return Ok(());
        }
        let kn = k1.replacen("gemv", "gemvn", 1);
        launch!(self, &kn, c, w.ptr, x, x_stride as i32, y, y_stride as i32, w.rows as i32, ncols, rpw, nt as i32, xq, pair.0, pair.1, pair.2, w.scale, pair.3);
        Ok(())
    }

    /// Prefill GEMM over a tile list (or one dense matrix when `tiles` is 0). Without int8 tensor cores the f16
    /// `gemm_*`/`gemms_*` kernels run; `gemm_tc` is compiled out of this build (`GEMM_TC 0`).
    fn gemm_raw(&self, ty: GgmlType, rows: usize, cols: usize, tiles: u64, n_tiles: usize, wd: u64, nd: usize,
                x: u64, x_stride: usize, _n_x: usize, xrow: u64, y: u64, y_stride: usize, act: i32, w_scale: f32) -> anyhow::Result<()> {
        if cols % 64 != 0 { bail!("{} matrix with {cols} columns: GEMM needs a multiple of 64", ty.name()); }
        if n_tiles == 0 { return Ok(()) }
        // Only the activation reaches the kernel: SAME_X and BATCHED (the same kernels here) are flags for the host.
        let (same, act) = (act & SAME_X != 0, act & 3);
        let _ = same;
        let grid = (rows.div_ceil(64) as u32, n_tiles as u32);
        let sm = tiles == 0 && (grid.0 * grid.1 < 112 || nd <= 32);
        let grid = if sm { (grid.0, nd.div_ceil(32) as u32) } else { grid };
        launch!(self, &kname(if sm { "gemms" } else { "gemm" }, ty), cfg(grid, 2 * GEMM_BM as u32, 0),
                tiles, wd, nd as i32, x, x_stride as i32, xrow, y, y_stride as i32, rows as i32, (cols / 64) as i32, act, w_scale);
        // Tile lists: gemm_* skips the tiles of at most 32 positions, which gemms_* takes.
        if tiles != 0 {
            launch!(self, &kname("gemms", ty), cfg(grid, 2 * GEMM_BM as u32, 0),
                    tiles, wd, nd as i32, x, x_stride as i32, xrow, y, y_stride as i32, rows as i32, (cols / 64) as i32, act, w_scale);
        }
        Ok(())
    }

    /// Decode attention: `attn_decode_gq` plus `attn_merge`, which also writes `out` quantized into xq.
    fn attn(&self, q: u64, kc: u64, vc: u64, qg: u64, out: u64, part: u64, step: u64, nh: usize, nkv: usize, hd: usize,
            scale: f32, gq_min: i32, nt: usize, xq: u64, sel: u64, sst: usize, sink: u64, win: usize, kvt: u64, ns1: usize) -> anyhow::Result<()> {
        if ![64, 128, 256, 512].contains(&hd) { bail!("attention head_dim {hd} unsupported (64, 128, 256 or 512)"); }
        let group = nh / nkv;
        let g = [8, 4, 2, 1].into_iter().find(|&g| group % g == 0 && g * hd <= 2048).unwrap();
        let ns = if kvt != 0 { ns1 } else { ATT_SPLITS };
        let pz = (nh * ns * (hd + 2)) as i32;
        if gq_min != 0 {
            launch!(self, &format!("attn_decode_{hd}"), cfg(((nh * nt) as u32, ns1 as u32), 256, (8 * hd * 4) as u32),
                    q, kc, vc, part, step, nh as i32, nkv as i32, hd as i32, scale, gq_min, pz, nt as i32, sel, sst as i32, sink, win as i32, kvt);
        }
        // No cp.async double buffering here: that path is the sm_80+ one (`ATT_RUN_CP`), so the K/V tile comes
        // through registers and the kernel needs no extra shared memory.
        let g4 = false;
        let (name, cp) = if g4 { ("attn_decode_gqa4_128".to_string(), 0) } else { (format!("attn_decode_gqa_{hd}"), 0) };
        if gq_min != i32::MAX {
            launch!(self, &name, cfg(((nh / g * nt) as u32, ns as u32), 256, (8 * hd * 4 + cp) as u32),
                    q, kc, vc, part, step, nh as i32, nkv as i32, hd as i32, scale, gq_min, pz, nt as i32, sel, sst as i32, sink, win as i32, kvt);
        }
        launch!(self, "attn_merge", LaunchConfig { grid_dim: (nh as u32, 1, nt as u32), block_dim: (256, 1, 1), shared_mem_bytes: 0 },
                part, qg, out, hd as i32, ns as i32, step, ns1 as i32, gq_min, xq, sink);
        Ok(())
    }

    fn grid(n: usize) -> LaunchConfig { cfg((n.div_ceil(256) as u32, 1), 256, 0) }
}

/// Pinned host allocation freed with the runtime that made it.
struct PinnedHost { hip: Arc<ffi::Hip>, p: *mut c_void }

unsafe impl Send for PinnedHost {}
unsafe impl Sync for PinnedHost {}

impl Drop for PinnedHost {
    fn drop(&mut self) { unsafe { (self.hip.host_free)(self.p) }; }
}

/// Pinned host memory used as a ring for `htod`: a region is free again once the copy that read it has completed.
struct Staging {
    /// Owns the pinned allocation; dropping it frees the ring.
    _pin: Buf,
    host: *mut u8,
    head: usize,
    busy: VecDeque<(usize, usize, Event)>,
}

/// A `hipEvent_t` that destroys itself.
struct Event { hip: Arc<ffi::Hip>, ev: ffi::Event }

impl Event {
    fn new(hip: &Arc<ffi::Hip>) -> anyhow::Result<Event> {
        let mut ev: ffi::Event = std::ptr::null_mut();
        hip.check(unsafe { (hip.event_create)(&mut ev, ffi::EVENT_DISABLE_TIMING) }, "hipEventCreate")?;
        Ok(Event { hip: hip.clone(), ev })
    }
    fn record(&self, stream: ffi::Stream) -> anyhow::Result<()> {
        self.hip.check(unsafe { (self.hip.event_record)(self.ev, stream) }, "hipEventRecord")
    }
}

impl Drop for Event {
    fn drop(&mut self) { unsafe { (self.hip.event_destroy)(self.ev) }; }
}

/// Recorded graphs kept before a forward runs uncaptured (CUDA's cap, same reason).
const GRAPHS_MAX: usize = 32;

const RING: usize = 64 << 20;
const RING_MIN: usize = 256 << 10;

fn alloc_raw(hip: &Arc<ffi::Hip>, bytes: usize) -> anyhow::Result<Buf> {
    let mut p: *mut c_void = std::ptr::null_mut();
    hip.check(unsafe { (hip.malloc)(&mut p, bytes.max(1)) }, "hipMalloc")?;
    // The Buf owns the allocation through its address; the free happens when the owner (the backend or the tensor)
    // drops, so the owner is a small RAII handle.
    struct DevAlloc { hip: Arc<ffi::Hip>, p: *mut c_void }
    unsafe impl Send for DevAlloc {}
    unsafe impl Sync for DevAlloc {}
    impl Drop for DevAlloc {
        fn drop(&mut self) { unsafe { (self.hip.free)(self.p) }; }
    }
    Ok(Buf::new(p as u64, bytes.max(1), DevAlloc { hip: hip.clone(), p }))
}

fn compile_cached(rtc: &ffi::Rtc, src: &str, arch: &str, tag: &str) -> anyhow::Result<Vec<u8>> {
    let dir = std::env::var_os("XDG_CACHE_HOME").map(Into::into)
        .unwrap_or_else(|| std::path::PathBuf::from(std::env::var_os("HOME").unwrap_or_default()).join(".cache"))
        .join("rusting_strata");
    let path = dir.join(format!("{:016x}.amdgcn", cache_key(src, arch, tag)));
    if let Ok(b) = std::fs::read(&path) { return Ok(b) }
    eprintln!("compiling HIP kernels for {arch} with hiprtc {tag} (first run only)");
    let b = rtc.compile(src, "kernels.cu", arch, &[])?;
    let _ = std::fs::create_dir_all(&dir).and_then(|_| std::fs::write(&path, &b));
    Ok(b)
}
